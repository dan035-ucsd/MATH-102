\begin{enumerate}[(a)]
    \item Given $r(x,y) = \sqrt{x^2+y^2} = (x^2+y^2)^\frac{1}{2}$
    \begin{align*}
        r_x &= \frac{1}{2}(x^2+y^2)^{-\frac{1}{2}}\cdot 2x \\
        &= x(x^2+y^2)^{-\frac 12} \\
        &= \frac xr \\
        r_{xx} &= \frac 1r - \frac 12 x(x^2+y^2)^{-\frac 32} \cdot 2x \\
        &= \frac 1r - x^2(x^2+y^2)^{-\frac 32} \\
        &= \frac 1r - \frac {x^2}{r^3} \\
        &= \frac{r^2-x^2}{r^3} \\
        &= \frac{y^2}{r^3}\\
        r_y &= \frac yr \\ 
        r_{yy} &= \frac{x^2}{r^3}
    \end{align*}
    \item Given $u(x,y) = f(r(x,y))$
    \begin{align*}
        \nabla^2 u &= u_{xx} + u_{yy} 
    \end{align*}
    Find $f_{xx}$ and $f_{yy}$
    \begin{align*}
        f_x &= f'(r)r_x \\
        f_{xx} &= f''(r)r_x + f'(r)r_{xx} \\
        &= f''(r)\frac xr + f'(r)\frac{y^2}{r^3} \\
        &= \frac 1r \left(f''(r)x + f'(r)\frac{y^2}{x^2 + y^2}\right) \\
        f_{yy} &= \frac 1r \left(f''(r)y + f'(r)\frac{x^2}{x^2 + y^2}\right)
    \end{align*}
    So we have
    \begin{align*}
        \nabla^2 u &= \frac 1r \left(f''(r)x + f'(r)\frac{y^2}{x^2 + y^2} + f''(r)y + f'(r)\frac{x^2}{x^2 + y^2} \right) \\
        &= \frac 1r \left( f''(r)(x+y)+f'(r)\right) \\
        &= f''(r) \frac{x+y}{r} + \frac 1r f'(r)
    \end{align*}
    \item So we have the ODE $\displaystyle f''(r) + \frac 1r f'(r) = 0$
    \begin{align*}
        f''(r) &= - \frac 1r f'(r) \\
        \frac{d}{dr} f'(r) &= - \frac 1r f'(r)
    \end{align*}
    Let $u = f'(r)$
    \begin{align*}
        \frac{du}{dr} &= -\frac 1r u \\
        \frac 1u du &= -\frac 1r dr \\
        \ln u &= - \ln r + c \\
        u &= \frac{C}{r} \\
        f'(r) &= \frac Cr \\
        \frac{d}{dr} f(r) &= \frac Cr \\
        f(r) &= C\ln|r| + B
    \end{align*}
\end{enumerate}
