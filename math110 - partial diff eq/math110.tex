\section{First Order PDEs}
\subsection{Lecture 1 - Introduction to PDEs}
\textbf{What is a ODE?} Recall that an ordinary differential equation (ODE) are 
equations for functions of a single variable, $u(t),\ t \in \mathbb{R}$
\begin{align*}
    \frac{du}{dt} = u (1-u) \ \text{ is an ODE}
\end{align*}
Here we call $t$ the independent variable and $u$ the state/dependent variable.
In general, an ODE has the form $F(t) = (t,u,\frac{du}{dt},\frac{d^2u}{dt^2},\dots)$ 
\\ \\
Note that the state variable can have many components.
\begin{align*}
    \mathbf{u}(t) = \begin{bmatrix}
        u_1(t) \\ u_2(t)
    \end{bmatrix} \in \mathbb{R}^2
\end{align*}
then 
\begin{align*}
    \frac{du}{dt} = \mathbf{u}\ \text{ is an ODE}
\end{align*}
\textbf{What is a PDE?} A PDE is an equation for a function with more than 1 
independent variable, e.g. $u(x,y)$.
\begin{align*}
    \text{Notation: } & u_x = \frac{\partial u}{\partial x} = \partial_x u \\
    & u_y = \frac{\partial u}{\partial y} = \partial_y u \\ 
    & \text{etc.}
\end{align*}
\underline{Examples of PDEs}
\begin{itemize}
    \item Heat/diffusion equation for $u(x,t)$ \begin{align*}
        u_t = ku_{xx} \quad k \text{ constant}
    \end{align*}
    \item Wave equation for $u(x,t)$ \begin{align*}
        u_{tt} = c^2u_{xx} \quad c \text{ constant}
    \end{align*}
    \item Laplace's equation for $u(x,y)$ \begin{align*}
        u_{xx} + u_{yy} = 0
    \end{align*}
\end{itemize}
\boxed{\text{ex}} Consider the heat flow in a bar of length $l$ 
\begin{itemize}
    \item $u(x,t)$ is the temperature of the bar at position $x$, time $t$
    \item Suppose the ends are held at temperature 0, and the initial temperature
    is $\phi(x)$
    \item The temperature evolves according to the heat equation $u_t = ku_{xx}$
    where $t \ge 0, 0 \le x \le l$, $k $ constant.
\end{itemize}
We also have:
\begin{itemize}
    \item Boundary conditions (BCs): 
    \begin{align*}
        u(0,t) = u(l,t) = 0\quad \forall t
    \end{align*}
    \item Initial conditions (ICs):
    \begin{align*}
        u(x,0) = \phi(x)
    \end{align*}
\end{itemize}
\underline{Remarks}
\begin{itemize}
    \item The order of the PDE is the order of the highest derivative that occurs.
    \item The heat equation above is $2^{\text{nd}}$ order
    \item A general $2^{\text{nd}}$ order PDE for $u(x,t)$ is: 
    \begin{align*}
        G(x,t,u,u_x,u_t,u_{xx},u_{tt},u_{xt}) = 0
    \end{align*}
    \item A PDE in the above form is \underline{linear} if $G$ is a linear function of 
    $u$ and all of its derivatives. It is nonlinear otherwise.
    \item A linear PDE is \underline{homogeneous} if  every term contains $u$ or some 
    derivative of $u$. It is nonhomogeneous/inhomogeneous otherwise.
\end{itemize}
\boxed{\text{ex}} Consider the linearity and homogeneity of the following equations.
\begin{enumerate}
    \item $u_t + e^{x^2t}u_{xt} = 0$ \\
    is a linear and homogeneous PDE.
    \item $u_t + 3xu_{xx} = \frac tx$ \\
    is a linear and inhomogeneous PDE because of the $\frac tx$ term.
    \item $u_t + uu_x = 0$ \\
    is a nonlinear PDE because of the $uu_x$ term.
    \item $u_{tt} + u_{x} + sin(u) = 0$ \\
    is nonlinear because of the $sin(u)$ term.
\end{enumerate}
\underline{Operating notation \& linearity} \\ 
We can write the heat equation $u_t - ku_{xx} = 0$ as $Lu = 0$ where 
\begin{align*}
    L = (\partial_t - k\partial_x^2)
\end{align*}
We call $L$ the differential operator. $L$ acts on (smooth) functions $u(x,t)$ to produce
a new function.
\begin{align*}
    L(u) = Lu &= (\delta_t - k\partial_x^2)u \\
    &= u_t - ku_{xx}
\end{align*}
$L$ is a \underline{linear} operator iff 
\begin{align*}
    L(u+v) &= Lu + Lv \\
    &\ \forall \text{ functions } u,v
\end{align*}
and 
\begin{align*}
    L(cu) &= cLu \\
    &\ \forall \text{ constants } c \in \mathbb{R} \\
    &\ \forall \text{ functions } u
\end{align*}
\boxed{\text{ex}} Check if the heat equation has a linear differential operator.
\begin{enumerate}
    \item \begin{align*}
        L(u + v) &= (\partial_t + k\partial^2_x)(u+v) \\
        &= u_t + ku_{xx} + v_t + k v_{xx} \\
        &= Lu + Lv \quad \checkmark
    \end{align*}
    \item \begin{align*}
        L(cu) &= cu_t + cku_{xx} \\
        &= cLu \text{ for constant } c \quad \checkmark
    \end{align*}
\end{enumerate}
\underline{Remarks} 
\begin{itemize}
    \item If $L$ is a linear operator then $Lu = f$ is a linear PDE
    \begin{align*}
        \underbrace{u_t + ku_{xx}}_{Lu} = \underbrace{e^{x^2t}}_{f(x,t)} \text{ is linear}
    \end{align*}
    \item If $L$ is a linear operator then the PDE $Lu = f$ is 
    \begin{itemize}
        \item homogeneous if $f = 0$
        \item inhomogeneous if $f \ne 0$ 
    \end{itemize}
    \item A linear, homogeneous PDE $Lu = 0$ has the superposition principle
    \item If $u_1, u_2$ are solutions to the PDE then $au_1 + bu_2$ is a solution for 
    constants $a,b$.
    \begin{align*}
        Lu_1 = Lu_2 = 0 \\
        \Rightarrow L(au_1 + bu_2) = 0
    \end{align*}
    \item The distinction between linear \& nonlinear PDEs is very important.
\end{itemize}
\underline{Solutions to PDEs}
\begin{itemize}
    \item A solution to a PDE is a function $u$ in which satisfies the PDE and any BCs/ICs upon
    direct substitution.
    \item The general solution to an ODE involves arbitrary constants. The general solution to 
    a PDE involves arbitrary functions.
    \item Typically, we are interested in specific solutions.
\end{itemize}
\boxed{\text{ex}} Consider the heat equation   $u_t - u_{xx} = 0$.
\begin{align*}
    u_1(x,t) &= x^2 + 2t \text{ is a soln} \\
    \rightarrow \partial_t u_1 - \partial_x^2 u_1 &= 2-2 = 0 \\
    u_2(x,t) &= e^{-t} \sin x \text{ is a soln} \\
    \rightarrow \partial_t u_2 - \partial^2_xu_2 &= -e^{-t}\sin x + e^{-t} \sin x = 0
\end{align*}
\boxed{\text{ex}} Consider the equation $u_x = tx$ (1st order, linear, inhomogeneous)
\begin{align*}
    Lu = f \text{ with } L &= \partial_x  \\
    f(x,t)&= tx
\end{align*}
We can solve this by direct integration on both sides wrt $x$.
\begin{align*}
    u(x,t) = \frac 12 tx^2 + A(t)
\end{align*}
Where $A(t)$ is an arbitrary function of $t$ \\
\boxed{\text{ex}} Consider the equation $u_{tt} - 4u = 0$ (2nd order, linear, homogeneous)
\begin{align*}
    Lu &= f \\
    \rightarrow L &= (\partial^2_t - 4) \\
    f(x,t) &= 0
\end{align*}
There is no explicit $x$-dependence
\begin{itemize}
    \item We can treat this as an ODE with $x$ as a parameter.
    \item 'Constants' depend on $x$
    \item General solution: 
    \begin{align*}
        u(x,t) = A(x)e^{-2t} + B(x)e^{2t}
    \end{align*}
\end{itemize}

\subsection{Lecture 2 - Characteristics}
\underline{Recap}:
\begin{itemize}
    \item Heat equation for $u(x,t)$ 
    \begin{align*}
        u_t - ku_{xx} = 0, \quad k > 0, \text{constant}
    \end{align*}
    \item We can write the heat equation as $Lu = 0$, where $L = (\partial_t - k \partial_x^2)$ 
    is a differential operator.
    \item L is linear $\leftrightarrow$ $L(u+v) = Lu + Lv$ and $L(cu) = cLu\ \forall $ 
    functions $u,v, $ constant $c$
    \item PDE: $Lu = f$ is linear if $L$ is linear 
    \item Linear PDE: $Lu = f$ is homogeneous if $f = 0$, inhomogeneous otherwise.
\end{itemize}
\underline{First-order linear PDE} \\
Consider functions $u(x,y)$
\begin{align*}
    u:\mathbb{R}^2 \rightarrow \mathbb{R}
\end{align*}
Often we think of $x,y$ as spatial variable and $t$ as time variables. The simplest PDE we 
can have following this is:
\begin{align*}
    u_x = 0
\end{align*}
Which is 1st-order, linear, homogeneous. Also note that there is no explicit $y$-dependence. 
Therefore we can treat this like an ODE, and solve with direct integration of both sides.
\begin{align*}
    u(x,y) = f(y)
\end{align*}
where $f(y)$ is an arbitrary function of $y$. \\ \\
Now consider $au_x + bu_y = 0$, which is a 1st-order, linear, homogeneous PDE with constants 
$a,b \in \mathbb{R}$.
\begin{align*}
    \text{Let } V &= (a,b) \in \mathbb{R}^2 \\
    \text{Recall } \nabla u &= (u_x,u_y) \\
    \text{Then } V \cdot \nabla u &= (a,b) \cdot (u_x,u_y) \\
    &= au_x + bu_y = 0
\end{align*}
$\Rightarrow$ Directional derivative in direction $V$ is 0. \\
$\Rightarrow$ $u$ is constant on lines parallel to $V$. \\ \\
Let $V^\perp = (b,-a)$, so that $V \cdot V^\perp = 0$. \\
Lines parallel to $V$ satisfy 
\begin{align*}
    V^\perp \cdot (x,y) &= \text{constant} \\
    \Rightarrow bx - ay &= \text{constant}
\end{align*}
Then $u$ is constant on lines parallel to $V$.
\begin{align*}
    \Rightarrow u(x,y) = f(bx-ay) \\
    \text{for an arbitrary function of }f
\end{align*}
\underline{Check}:
\begin{align*}
    &(a\partial_x + b\partial_y)\ f(bx - ay) \\
    &= abf' - baf' = 0 \\
    &\Rightarrow f' \text{ is a derivative of } f:\mathbb{R} \rightarrow \mathbb{R}
\end{align*}
\textbf{Characteristic Coordinates.} In this example, the lines parallel to $V$ are 
called characteristics. Here $u$ is constant along the characteristics. \\ \\
We can write points $(x,y)$ in characteristic coordinates $(\xi, \eta)$. Where  
$\xi$ is the $x$-intercept of the characteristic line, and $\eta$ is 
the distance along that line to the point $(x,y)$. 
\begin{align*}
    \text{So } x &= \xi + a\eta = x(\xi,\eta) \\
    y &= b\eta = y(\xi,\eta) \\
    \xi &= x - a\eta \\
    \eta &= \frac{y}{b} \\
    \xi &= x - \frac{a}{b}y \\
    b\xi &= bx - ay
\end{align*}
Transform PDE to $(\xi, \eta)$ coordinates. \\ \\
Chain rule:
\begin{align*}
    \frac{\partial u}{\partial\eta} &= \frac{\partial u}{\partial x} \cdot 
    \frac{\partial x}{\partial \eta} + \frac{\partial u}{\partial y} \cdot 
    \frac{\partial y }{\partial \eta} \\
    \Rightarrow u_\eta &= au_x + bu_y = 0
\end{align*}
Using characteristic coordinates $(\xi,\eta)$, we've transformed to the PDE 
$u_\eta = 0$ for $u(\xi,\eta)$. \\ \\
\underline{Solution}: 
\begin{align*}
    u &= f(\xi) \\
    \text{where } \xi &= \frac 1b (bx - ay) \\
    \Rightarrow u &= g(bx - ay)
\end{align*}
Where g is an arbitrary function. \\ \\
\underline{Remarks}: 
\begin{itemize}
    \item $\eta$ is coordinates along characteristics
    \item $u$ is constant on characteristics $\Rightarrow$ $u_\eta = 0$
    \item $\xi$ labels different characteristics. There are other choices for $\xi$
    but any choice $\tilde \xi$ would have $\tilde \xi = h(\xi)$
\end{itemize}
\underline{Method of Characteristics} \\
Method for solving PDEs of form 
\begin{align*}
    a(x,y) u_x + b(x,y) u_y = 0 \quad \text{for } u(x,y)
\end{align*}
Idea: Let $A(x,y) = (a(x,y),b(x,y))$ \\
The directional derivative $A \cdot \nabla u = 0  \\
\Rightarrow u$ is constant along curves w/ slope $A$.
\begin{itemize}
    \item These are characteristic curves.
    \item Find coord $\eta$ along chars
    \item Find coord $\xi$ which labels chars
    \item PDE transforms to $u_\eta = 0, $ general soln: $u = f(\xi)$
\end{itemize}
\boxed{\text{ex}} $u_x + yu_y = 0$ \\
The directional derivative along $(1,y)$ are 0. \\
The chars $(x(\eta), y(\eta))$ satisfy 
\begin{align*}
    \frac{dx}{d\eta} = 1,\ \frac{dy}{d\eta} = y
\end{align*}
At $\eta = 0$, take $(x(0), y(0)) = (0,\xi)$ \\
Solve: 
\begin{align*}
    \frac{dx}{d\eta} &= 1,\ x(0) = 0 \\
    \Rightarrow x &= \eta \\
    \frac{dy}{d\eta} &= y,\ y(0) = \xi \\
    \Rightarrow y &= \xi e^\eta 
\end{align*}
This follows that $\xi = ye^{-x}$. \\
\begin{align*}
    \Rightarrow u_\eta &= u_x \frac{\partial x}{\partial \eta} + u_y\frac{\partial y}{\partial \eta} \\
    &= u_x + \xi e^\eta u_y \\
    &= u_x + yu_y = 0 \\
    \Rightarrow u_\eta &= 0 \\
    \Rightarrow u &= f(\xi) = f(ye^{-x}) \text{ with arbitrary function }f
\end{align*}
\underline{General solution}:\\
Suppose we are given $u(0,y) = y^2$
\begin{align*}
    &\Rightarrow \text{Specific solution} \\
    &u(0,y) = f(y) = y^2 \\ 
    &\Rightarrow u(x,y) = (ye^{-x})^2
\end{align*}
Alternatively, from parametric equations,
\begin{align*}
    &\frac{dx}{d\eta} = 1 \qquad \frac{dy}{d\eta} = y \\
    &\Rightarrow \frac{dy}{dx} = y  \\
    &\Rightarrow y = Ce^{x}
\end{align*}
    $\Rightarrow u\ \text{is constant along chars.} \\
    \Rightarrow u(x,Ce^x) = u(0,C) \\
    = u(0,ye^{-x}) \\
    = f(e^{-x}y) \text{ for arbitrary }f$ \\ \\
\boxed{\text{ex}} $u_x + 2xy^2u_y = 0$ \\
Characteristic curves: $(x(\eta), y(\eta))$
\begin{align*}
    \frac{dx}{d\eta} = 1 &\qquad \frac{dy}{d\eta} = 2xy^2 \\
    &\frac{dy}{dx} = 2xy^2
\end{align*} 
\begin{align*}
    \int \frac{1}{y^2}dy &= \int 2xdx \\
    -\frac{1}{y} &= x^2 - c \\
    y &= \frac{1}{c-x^2} \\ 
\end{align*}
$u$ is constant along chars. \\
$\Rightarrow u(x,y) = f(c) = f(x^2 + \frac 1y)$ is the general soln. \\ \\
\underline{Advection, transport}\\
Consider $u(x,t)$ describing quantity $u$ at position $x$, time $t$.
\begin{align*}
    u_t + cu_x = 0 \quad (c>0)
\end{align*}
Chars: $(t(\eta), x(\eta))$ satisfy 
\begin{align*}
    &\frac{dt}{d\eta} = 1 \qquad \frac{dx}{d\eta} = c \\
    &\Rightarrow \frac{dx}{dt} = c  \\ &\Rightarrow x = ct + \xi\ \text{ via integration}
\end{align*}
General solution:
\begin{align*}
    u = f(\xi) = f(x-ct)
\end{align*}
\subsection{Lecture 3 - Conservation Laws}
\underline{Recap}:
\begin{itemize}
    \item Solve PDEs of the form $a(x,y)u_x + b(x,y)u_y = 0$
    \item The char curves $(x(\eta),y(\eta))$ satisfy \begin{align*}
        \frac{dx}{d\eta} = a(x,y) \qquad \frac{dy}{d\eta} = b(x,y)
    \end{align*}
    \item $u$ is constant along chars.
\end{itemize}
\boxed{\text{ex}} Given $\frac{1}{x^2} u_x + u_y = 0$
\begin{align*}
    \frac{dx}{d\eta} = \frac{1}{x^2} \qquad \frac{dy}{d\eta} = 1 
\end{align*} 
\underline{Method 1}: divide to get parametric eq.
\begin{align*}
    \frac{dy}{dx} &= x^2 \\
    y &= \frac 13 x^3 + \xi
\end{align*}
where constant $\xi$ labels the char. \\ \\
The general solution is $u = f(\xi) = f(y-\frac 13 x^3)$ for an arbitrary function $f$. This 
can be confirmed by plugging $u$ back into the PDE. \\ \\
\underline{Method 2}: solve parametric equations.
\begin{align*}
    &\frac{dx}{d\eta} = \frac{1}{x^2} \qquad \frac{dy}{d\eta} = 1 \\
    &\Rightarrow \int x^2 dx = d\eta \qquad y = \eta + B \\
    &\Rightarrow \frac{1}{3} x^3 = \eta + A \qquad y = \eta + B
\end{align*}
We can absorb the two constants into one $\xi$
\begin{align*}
    &\Rightarrow \frac 13 x^3 = \eta + \xi \qquad y = \eta \\
    &\Rightarrow y = \frac 13 x^3 - \xi
\end{align*}
We should check that the chars cover the whole plane, i.e. no chars have been 'missed', via 
sketching or finding 
\begin{align*}
    \eta &= \eta(x,y) \\
    \xi &= \xi(x,y)
\end{align*} 
Coord transformation is also useful
\begin{align*}
    \begin{matrix}
        \frac 13 x^3 = \eta + \xi \\
        y = \eta
    \end{matrix}\ \Rightarrow \ 
    \begin{matrix}
        x = (3\eta + 3\xi)^\frac 13 \\
        y = \eta
    \end{matrix}
\end{align*}
This transform our PDE
\begin{align*}
    \frac 1{x^2}u_x + u_y = 0\ &\Leftrightarrow\ u_\eta = 0 \\
    \text{Gen soln: } u &= f(\xi) \\
    &= f(\frac 13 x^3 - y)
\end{align*}
with arbitrary function $f$ \\ \\
\boxed{\text{ex}} Advection \\
Consider function $u(x,t)$ describing a quantity $u$ at position $x$ and time $t$
\begin{align*}
    u_t + cu_x = 0 \tag{\text{Advection}}
\end{align*}
with constant $c > 0$ \\
Chars $(t(\eta), x(\eta))$ satisfy 
\begin{align*}
    \frac{dt}{d\eta} = 1 \qquad \frac{dx}{d\eta} = c
\end{align*}
Solve 
\begin{align*}
    t &= \eta + A \\
    x &= c\eta + B
\end{align*}
Absorb into one constant using $A = 0, B = \xi$
\begin{align*}
    \Rightarrow t &= \eta \\
    x &=c\eta + \xi \\
    \Rightarrow x &= ct + \xi
\end{align*}
By inverting, $\eta = t,\ \xi = x - ct$. \\
Now we have an transformed PDE
\begin{align*}
    u_\eta= u_t + cu_x = 0
\end{align*}
General solution is 
\begin{align*}
    u(x,t) &= f(\xi) \\
    &= f(x-ct) 
\end{align*}
for an arbitrary $f$. \\ \\
Suppose we had initial condition $u(x,o) = \phi (x)$, then $u(x,0) = f(x) = \phi(x) \Rightarrow$
solution is $u(x,t) = \phi(a-ct)$. \\ \\
\underline{Remarks}: \\
We can solve more complicated PDEs using chars, for example,
\begin{align*}
    u_t + cu_x = f(u)
\end{align*}
Transform to coords $(\xi,\eta)$ with $t = \eta$, $x = c\eta + \xi$
\begin{align*}
    u\eta = f(u)
\end{align*}
which can be solved as an ODE \\ \\
\textbf{Conservation Laws.} (Logan 1.2)
\begin{itemize}
    \item Where do PDEs come from?
    \item For a conserved quantity:
    \begin{align*}
        \begin{matrix}
            \text{rate of change} \\ \text{in domain}
        \end{matrix}\ \implies\
        \begin{matrix}
            \text{net flow into domain} \\ \text{+ creation in domain}
        \end{matrix} \\
        \text{e.g. Animal population in a fixed region} 
    \end{align*}
\end{itemize}
\underline{Quantitatively}
\begin{itemize}
    \item Let $u(x,t)$ be density of some quantity in 1d domain.
    \item The amount of quantity in section $dx$ is $u(x,t)Adx$.
    \item Let $\phi(x,t)$ be the flux, which is the rate of quantity crossing 
    section $x$ at time $t$. Note that $\phi(x,t) dt$ is the amount of stuff, 
    and so is $u(x,t)dx$.
    \item $A\phi(x,t)$ is amount of quantity passing $x$ at time $t$
    \item Let $f(x,t)$ be the rate at which quantity is created.
    \item $f(x,t)Adx$ is amount of quantity created in width $dx$ per unit time.
\end{itemize}
\underline{Remarks}:
\begin{itemize}
    \item $\phi(x,t) > 0 \implies $ flux is L to R 
    \item $\phi(x,t) < 0 \implies $ flux is R to L 
    \item $f(x,t) > 0$ is source at $x$ (creation)
    \item $f(x,t) < 0$ is sink at $x$ (destroyed)
\end{itemize}
\underline{Balance}: 
\begin{align*}
    \begin{matrix}
        \text{change in} \\ \text{quantity in} \\ \text{section} [a,b]
    \end{matrix}\ =\ 
    \text{flux} + \text{creation}
\end{align*}
\begin{align*}
    \Rightarrow \frac{d}{dt} \int_{a}^{b}u(x,t)Adx &= A \phi(a,t) - A \phi(b,t) + \int_{a}^{b}f(x,t)Adx \\
    \Rightarrow \int_{a}^{b} u_t Adx &= -\int_{a}^{b}\phi_x Adx + \int_{a}^{b}fAdx \\
    \Rightarrow \int_{a}^{b}(u_t + \phi_x - f)dx &= 0 \\
\end{align*}
This is true for any domain $[a,b]$
\begin{align*}
    u_t + \phi = f \text{ where}\\
    \begin{matrix}
        \text{change in} \\ \text{quantity}
    \end{matrix}\ &= \ \text{net flux + net creation}
\end{align*}
\underline{Remarks}: \\
Can get a PDE for $u$ by
\begin{itemize}
    \item Specifying $f(x,t)$
    \item Choosing a relationship between $\phi$ and $u,u_x,$ etc. This is called constitutive law.
    \item E.g. $\phi = cu \rightarrow$ advection.
\end{itemize}
\underline{Diffusion}:
\begin{itemize}
    \item Conservation law with no sources \begin{align*}
        u_t + \phi_x = 0
    \end{align*}
    \item Suppose $u(x,t)$ is density of a gas of particles. Then
    \begin{itemize}
        \item Particles move from high $u$ to low $u$.
        \item Flux increase with density gradient.
        \item Fick's law: 
        \begin{align*}
            \phi(x,t) = -Du_x
        \end{align*}
        where $D>0$ is diffusion constant.
        \item This gives the diffusion equation
        \begin{align*}
            u_t = Du_{xx}
        \end{align*}
        which is the same as the heat equation.
    \end{itemize}
\end{itemize}
\underline{Remark}:
\begin{itemize}
    \item Can combine diffusion with other effects. \\
    e.g. Advection-diffusion eq.
    \begin{align*}
        \text{flux: } \phi = cu - Du_x
    \end{align*}
    \underline{Balance}: 
    \begin{align*}
        u_t + \phi_x &= 0 \\
        u_t + cu_x &= Du_{xx}
    \end{align*}
\end{itemize}

\section{Intro to Heat, Wave, Laplace's equations}
\subsection{Lecture 4 - Diffusion Equation}
\underline{Recap}:
\begin{itemize}
    \item Conservation Laws (Logan 1.2)
    \item Density: $u(x,t)$
    \item Flux: $\phi(x,t)$ (movement of $u$)
    \item Sources: $f(x,t)$ (creation of $u$)
    \item Balance in domain $[a,b]$
    \begin{align*}
        \frac{d}{dt}\int_{a}^{b}Au(x,t)dx &= A\phi(a,t) - A\phi(b,t) + \int_{a}^{b}Af(x,t)dx \\
        \int_{a}^{b}u_tdx &= - \int_{a}^{b}\phi_xdx + \int_{a}^{b}fdx \\
        \int_{a}^{b}(u_t + \phi_x - f)dx &= 0
    \end{align*}
    This holds for any $[a,b]$
    \begin{align*}
        u_t + \phi_x &= f \\
        \begin{matrix}
            \text{change in} \\ \text{quantity}
        \end{matrix}
         + \text{flux} &= \text{source}
    \end{align*}
\end{itemize}
\textbf{Diffusion equation}. \\
Suppose $u(x,t)$ is density/concentration of substance. We have Fick's Law for flux:
\begin{align*}
    \phi(x,t) = -Du_x,\ D > 0
\end{align*}
\begin{itemize}
    \item Movement from high $u$ to low $u$
    \item Steeper gradient gives higher flux 
\end{itemize}
Conservation/balance Law: 
\begin{align*}
    u_t + \phi_x &= 0 \\
    \Rightarrow u_t &= Du_{xx} \tag{Diffusion/heat eq}
\end{align*}
\underline{Remarks}:
\begin{itemize}
    \item $D$ is called diffusivity or thermal conductivity.
    \item Some physical settings have a non-constant diffusivity $D(x)$
    \begin{align*}
        \phi = - D(x)u_x
    \end{align*}
    This yields the PDE
    \begin{align*}
        u_t &= (Du_x)_x \\
        &= D'u_x + Du_{xx} \\
        \text{where } D' &= \frac{dD}{dx}
    \end{align*}
    \item Can also find nonlinear diffusion equations where $D = D(u)$
    \begin{align*}
        u_t = (D(u)u_x)_x
    \end{align*}
    \item We can combine diffusion with other terms, e.g. diffusion-advection
    \begin{align*}
        u_t + cu_x = Du_{xx} \\
        \phi(x,t) = - Du_x + cu
    \end{align*}
\end{itemize}
\underline{Dimensions/units}:\\
Suppose $u(x,t)$ is temperature, then 
\begin{align*}
    [u] &= \text{temp} \\
    [u_t] &= \frac{\text{temp}}{\text{time}} \\
    [u_{xx}] = \frac{\text{temp}}{\text{length}^2}
\end{align*}
So $u_t = Du_{xx}$
\begin{align*}
    \Rightarrow[D] = \frac{\text{length}^2}{\text{time}}
\end{align*}
Consider heat flow in a region of length $L$. The time-scale $T$ for temperature change is 
\begin{align*}
    T = \frac{L^2}{D} \ \text{so } \ D = \frac{L^2}{T}
\end{align*}
\textbf{Boundary \& Initial Conditions}.
Diffusion eqs usually have BCs and ICs. Consider heat flow in 1d with domain $0<x<l$. \\ \\
IC has form 
\begin{align*}
    u(x,0) = u_0(x)
\end{align*}
where $u_0(x)$ is initial temperature profile. \\ \\
There are 3 types of common BCs.
\begin{enumerate}
    \item Dirichlet BCs \\
    Specify $u$ at the boundary.
    \begin{align*}
        u(0,t) &= g_1(t) \\
        u(l,t) &= g_2(t)
    \end{align*}
    where $g_1,g_2$ are given i.e. $u$ is known at boundary.
    \item Neumann BCs \\
    Specify $u_x$ at the boundary.
    \begin{align*}
        u_x(0,t) &= h_1(t) \\
        u_x(l,t) &= h_2(t)
    \end{align*}
    where $h_1, h_2$ are given, i.e. flux is known at boundary.
    \item Robin BCs \\
    Specify $u,u_x$ given at boundary.
    \begin{align*}
        au(0,t) + bu_x(0,t) = \psi_1(t) \\
        au(l,t) + bu_x(l,t) = \psi_2(t) 
    \end{align*}
    where $\psi_1, \psi_2$ are given. E.g. Newton's Law of Cooling states that heat flow 
    $\propto$ temp difference between object and environment.
    \begin{align*}
        bu_x(0,t) &= - au(0,t) + \psi_1(t) \\
        \text{flux} &= \text{object} + \text{environment}
    \end{align*}
\end{enumerate}
\underline{Remarks}:
\begin{itemize}
    \item A well-posed problem is a PDE with ICs and BCs s.t. it has a unique solution with 
    a correct physical meaning (stability).
    \item The above terminology for BCs applies to general PDEs.
\end{itemize}
\textbf{Steady-state solutions}. Many time-dependent PDEs have solutions which approach steady 
states $u = u(x)$, which are time-independent. E.g. diffusion/heat equation:
\begin{align*}
    u_t = Du_{xx}
\end{align*}
Steady state solution $u(x)$ has $u_t = 0$, so 
\begin{align*}
    Du'' = 0 \quad (u' = u_x)
\end{align*}
This gives a 2nd-order ODE, where 
\begin{align*}
    u(x) = Ax + B
\end{align*}
for constants $A,B$ set by BCs. \\ \\
\boxed{\text{ex}} Consider the diffusion-decay equation
\begin{align*}
    u_t = Du_{xx} - ru \\
    0 < x < L,\ t,D,r > 0
\end{align*}
$u$ has the flux $\phi = - Du_x$ and is destroyed at rate $ru$. Take BCs: 
\begin{align*}
    u(0,t) &= 0 \\ 
    -Du_x(L,t) &= -1 \\
    t > 0
\end{align*}
and IC
\begin{align*}
    u(x,0) = g(x) \\
    0 < x < L
\end{align*}
\underline{Remarks}: \\
This model could describe a diffusing pollutant ($Du_{xx}$) which decays at rate $ru$.
\begin{align*}
    \text{At }0:&\ u(0,t) = 0 \\
    &\Rightarrow u\text{ is held at 0} \\
    \text{At }L:& -Du_x(L,t) = -1 \\
    &\Rightarrow \text{constant flux of }u \text{ into domain}
\end{align*}
Steady state problem:
\begin{align*}
    u = u(x), \text{ so } u_t = 0
\end{align*}
BVP:
\begin{align*}
    Du'' - ru = 0 \\ 0 < x < L
\end{align*}
BCs: $u(0) = 0$, $u'(L) = \frac{1}{D}$ \\
Here we ignore IC and assume transients have decayed. \\
Solution of BVP:
\begin{align*}
    u'' = \frac rD u \\
    u(0) = 0,\ u'(L) = \frac 1D
\end{align*}
General solution:
\begin{align*}
    u = A\sinh\left(x\sqrt{\frac{r}{D}}\right) + B\cosh\left(x\sqrt{\frac{r}{D}}\right) \\
    \text{consts } A,B
\end{align*}
BC1: 
\begin{align*}
    u(0) &= 0 \\
    \Rightarrow B &= 0 \\
    \Rightarrow u(x) &= A\sinh\left(x\sqrt{\frac{r}{D}}\right)
\end{align*} 
BC2: 
\begin{align*}
    u'(L) &= \frac 1D \\
    \Rightarrow A \sqrt\frac{r}{D} \cosh \left(L \sqrt{\frac{r}{D}}\right) &= \frac 1D \\
    \Rightarrow u(x) &=  \frac{\sinh\left( x \sqrt \frac rD\right)}{\sqrt{rD} \cosh\left(L \sqrt \frac rD\right)}
\end{align*}
\underline{Remarks}:
\begin{itemize}
    \item An ODE BVP may not have a solution. Physically the PDE may not have a steady 
    state solution.
    \item The steady state solution could be unstable in general. (Here it turns out that 
    $\lim_{t\rightarrow \infty} u(x,t) \rightarrow $ steady state.)
\end{itemize}
\textbf{Summary}. Diffusion in 1d, for $u(x,t)$
\begin{align*}
    u_t = Du_{xx} \\
    D > 0
\end{align*}
Can consider higher dimensions for $u(x,y,t)$ \\ \\
\underline{Preview}: Vibrations and waves \\
Consider the 1d wave equation for $u(x,t)$:
\begin{align*}
    u_{tt} = C^2u_{xx}
\end{align*}
This describes, for example, waves on a string. $C$ has dimensions length/time and 
is the speed of waves. 2nd-order in time $\rightarrow$ waves (contrasts with 1st-order 
in time $\rightarrow$ diffusion). \\ \\
We have the string fixed at length $x = 0, L$ and $u(x,t)$ is the height of the string
at time $t$. And we have the following equation of motion for the string
\begin{align*}
    u_{tt} = \sqrt \frac{T}{\rho}u_{xx}
\end{align*}
where $T$ is string tension, and $\rho$ is the density of string.

\subsection{Lecture 5 - Wave Equation}
\underline{Recap}:
\begin{itemize}
    \item Diffusion: $u_t = Du_{xx}$, $D > 0$
    \item Wave: $u_{tt} = c^2u_{xx}$, $c$ is speed 
\end{itemize}
\textbf{Vibrations of a string}.
\begin{itemize}
    \item String fixed at $x=0, x=l$.
    \item $u(x,t)$ is the height of the string at time $t$, position $x$
    \item Density $\rho = \frac{\text{mass}}{\text{length}}$
\end{itemize}
\underline{Assumptions}:
\begin{itemize}
    \item Small deflections, i.e. $|u_x|$ is small.
    \item Only force acting on string is tension (no gravity, damping, etc.)
    \item No horizontal motion of string (only vertical)
\end{itemize}
Let $T(x,t)$ be the tension, $T$ acts tangent to the string. \\
Let $\theta(x,t)$ be the angle of the tangent to the string, so
\begin{align*}
    \cos \theta = \frac{1}{\sqrt{1+u_x^2}}  && \sin \theta = \frac{u_x}{\sqrt{1+u_x^2}}
\end{align*}
So $T$ is the tension force along the string. Therefore, total force on the string 
in region $[a,b]$ is
\begin{align*}
    \text{horiz:}\quad &T\cos\theta\big\vert^b_a = \frac{T}{\sqrt{1+u_x^2}}\Big\vert^b_a \\ \\
    \text{vert:}\quad &T\sin\theta\big\vert^b_a = \frac{Tu_x}{\sqrt{1+u_x^2}}\Big\vert^b_a
\end{align*}
We can simplify this using the small deflection assumption, $|u_x| \ll 1$ 
\begin{align*}
    \text{Taylor series}: (1+u^2_x)^\frac 12 &= 1 + \frac 12 u_x^2 - \frac 18 u _x^4 + \dots\\
    &\approx 1
\end{align*}
where each $u_x$ term gets smaller. \\ \\
So the forces become
\begin{align*}
    \text{horiz}: \quad &T\cos\theta\big\vert^b_a \approx T \big\vert^b_a = \int_{a}^{b}T_xdx \\
    \text{vert}: \quad &T\sin\theta\big|^b_a \approx Tu_x \big|^b_a = \int_{a}^{b}(Tu_x)_xdx
\end{align*}
The accelerations are
\begin{align*}
    \text{horiz}: \quad &0 \quad \text{(by assumption)} \\
    \text{vert}: \quad &\int_{a}^{b}u_{tt}dx
\end{align*} 
Using Newton's 2nd Law, $F = ma$
\begin{align*}
    \text{horiz}: \quad &\int_{a}^{b} T_xdx = 0 \\
    \text{vert}: \quad & \int_{a}^{b} (Tu_x)_xdx = \int_a^b \rho u_{tt}dx
\end{align*}
This holds for all intervals $[a,b]$  \\
$\Rightarrow T_x = 0$ so $T(x,t) = T(t)$ \\ 
and $(Tu_x)_x = Tu_{xx}$
\begin{align*}
    \text{vert}: Tu_{xx} &= \rho u_{tt} \\
    \Rightarrow u_{tt} &= c^2u_{xx} \text{ for } c = \sqrt{\frac T\rho}
\end{align*}
\underline{Remarks}:
\begin{itemize}
    \item Typically assume $T(t)$ is also time-independent so 
    $T = $ const, and speed $c = \sqrt{\frac T\rho}$ is const.
    \item For constant $c$, the general solution is 
    \begin{align*}
        u(x,t) = F(x-ct) + G(x+ct)
    \end{align*}
    where $F$ is a right-moving wave, and $G$ is a left-moving wave
    \item For a plucked string, our BCs are 
    \begin{align*}
        u(0,t) = u(l,t) = 0 
    \end{align*}
    \item To get a specific solution, we need \underline{2 ICs}, since
    the equation is 2nd order in time. E.g. we need
    \begin{align*}
        \text{Init position}: u(x,0) &= f(x) \\
        \text{Init velocity}: u(x,0) &= g(x) \\
        \text{for}\ 0 < x &< l
    \end{align*}
    \item There are many variations, e.g.:
    \begin{itemize}
        \item Air resistance term, $ru_t$:
        \begin{align*}
            u_{tt} - c^2u_{xx} + ru_t = 0
        \end{align*}
        \item External force, $f(x,t)$:
        \begin{align*}
            u_{tt} - c^2u_{xx} = f(x,t)
        \end{align*}
    \end{itemize}
\end{itemize}
\textbf{Standing waves}. Consider a vibrating string
\begin{align*}
    u_{tt} &= c^2u_{xx} \\
    \text{BCs}: u(0,t) &= u(l,t) = 0 
\end{align*}
Standing waves are solutions of the form 
\begin{align*}
    u(x,t) =& \phi(x)\cos\omega t \\
    \text{or }& \phi(x) \sin\omega t \\
    \text{or in general }& \phi(x)e^{i\omega t}
\end{align*}
Each part of the string oscillates with:
\begin{itemize}
    \item Same freq $\omega$, in phase.
    \item Different amplitude $\phi(x)$
\end{itemize}
Solutions only exist for certain $\omega$
\begin{align*}
    \text{substitute } u &= \phi(x) \cos\omega t \text{ into wave eq} \\
    - \omega^2 \phi \cos \omega t &= c^2 \phi'' \cos\omega t \\
    \text{with } \phi(0)\cos\omega t &= \phi(l) \cos \omega t = 0\ \forall \ t
\end{align*}
This gives us an ODE BVP:
\begin{align*}
    \phi '' = -\frac{\omega^2}{c^2} \phi,\ \phi(0) = \phi(l)= 0
\end{align*}
We have the following general solution
\begin{align*}
    \phi(x) = A\cos\frac{\omega x}{c} + B\sin\frac{\omega x}{c}
\end{align*}
BC1: $\phi(0) = A = 0$ 
\begin{align*}
    \phi(x) = B \sin \frac{\omega x}{c}
\end{align*}
BC2: $\phi(l) = B \sin \frac{\omega l }{c}$
\begin{align*}
    B = 0 \text{ or } \sin\frac{\omega l}{c} = 0
\end{align*}
If $B = 0$ then $\phi(x) = 0$ and $u(x,t) = 0$. We are interested in nonzero 
solutions so we need $\sin \frac{\omega l}{c} = 0$ \\ \\
\underline{Note}: $l$ and $c$ are fixed by the problem, $\omega$ is the 
frequency of the standing wave. The condition $\sin \frac{\omega l}{c} = 0$
means the allowed frequencies $\omega$ of a standing wave satisfy 
$\sin\frac{\omega l}{c} = 0$
\begin{align*}
    \Rightarrow \frac{\omega l}{c} = n\pi \text{ for } n \in \mathbb{Z}
\end{align*}
Allowed frequencies: $\omega_n = \frac{n\pi c}{l}$ \\
Modes: $\phi_n(x) = \sin\omega_n x = \sin \left(\frac{n\pi x}{l}\right),\ n \in \mathbb{Z}$
\begin{itemize}
    \item $B\phi_n(x)$ satisfies the BVP for any constant $B$, modes defined
    up to a multiplicative constant.
    \item $n=0$ gives $\phi_n = 0$ so it is trivial.
    \item $n<0$ gives the same mode as $n>0$ up to a sign: $\phi_n(x) = - \phi_{-n}(x)$
    \item We can take the modes to have $n>0$
    \begin{align*}
        \phi_n(x) = \sin\left(\frac{n\pi x}{l}\right),\ n = 1,2,3,\dots
    \end{align*}
\end{itemize}
The standing waves are 
    \begin{align*}
        u_n(x,t) &= \cos\left(\frac{n\pi ct}{l}\right)\sin \left(\frac{n\pi x}{l}\right) \\
        &= \cos(\omega_n t)\phi_n(x),\ n = 1,2,3,\dots
    \end{align*}
\underline{Remarks}:
\begin{itemize}
    \item The BVP problem is an eigenvalue problem. The frequencies $\omega_n$
    are the $c$-values and modes $\phi_n$ are eigenfunctions.
    \item Eigenfunctions $\phi_n$ are defined up to a constant multiplier,
    like eigenvectors.
    \item The frequencies $\omega_n$ are the fundamental frequencies of the string.
\end{itemize}
\underline{Quantum Mechanics} (Logan 1.6) \\
Consider a particle in 1d. Classically, the particle has a position $x\in \mathbb{R}$.
In quantum mechanics, the state of the particle is a function $\Psi(x,t)\in \mathbb{C}$
called the wave function. The probability density of finding the particle at $x$ is 
$|\Psi(x,t)|^2$. $\Psi$ obeys the Schrödinger equation (SE):
\begin{align*}
    i\hbar \Psi_t &= -\frac{\hbar^2}{2m}\Psi_{xx} \quad (\text{free particle}) \\
    \text{where } &\hbar \text{ is Planck's constant} \\
    &m \text{ is particle mass}
\end{align*}
If the 'energy cost' of being at position $x$ is $V(x)$, then the SE is 
\begin{align*}
    i\hbar \Psi_t = -\frac{\hbar^2}{2m} \Psi_{xx}+V(x)\Psi
\end{align*}
\underline{Remarks}:
\begin{itemize}
    \item The particle is typically more likely to be at places where $V(x)$ is lower.
    \item Probability is conserved
    \begin{align*}
        \int_{-\infty}^{\infty}|\Psi|^2dx = 1 \ \forall t
    \end{align*}
    \item Looking for solutions of the form 
    \begin{align*}
        \Psi(x,t) = e^{-iEt/\hbar}\cdot \phi(x)
    \end{align*}
    gives the following time=independent SE:
    \begin{align*}
        E\phi = -\frac{\hbar^2}{2m}\phi'' + V(x)\phi
    \end{align*}
    Where $E$ has the interpretation of 'energy'
\end{itemize}
For a particle in a '1d box,' $0 < x < l$, we have 
\begin{align*}
    E\phi = -\frac{\hbar^2}{2m}\phi'',\ 0 < x < l
\end{align*}
with the following BCs
\begin{align*}
    \phi(0) = \phi(l) = 0 \\
    \Rightarrow \text{standing wave solutions}
\end{align*}
\subsection{Lecture 6 - Higher Dimensional PDEs}
\underline{Recap}:
\begin{itemize}
    \item 1st order PDEs, characteristics, change of coords.
    \item 2nd order PDEs (1 space dimension), diffusion/heat eq, with advection, decay, etc.
    \begin{align*}
        u_t = Du_{xx}
    \end{align*}
    \item Steady state solutions
    \item Wave equation
    \begin{align*}
        u_{tt} = c^2u_{xx}
    \end{align*}
    \item Standing waves, with BCs and ICs. 
    \item Dirichlet ($u$ is known on boundary), and Neumann ($u_x$ is known on boundary).
    \item ODEs: separation of variables, homogeneous solutions, 2nd order constant coefficients.
\end{itemize}
Higher dimensional PDEs (Logan 1.7) 
\begin{itemize}
    \item Most physical processes occur in more than 1d.
    \item We can generalize 1d derivations using vector calculus.
\end{itemize}
\textbf{Heat flow in 2d/3d}. 2d and 3d are similar, so we work in 3d. Consider heat flow in 
domain $\Omega \subseteq \mathbb{R}^3$. 
\begin{itemize}
    \item Let $u(x,y,z,t)$ be the temperature at point $(x,y,z) \in \Omega$ at time $t$. 
    \item Let $Cu$ be the thermal energy density.
    \item Let $\Phi(x,y,z,t)$ be the energy flux vector, i.e. $\Phi$ is the vector field describing 
    heat flow.
    \item The rate of heat flow across a surface element, $dS$ is $\Phi \cdot dS$ 
    \item Recall that $dS = n \cdot dA$, where $n$ is the normal vector, and $dA$ is the area element.
    \item In 2d, flow across a line element is $\Phi \cdot n dl$
    \item Let $f(x,y,z,t)$ be the rate at which heat is produced.
    \item Consider thermal energy in subdomain $B \subseteq \Omega$
\end{itemize}
\underline{Conservation/balance}:
\begin{align*}
    \frac{d}{dt} \int_{B} CudV &= - \int_{\partial B} \Phi \cdot dS + \int_{B}fdV \\ \\
    \begin{matrix}
        \text{change in} \\ \text{energy} 
    \end{matrix} &= 
    \begin{matrix}
        \text{flux out} \\ \partial B\text{ is boundary}
    \end{matrix} + \text{sources/sinks}
\end{align*}
\underline{Remark}: this applies to virtually every spatially varying density. \\ \\
Using the \underline{divergence theorem}, we can get the PDE. \\
Let $\vec F$ be a (smooth) vector field in region $B \subseteq \mathbb{R}^3$. Then 
\begin{align*}
    \int_{\partial B} F \cdot dS = \int_{B} \nabla \cdot F dB
\end{align*}
where $\vec F = (F_1,F_2,F_3)$ and $\nabla \cdot F = \partial_x F_1 + \partial_y F_2 + \partial_z F_3$ \\ \\
Apply this to conservation equation:
\begin{align*}
    \frac{d}{dt} \int_{B}CudV &= - \int_{\partial B} \Phi \cdot dS + \int_{B}fdV \\
    \int_{B} Cu_t dV &= - \int_{B} \nabla \cdot \Phi dV + \int_{B} fdV
\end{align*}
This is true for any $B \subseteq \Omega$
\begin{align*}
    \Rightarrow Cu_t = - \nabla \cdot \Phi + f
\end{align*}
To get a PDE for $u$, we need a relationship between $\Phi$ and $u$. In 1 space dimension, we had 
$u = u(x,t)$, $\phi(x,t)$, and $\phi = -Ku_x$ with constant $K > 0$ (in 1d, $\phi$ is effectively 
a scalar). \\ \\ 
Fick's Law in higher dimensions:
\begin{align*}
    \Phi &= -K \nabla u \quad (\text{constant } K > 0) \\
    \nabla u &= (u_x, u_y, u_z)
\end{align*}
This gives the PDE 
\begin{align*}
    Cu_t &= - \nabla \cdot \Phi + f \\
    &= K\nabla \cdot (\nabla u)
\end{align*}
where
\begin{align*}
    \nabla \cdot (\nabla u) &= \nabla \cdot (u_x,u_y,u_z) \\
    &= u_{xx} + u_{yy} + u_{zz} \\
    &:= \nabla ^2 u \quad (\text{Laplacian of }u)
\end{align*}
This gives the PDE 
\begin{align*}
    Cu_t = K \nabla^2u + f \\
    \text{for } (x,y,z) \in \Omega
\end{align*}
\underline{Definition}: The Laplacian \\
Let $u$ be a scalar function which depends on space and possibly time. The Laplacian 
of $u$, $\nabla^2u$ is the sum of 2nd derivatives with respect to space (in Cartesian
coords). \\ \\
\underline{Remarks}:
\begin{itemize}
    \item The Laplacian $\nabla^2$ is a linear differential operator.
    \item We've only defined $\nabla^2$ for scalar functions.
    \item Laplacian of $u$ is sometimes written as $\Delta u$
    \begin{align*}
        \text{e.g. } \Delta u = \nabla^2 u = \nabla\cdot(\nabla u)
    \end{align*}
\end{itemize}
\underline{Heat/diffusion eq in higher dimensions}.
\begin{align*}
    u_t = D \nabla^2 u,\ \text{constant } D > 0
\end{align*}
for $u = u(x,y,z,t)$ and $(x,y,z) \in \Omega \subseteq \mathbb{R}^3$  \\ \\
\underline{Heat/diffusion eq with heat source}.
\begin{align*}
    u_t = D \nabla^2 u + f
\end{align*}
for $f = f(x,y,z,t)$ \\ \\
\underline{Remarks}:
\begin{itemize}
    \item The time-independent cases also give important equations \\
    Poisson equation:
    \begin{align*}
        \nabla^2 u = - f\quad \in \Omega \\
        u = u(x,y,z)
    \end{align*}
    Laplace equation:
    \begin{align*}
        \nabla^2 u = 0\quad \in \Omega \\
        u = u(x,y,z)
    \end{align*}
    Wave equation:
    \begin{align*}
        u_{tt} = c^2 \nabla^2 u \quad \in \Omega
    \end{align*}
    \item These equations are fundamental for describing spatially varying quantities 
    in 2d, 3d. E.g. fluid flow, vibrations, quantum mechanics, electromagnetism.
\end{itemize}
\underline{Boundary Conditions}. Just like in 1d, PDEs in higher dimensions need BCs. \\
Consider the heat equation $u_t = D \nabla^2 u$ in 1d, where $\Omega = [0,L] \subseteq \mathbb{R}$
\begin{itemize}
    \item Dirichlet BCs: $\begin{cases}
        u(0,t) = g_1(t) \\
        u(L,t) = g_2(t)
    \end{cases}$ i.e. $u$ is known on boundary.
    \item Neumann BCs:$\begin{cases}
        u_x(0,t) = g_1(t) \\
        u_x(L,t) = g_2(t)
    \end{cases}$ i.e. $u_x$ is known on boundary (flux).
\end{itemize}
In 2d, we have the area $\Omega \in \mathbb{R}^2$ and $\partial \Omega$ is the boundary (curve).
\begin{itemize}
    \item Dirichlet BC: $u(x,y,t) = g(x,y,t)$ for $(x,y) \in \partial \Omega$ (boundary). \\
    This can be written as 
    \begin{align*}
        u\big|_{\partial \Omega} = g
    \end{align*}
    i.e. $u$ is known on boundary.
    \item Neumann BC: $n(x,y) \cdot \nabla u(x,y) = g(x,y,t)$ for $(x,y) \in \partial \Omega$ 
    where $n$ is the unit normal to $\partial \Omega$ at $(x,y)$. \\
    This can be written as
    \begin{align*}
        \vec n \cdot \nabla u \big|_{\partial \Omega} = g
    \end{align*}
    i.e. flux out of domain $n\cdot \nabla u$ is known.
\end{itemize}
\boxed{\text{ex}} Vibrating drum. \\ 
A drum occupies region $\Omega \subseteq \mathbb{R}^2$, $u(x,y,t)$ is height of the drum, and 
the drum is clamped on its boundary.
\begin{align*}
    u\big|_{\partial \Omega} = 0 \quad (\text{Dirichlet BC})
\end{align*}
The height of the drum obeys 
\begin{align*}
    u_{tt} = c^2 \nabla^2 u\ \text{ on } \Omega \\
    u\big|_{\partial \Omega} = 0 
\end{align*}
where $c^2 = \frac{T}{\rho} = \frac{\text{tension}}{\text{density}}$ \\ \\ 
Standing wave solutions:
\begin{align*}
    \nabla^2 u = -\frac{\omega^2}{c^2}u,\quad u\big|_{\partial \Omega} = 0
\end{align*}

\section{Unbounded Domains}
\subsection{Lecture 7 - Classification \& Unbounded Heat Equation}
\underline{Recap}:
\begin{itemize}
    \item Characteristics: 1st order PDEs
    \item Conservation laws and derivations for PDEs.
    \item Special solutions: steady states and standing waves.
    \item The Laplacian, $\nabla^2$, which acts on spatial variables, e.g. for $u = u(x,y)$ 
    or $u(x,y,t)$, we have $\nabla^2 u = u_{xx} + u_{yy}$.
\end{itemize}
Consider Laplace's equation for $u = u(x,y)$
\begin{align*}
    \nabla^2 u = u_{xx} + u_{yy} = 0
\end{align*}
Functions which satisfy $\nabla^2 u = 0$ are called \underline{harmonic}, e.g. $x+y,\ xy,\ 
x^2-y^2$. 
\begin{itemize}
    \item Laplace's equation comes with a domain $\Omega \subseteq \mathbb{R}^2$
    \begin{align*}
        \nabla^2 u = 0, \text{ for } (x,y) \in \Omega
    \end{align*}
    Note that $\Omega = \mathbb{R}^2$ is a possibility
    \item Laplace's equation also comes with BCs:
    \begin{itemize}
        \item Dirichlet BCs: Specify $u$ on $\partial \Omega$
        \item Neumann BCs: Specify $n\cdot \nabla u$ on $\partial \Omega$
    \end{itemize}
    where $\partial \Omega$ is the boundary of $\Omega$ and $\vec{n}$ is the outward unit normal 
    to $\partial \Omega$
\end{itemize} 
\textbf{Classification of PDEs}. We have 3 main PDEs for functions of 2 variables.
\begin{enumerate}[1)]
    \item Diffusion: $u = u(x,t)$
    \begin{align*}
        u_t - ku_{xx} = 0,\quad k > 0
    \end{align*}
    \item Wave: $u = u(x.t)$
    \begin{align*}
        u_{tt} - c^2u_{xx} = 0
    \end{align*}
    \item Laplace: $u = u(x,y)$
    \begin{align*}
        u_{xx} + u_{yy} = 0
    \end{align*}
\end{enumerate}
\underline{Remarks}:
\begin{itemize}
    \item 1) and 2) are evolution equations which specify how $u$ changes in time. Typically, 
    prescribe an initial state, along with BCs, ask for state at time $t$.
    \item We think of 3) as a static problem.
    \item Choosing $t$ to be a 'time' variable in 1) and 2), but not in 3), is not as arbitrary 
    as it seems.
\end{itemize}
\underline{Classification}: \\
Let $u(x,y)$ be a function of 2 variables. Here we think $x,y,$ as general variables (not 
necessarily space or time). Consider a 2nd order PDE of the form:
\begin{align*}
    Au_{xx} + Bu_{xy} + Cu_{yy} + F(x,y,u,u_x, u_y) = 0 
\end{align*}
where $A,B,C$ are constants and $F$ is some arbitrary function. \\ \\ 
The \underline{discriminant} of this PDE is 
\begin{align*}
    \boxed{D = B^2 - 4AC}
\end{align*}
A PDE is called $\begin{cases}
    \text{hyperbolic if } D > 0 \\
    \text{parabolic if } D = 0 \\
    \text{elliptic if } D < 0
\end{cases}$ \\
E.g.:
\begin{enumerate}[1)]
    \item Heat equation $u_t - ku_{xx} = 0$, $k > 0$. Setting $y = t$ and comparing, we get 
    $A = -k,\ B = 0,\ C = 0 \Rightarrow D = B^2 -4AC = 0$. This means the heat equation is 
    parabolic.
    \item Wave equation $u_{tt} - c^2u_{xx} = 0$. Setting $y = t$ gives $A = -c^2,\ B = 0,
    \ C = 1 \Rightarrow D = B^2 - 4AC = 4c^2 > 0$. This means the wave equation is hyperbolic.
    \item Laplace's equation $u_{xx} + u_{yy} = 0$. This gives $A = 1,\ B = 0,\ C = 1 
    \Rightarrow D = B^2 -4AC = -4 < 0$. This means Laplace's equation is elliptic.
\end{enumerate}
\underline{Remarks}:
\begin{itemize}
    \item PDEs of the same class have similar behavior, e.g. hyperbolic equations are wave-time.
    \item Terminology comes from conic sections.
    \item In general, the coefficients $A,B,C$ can be functions of $(x,y)$, in which case, the 
    PDE class could vary over the plane.
\end{itemize}
Having set up our 3 main classes of PDEs, we will proceed to solve them. \\ \\
\textbf{Unbounded Domains}.
\begin{itemize}
    \item Infinite or semi-infinite domains are easier because BCs are looser.
    \item We start with the heat equation.
\end{itemize}
\underline{Cauchy Problem}: (Logan 2.1) \\
Consider the heat equation for $u(x,t)$ on the real line:
\begin{align*}
    u_t = ku_{xx},\ k > 0 \\
    t > 0 \\
    x \in \mathbb{R} 
\end{align*}
We have IC $u(x,0) = \phi(x)$ with $x \in \mathbb{R}$. What about BCs?
\begin{itemize}
    \item For infinite domains, we typically also have some decay conditions. Take $u_x 
    \rightarrow 0$ as $x \rightarrow \pm \infty$.
    \item This is a pure IC problem, or a Cauchy problem.
\end{itemize}
Given a solution to the heat equation, we can generate more solutions. E.g. Suppose $G(x,t)$
solves the heat equation. 
\begin{align*}
    G_t - kG_{xx} = 0 \\
   \text{or } (\partial_t -k\partial^2_x) G = 0 
\end{align*}
Then $G_x$ also solves the heat equation:
\begin{align*}
    (\partial_t -k\partial^2_x) G_x &= G_{xt} - kG_{xxx} \\
    &= (G_t - kG_{xx})_x \\
    &= 0
\end{align*}
So if $G$ solves the heat equation then $G_x$ also solves the heat equation. It turns out that 
$G(x-y,t)$ also solves the heat equation.
\begin{align*}
    (\partial_t -k\partial^2_x) G(x-y,t) = 0 
\end{align*}
and or any function $\psi(y)$,
\begin{align*}
    \int_{-\infty}^{+\infty} G(x-y, t) \psi(y)dy
\end{align*}
solves the heat equation. \\ \\ 
\boxed{\text{ex}} $w(x,t) = 2kt + x^2$ solves $w_t - kw_{xx} = 0$. Consider $w_x$ as a solution.
\begin{align*}
    w_x = 2x
\end{align*}
This also solves the heat equation. Now consider $w(x-y,t) = 2kt - (x-y)^2$
\begin{align*}
    &\partial_t w(x-y,t) = 2k \\
    &\partial_{x}^2 w(x-y,t) = 2
\end{align*}
What about $\int w(x-y,t) \psi(y)dy$? This will be in HW3. \\ \\ 
Returning to the Cauchy problem, we have 
\begin{align*}
    u_t - ku_{xx} = 0 \\
    k,t > 0,\ x \in \mathbb{R} \\
    u(x,0) = \phi(x) \\
    u_x \rightarrow 0, \text{ as } x \rightarrow \pm \infty
\end{align*}
\underline{Strategy}: First solve special case where $\phi$ is a step-function then use above
moves to generate general solutions for any $\phi$. \\ \\
For step-function $\phi$ we have:
\begin{align*}
    u_t - ku_{xx} = 0 \\
    k,t > 0,\ x \in \mathbb{R} \\
    u(x,0) = \begin{cases}
        \alpha & x > 0 \\
        0 & x < 0
    \end{cases}
\end{align*}
\underline{Dimensional analysis}: The variables, constants here are: $x,t,u,k,\alpha$. We
can interpret 
\begin{center}
    $\begin{matrix}
    [u] = \text{temp} & [x] = \text{length} \\
    [\alpha] = \text{temp} & [t] = \text{time}
\end{matrix}$
\end{center}
We also have 
\begin{center}
    $\begin{matrix}
        [u_t] = \displaystyle\frac{\text{temp}}{\text{time}} & [u_{xx}] = \displaystyle\frac{\text{temp}}{\text{length}^2}
    \end{matrix}$
    \\ and $[k] = [\displaystyle\frac{u_t}{u_{xx}}] = \frac{\text{length}^2}{\text{time}}$
\end{center}
There are two dimensionless quantities:
\begin{align*}
    &\frac{u}{\alpha}\ \text{ from temperature parameters} \\
    &\frac{x}{\sqrt{4kt}}\ \text{ from length, time} \\
    &\Rightarrow \frac{u(x,t)}{\alpha} = f\left(\frac{x}{\sqrt{4kt}}\right) 
\end{align*}
\subsection{Lecture 8 - Cauchy Problem}
\underline{Recap}:
\begin{itemize}
    \item Cauchy problem for heat equation (Logan 2.1)
    \item Goal: solve for $u(x,t)$
    \begin{align*}
        u_t - ku_{xx} = 0.\ k > 0 \\
        x \in \mathbb{R},\ t > 0 \tag{$\ast$}\\
        \text{IC: }\ u(x,0) = \phi(x)
    \end{align*}
    \item Decay conditions, e.g. $u,u_x \rightarrow 0 $ as $x \rightarrow \pm \infty$ \\
    Here $u$ is bounded, therefore $\phi(x)$ is bounded.
    \item The solution will look like $\phi(x)$ 'spreading out' over time.
\end{itemize}
How to solve the Cauchy problem?
\begin{enumerate}
    \item First find solution when $u(x,0)$ is a step-function.
    \item Take linear combinations \& translates of this solution to solve the general
    Cauchy problem $(\ast)$.
\end{enumerate}
So for the Cauchy problem above, we have 
\begin{enumerate}
    \item Step-function ICs
    \begin{align*}
        u_t - ku_{xx} = 0 \\
        x \in \mathbb{R}, t > 0  \\
        u(x,0) = \begin{cases}
            \alpha & x > 0 \\
            0 & x < 0
        \end{cases}
    \end{align*}
    Use \underline{dimensional analysis} as with last lecture, we continue with 
    \begin{align*}
        \frac{u(x,t)}{\alpha} = f\left(\frac{x}{\sqrt{4kt}}\right) 
    \end{align*}
    \underline{Remarks}:
    \begin{itemize}
        \item Any choice of dimensions for $u, x, t$ would give us this equation above.
        \item Having obtained this expression, we 'forget' about dimensions and set $\alpha = 1$.
        \begin{align*}
            \Rightarrow u(x,t) &= f\left(\frac{x}{\sqrt{4kt}}\right) \\
            &= f(z) \\
            \text{for } z(x,t) &= \frac{x}{\sqrt{4kt}}
        \end{align*}
        \item Trick: substitute $u(x,t) = f(z)$ into heat eq. and hope that we get an ODE for $f$
        \item We don't know if this will work, the above argument is slippery, full explanation 
        involves 'symmetry.'
    \end{itemize}
    Now we transform the heat PDE into an ODE, recall that we have 
    \begin{align*}
        u_t - ku_{xx} &= 0 \\
        u(x,0) &= \begin{cases}
            1 & x > 0 \\
            0 & x < 0
        \end{cases}
    \end{align*}
    We look for solutions of the form $u(x,t) = f(z(x,t))$ where 
    $z(x,t) = \displaystyle\frac{x}{\sqrt{4kt}}$.\\ 
    We compute that 
    \begin{align*}
        u_t &= f'(z) \frac{\partial z }{\partial t} \\
        &= -\frac 12 \cdot \frac{x}{\sqrt{4kt^3}}f'(z) \\
        &= \frac{-z}{2t} f'(z) \\
        u_x &= f'(z) \frac{\partial z}{\partial x} \\
        &= \frac{1}{\sqrt{4kt}}f'(z) \\
        u_{xx} &= \frac{1}{4kt}f''(z)
    \end{align*}
    So in all we have 
    \begin{align*}
        \frac{-z}{2t} f'(z) - k\frac{1}{4kt}f''(z) &= 0 \\
    \frac{-z}{2t} f'(z) - \frac{1}{4t}f''(z) &= 0 \\
        f'' + 2zf' &= 0
    \end{align*}
    We've reduced the PDE to an ODE for $f(z)$. If we got an expression which involved other
    functions of $x,t$, we would NOT have an ODE. \\ \\ The solutions of the ODE above is
    \begin{align*}
        f'' + 2zf' &= 0 \\
        \text{Set }h(z) &= f'(z) \\
        \frac{dh}{dz} &= -2zh \\
        \int \frac 1h dh &= \int -2 zdz \\
        \ln h &= -z^2 + c \\
        h(z) &= c_1e^{-z^2} \\
        f'(z) &= c_1e^{-z^2} \\
        f(z) &= c_1 \int_{0}^{z}e^{-s^2}ds + c_2 
    \end{align*}
    We plug $z = \frac{x}{\sqrt{4kt}}$ into the expression above
    \begin{align*}
        u(x,t) = \int_{0}^{\frac{x}{\sqrt{4kt}}} e^{-s^2}ds + c_2 \tag{$\ast \ast$}
    \end{align*}
    Now we use the IC $u(x,0) = \begin{cases}
        1 & x > 0 \\
        0 & x < 0
    \end{cases}$
    Fix $x > 0$ and send $t \rightarrow 0$ in $(\ast \ast)$ so that 
    $\frac{x}{\sqrt{4kt}} \rightarrow \infty$. Now we have the following 
    \begin{align*}
        1 = u(x,0) = c_1 \int_{0}^{\infty} e^{-s^2} ds + c_2
    \end{align*}
    Now fix $x < 0$ and send $t \rightarrow 0$ so that $\frac{x}{\sqrt{4kt}} \rightarrow - \infty$.
    We have the following 
    \begin{align*}
        0 = u(x,0) = c_1 \int_{0}^{-\infty} e^{-s^2} ds + c_2
    \end{align*}
    The key result here is 
    \begin{align*}
        \int_{0}^{\infty} e^{-s^2} ds = \int_{-\infty}^{0} e^{-s^2} ds = \frac{\sqrt{\pi}}{2} = \frac{1}{2} \int_{-\infty}^{\infty} e^{-s^2}ds
    \end{align*}
    This means our IC becomes
    \begin{align*}
        c_1 \frac{\sqrt \pi}{2} + c_2 &= 1 \\
        -c_1 \frac{\sqrt{\pi}}{2} + c_2 &= 0 \\
        c_2 &= \frac 12 \\
        c_1 = \frac{1}{\sqrt \pi}
    \end{align*}
    \textbf{Definition}. The error function $\text{erf}(z)$ is defined by 
    \begin{align*}
        \text{erf}(z) &= \frac{2}{\sqrt{\pi}} \int_{0}^{z}e^{-s^2}ds \\
        \Rightarrow \text{erf}(0) &= 0 \\
        \Rightarrow \lim_{z \rightarrow\infty} \text{erf}(z) &= 1 \\
        \Rightarrow u(x,t) &= \frac 12 \left( 1 + \text{erf}\left(\frac{x}{\sqrt{4kt}}\right)\right)
    \end{align*}
    This gives us the solution to the heat equation $u_t - ku_{xx} = 0$ for $u(x,0) = \begin{cases}
        1 & x > 0 \\
        0 & x < 0
    \end{cases}$
    \item Now that we solved it for the special case of $u(x,0)$ being a step function, how do
    we solve the general Cauchy problem with $u(x,0) = \phi(x)$?  \\ \\ 
    \underline{Lemma}. Suppose $v(x,t)$ solves the heat equation. Then the following are also 
    solutions.
    \begin{itemize}
        \item $v(x-y,t)$ for any $y$.
        \item $a_1v(x-y_1,t) + a_2v(x-y_2,t)$ for any $a_1, a_2, y_1, y_2$.
        \item $\int_{a}^{b} v(x-y,t)\psi(y)dy$ for any $a,b$, function $\psi$.
        \item $v_x(x,t)$
    \end{itemize}
    Let $F(x,t)$ be the solution with the step function IC.
    \begin{align*}
        \Rightarrow F(x,t) = \frac 12 (1 + \text{erf}(\frac{x}{\sqrt{4kt}}))
    \end{align*}
    Goal: use above moves to transform $F$ into a solution of $(\ast)$. We will start small and 
    go with a translation of the step function.
    \begin{align*}
        F(x - \epsilon, t) \\
        u(x,0) = \begin{cases}
            \alpha & x > \epsilon \\
            0 & x < \epsilon
        \end{cases}
    \end{align*}
    Next, consider the difference
    \begin{align*}
        F(x + \frac{\epsilon}{2},t) - F(x - \frac \epsilon 2, t)
    \end{align*}
    This gives a rectangle, with width $\epsilon$, height 1, centered at 0. Next, we try
    \begin{align*}
        \sum_{n = - \infty}^{\infty} \phi(n\epsilon) \left[F(x-n\epsilon + \frac{\epsilon}{2},t) - F(x-n\epsilon - \frac{\epsilon}{2},t)\right]
    \end{align*}
    This gives a series of rectangles, with width $\epsilon$, height $\phi(n\epsilon)$. 
    Think Riemann sums. This is so cool! The solution is 
    \begin{align*}
        \sum_{n = - \infty}^{\infty} \phi(n\epsilon) \left[F(x-n\epsilon + \frac{\epsilon}{2},t) - F(x-n\epsilon - \frac{\epsilon}{2},t)\right] \\
        \approx \epsilon F_x(x-n\epsilon, t) \\
        \rightarrow \int_{-\infty}^{\infty} F_x ( x-y,t) \psi(y)dy
    \end{align*}
\end{enumerate}

\subsection{Lecture 9 - Arbitrary Initial Conditions}
\underline{Recap}: 
\begin{itemize}
    \item Heat equation with step-function IC
    \begin{align*}
        u_t - ku_{xx} = 0 \\
        x \in \mathbb{R},\ t > 0 \\
        u(x,0) = \begin{cases}
            1 & x > 0 \\
            0 & x < 0
        \end{cases}
    \end{align*}
    \item For solns, try $u(x,t) = f(z)$ for $z(x,t) = \frac{x}{\sqrt{4kt}}$ \\
    $\Rightarrow f'' + 2zf' = 0$
    \item Soln 
    \begin{align*}
        F(x,t) = \frac 12 (1 + \text{erf}(\frac{x}{\sqrt{4kt}}))
    \end{align*}
    where 
    \begin{align*}
        \text{erf}(z) &= \frac{2}{\sqrt{\pi}}\int_{0}^{z}e^{-s^2}ds \\
        \text{erf}(0) &= 0 \\
        \lim_{z \rightarrow \pm \infty} \text{erf}(z) &= \pm 1
    \end{align*}
    \item If $v(x,t)$ is a soln to heat eq, then $v_x(x,t),\ v(x-y,t),\ \int v(x-y,t)\psi(y)dy$ 
    are also solns to the heat equation.
\end{itemize}
\underline{Goal}: 
\begin{align*}
    \text{Solve }\ u_t - ku_{xx} = 0 \\
    x \in \mathbb{R},\ t > 0 \\
    u(x,0) = \phi(x)\quad \text{(bounded)}
\end{align*}
\underline{Strategy}: 
\begin{center}
    Step-function IC \\
    $\downarrow$ \\
    'Point source' IC \\
    $\downarrow$ \\
    General IC
\end{center}
So the following are solutions
\begin{center}
    $F(x,t)$ \\
    $\downarrow$ \\
    $F(x-\frac{\epsilon}{2},t)$ \\
    $\downarrow$ \\
    $F(x+\frac \epsilon 2 , t) - F(x-\frac \epsilon 2, t)$ \\
    $\downarrow$ \\
    $\epsilon F_x(x,t)$
\end{center}
where $F_x(x,t)$ is a solution for a 'point source.' \\ \\
The IC $\phi$ can be thought of as a set of point sources of strength $\phi(y)$ at each 
$y \in \mathbb{R}$.
\begin{align*}
    \int_{-\infty}^{\infty} \phi(y)F_x(x-y,t)dy 
\end{align*}
\underline{Remark}: The point source function is also called the Dirac delta function
\begin{align*}
    \delta(x) = \begin{cases}
        0 & x \ne 0 \\
        1 & x = 0
    \end{cases}\ 
    \text{ s.t. } \int_{-\infty}^{\infty} \delta(x)dx = 1
\end{align*}
Let $G(x,t) = F_x(x,t)$
\begin{align*}
    \Rightarrow G(x,t) = \partial_x \left[\frac 12 + \frac 12 \text{erf}\left(\frac{x}{\sqrt{4kt}}\right)\right]
\end{align*}
Recall that $\text{erf}(z) = \frac{2}{\sqrt{\pi}}\int_{0}^{z}e^{-s^2}ds$
\begin{align*}
    \Rightarrow G(x,t) = \frac{1}{\sqrt{4\pi k t}}\exp\left( \frac{-x^2}{4kt} \right) 
\end{align*}
$G$ here is called the \underline{heat kernel} or the fundamental solution to the heat equation. \\ \\
The solution to the Cauchy problem 
\begin{align*}
    u_t - ku_{xx} &= 0 \\
    x\in \mathbb{R},\ t &> 0 \\
    u(x,0) &= \phi(x) \text{ (bounded)}
\end{align*}
is 
\begin{align*}
    u(x,t) &= \int_{-\infty}^{\infty} G(x-y,t)\phi(y)dy \\
    &= \int_{-\infty}^{\infty} \frac{1}{\sqrt{4\pi kt}}e^{-\frac{(x-y)^2}{4kt}}\cdot \phi(y)dy
\end{align*}
We can simplify this by substituting 
\begin{align*}
    s &= \frac{x-y}{\sqrt{4kt}} \\
    ds &= \frac{-dy}{\sqrt{4kt}}
\end{align*}
gives the equivalent formula
\begin{align*}
    u(x,t) = \frac{1}{\sqrt{\pi}} \int_{-\infty}^{\infty} e^{-s^2}\phi(x-s\sqrt{4kt})ds
\end{align*}
\underline{Remark}:
\begin{itemize}
    \item The above expression is called the Poisson integral representation.
    \item The solution is an integral, often with no closed form.
\end{itemize}
\boxed{\text{ex}} Solve the following and give the solution in terms of erf.
\begin{align*}
    u_t - ku_{xx} &= 0 \\
    x\in \mathbb{R},\ t &> 0 \\
    \text{with } u(x,0) &= \phi(x) = \begin{cases}
        e^{\lambda x} & x < 0 \\
        0 & x > 0 
    \end{cases}
\end{align*}
Using the Poisson integral above, we get 
\begin{align*}
    u(x,t) &= \frac{1}{\sqrt{\pi}} \int_{-\infty}^{\infty} e^{-s^2}\phi(x-s\sqrt{4kt})ds \\
    &= \frac{1}{\sqrt{\pi}} \int_{\frac{x}{\sqrt{4kt}}}^{\infty} e^{-s^2} e^{\lambda x} e^{-\lambda s\sqrt{4kt}}ds
\end{align*}
The limits are values of $s$ for which $\phi(x-s\sqrt{4kt})$ is $\ne 0$
\begin{align*}
    \Rightarrow x - s\sqrt{4kt} < 0
    \Rightarrow u(x,t) = \frac{e^{\lambda x}}{\sqrt \pi} \int_{\frac{x}{\sqrt{4kt}}}^{\infty} e^{-(s^2+2\lambda s\sqrt{kt})}ds
\end{align*}
Complete the square
\begin{align*}
    s^2 + 2s\lambda \sqrt{kt} = (s+\lambda\sqrt{kt})^2 - \lambda^2kt
\end{align*}
Then we have 
\begin{align*}
    u(x,t) &= \frac{e^{\lambda x}}{\sqrt{\pi}} \int_{\frac{x}{\sqrt{4kt}}}^{\infty} e^{-(s+\lambda\sqrt{kt})^2}e^{\lambda^2kt}
\end{align*}
Substitute $\theta = s + \lambda \sqrt{kt}$, $d\theta = ds$
\begin{align*}
    \Rightarrow u(x,t) = \frac{e^{\lambda^2kt+\lambda x}}{\sqrt \pi} \int_{\frac{x}{\sqrt{4kt}}}^{\infty}e^{-\theta^2}d\theta
\end{align*}
Note that 
\begin{align*}
    \text{erf}(b) &= \frac{2}{\sqrt \pi} \int_{0}^{b}e^{-s^2}ds \\
    \frac{2}{\sqrt \pi} \int_{0}^{b} e^{-s^2} ds + \frac{2}{\sqrt{\pi}} \int_{b}^{\infty}e^{-s^2}ds &= 1 \\
    \text{erf}(b) + \frac{2}{\sqrt{\pi}} \int_{b}^{\infty} e^{-s^2}ds &= 1 \\
    \Rightarrow \frac{2}{\pi} \int_{b}^{\infty} e^{-s^2} ds = 1 - \text{erf}(b)
\end{align*}
Therefore 
\begin{align*}
    u(x,t) = \frac{1}{2} e^{\lambda^2kt+\lambda x}\left(1-\text{erf}\left(\frac{x}{\sqrt{4kt}}+\lambda\sqrt{kt}\right)\right)
\end{align*}
\boxed{\text{ex}} Solve the following and give the solution in terms of erf.
\begin{align*}
    u_t - ku_{xx} &= 0 \\
    x\in \mathbb{R},\ t &> 0 \\
    u(x,0) = \phi(x) &= \begin{cases}
        x & 0 < x < 1 \\
        0 & \text{otherwise}
    \end{cases} 
\end{align*}
The solution is 
\begin{align*}
    u(x,t) = \frac{1}{\sqrt{\pi}} \int_{-\infty}^{\infty} e^{-s^2} \phi ( x - s\sqrt{4kt}) ds 
\end{align*}
\begin{align*}
    \phi \ne 0 \text{ between} &\\
    x - s \sqrt{4kt} = 0 \Rightarrow s &= \frac{x}{\sqrt{4kt}} = b \\
    x - s \sqrt{4kt} = 1 \Rightarrow s &= \frac{x-1}{\sqrt{4kt}} = a
\end{align*}
\begin{align*}
    \Rightarrow u &= \frac{1}{\sqrt{\pi}} \int_{a}^{b} e^{-s^2} (x - s\sqrt{4kt})ds \\
    &= \frac{1}{\sqrt{\pi}} \left[x\int_{a}^{b} e^{-s^2} ds - \sqrt{4kt} \int_{a}^{b}se^{-s^2} ds  \right] \\
    &= \frac{1}{\sqrt{\pi}} \left[x\cdot \frac{\sqrt{\pi}}{2}(\text{erf}(b)-\text{erf}(a)) - \sqrt{4kt}[\frac{-1}{2}e^{-s^2}]^b_a \right] \\
    &= \frac{x}{2}(\text{erf}(b)-\text{erf}(a)) + \sqrt{\frac{kt}{\pi}}(e^{-b^2} - e^{-a^2})
\end{align*}
for $\displaystyle a = \frac{x-1}{\sqrt{4kt}},\ b = \frac{x}{\sqrt{4kt}}$ \\ \\ \\ 
\underline{Remarks}:
\begin{itemize}
    \item The solutions here have infinite propagation speed, which is unphysical. Even if $u(x,0) = 0$,
    we can have $u(x,t) \ne 0$, for small $t$, and $x \rightarrow \infty$. 
    \item These solutions conserve mass. Suppose $u(x,t)$ is integrable and let 
    \begin{align*}
        M(t) = \int_{-\infty}^{\infty} u(x,t) dx 
    \end{align*}
    then \begin{align*}
        \frac{dM}{dt} &= \int_{-\infty}^{\infty} u_t dx \\
        &= k\int_{-\infty}^{\infty} u_{xx}dx \\
        &= k \left[u_x\right]^\infty_{-\infty} \\
        \Rightarrow \frac{dM}{dt} &= 0 \text{ if } u_x \rightarrow 0 \text{ as } x \rightarrow \pm \infty \\
        \Rightarrow M &\text{ is constant.}
    \end{align*}
\end{itemize}

\subsection{Lecture 10 - Unbounded Wave Equation}
\underline{Recap}:
\begin{itemize}
    \item Heat equation on $\mathbb{R}$
    \begin{align*}
        u_t - ku_{xx} = 0 \\
        k > 0,\ t > 0 \\
        x \in \mathbb{R} \\
        u(x,0) = \phi(x)\\ 
        \text{ (bounded)}
    \end{align*}
    \item Solution to the heat equation with arbitrary initial condition.
    \begin{align*}
        u(x,t) = \int_{-\infty}^{+\infty} \frac{1}{\sqrt{4\pi kt}} e^{-\frac{(x-y)^2}{4kt}} \phi(y)dy
    \end{align*}
    \item Solution describes $u(x,t)$ 'spreading out' or diffusing
    \item If $u_x \rightarrow 0$ as $|x| \rightarrow \infty$ then 
    total mass 
    \begin{align*}
        M(t) = \int_{-\infty}^{\infty} u(x,t)dx
    \end{align*}
    is conserved ($\frac{dM}{dt} = 0$)
\end{itemize}
\textbf{Cauchy problem for wave equation.} (Logan 2.2) \\
The wave equation for $u(x,t)$ is 
\begin{align*}
    u_{tt} - c^2u_{xx} = 0 \\
    c > 0 
\end{align*}
This has one space dimension, or 1d. \\ \\ 
We can find the general solution more directly than for the heat equation. \\
\underline{Method 1}:
\begin{align*}
    \text{Let } \xi &= x - ct \\
    \eta &= x + ct
\end{align*}
This is the same transformation as in HW3, where we write $u(x(\xi,\eta), t(\xi,\eta))$ 
and find $u_{\xi\xi}$, etc. Here we write $u(\xi(x,t),\eta(x,t))$
\begin{align*}
    \Rightarrow u_x &= u_\xi \xi_x + u_\eta \eta_x \\
    &= u_\xi + u_\eta \\
    \Rightarrow u_{xx} &= \partial_\xi (u_\xi + u_\eta) \cdot \xi_x + \partial_\eta(u_\xi + u_\eta)\cdot \eta_x \\
    &= u_{xx} + 2u_{\xi\eta} + u_{\eta\eta}
\end{align*}
and 
\begin{align*}
    u_t = c^2u_{\xi\xi} - 2c^2u_{\xi\eta} + c^2u_{\eta\eta} \\
    \Rightarrow u_{tt} - c^2u_{xx} = -4c^2u_{\xi\eta}
\end{align*}
so 
\begin{align*}
    u_{tt} - c^2 u_{xx} = 0 \\
    \Leftrightarrow u_{\xi\eta} = 0 \\
\end{align*}
We now solve $u_{\xi\eta = 0}$
\begin{align*}
    u_\xi = f(\xi) \\
    \Rightarrow u(\xi,\eta) = \int f(\xi)d\xi + G(\eta)
\end{align*}
Notice that an integral of an arbitrary function is another arbitrary function
\begin{align*}
    \Rightarrow u(\xi,\eta) &= F(\xi) + G(\eta) \\
    \Rightarrow u(x,t) &= F(x-ct) + G(x+ct) \\
    \text{for arbitrary functions } F,\ G
\end{align*}
\underline{Method 2}: 
\begin{align*}
    \text{Let } L = \partial_t^2 - c^2\partial_x^2
\end{align*}
So $Lu = (\partial_t^2 - c^2\partial_x^2)u = u_{tt} - c^2u_{xx}$. We can factorize this 
differential operator.
\begin{align*}
    \text{Let } L_1 &= \partial_t - c\partial_x \\
    L_2 &= \partial_t + c\partial_x 
\end{align*}
\underline{Lemma}: $L = L_1L_2$. Note that $L_1L_2 u$ means apply $L_2$, then apply $L_1$.
\begin{align*}
    L_1L_2 &= L_1[(\partial_t + c\partial_x)u] \\
    &= L_1[u_t + cu_x] \\
    &= (\partial_t - c\partial_x)[u_t + cu_x] \\
    &= u_{tt} + cu_{xt} - cu_{tx} - c^2u_{xx} \\
    &= u_{tt} - c^2u_{xx} \\
    &= Lu
\end{align*}
And so $Lu = 0 \Leftrightarrow L_1 L_2 u = 0$. The coordinates $\xi,\eta$ from method 1 comes 
characteristics of 1st order PDEs
\begin{align*}
    L_1 v = 0 \\
    L_2 v = 0
\end{align*}
\underline{D'Alembert's solution of the wave equation}. The Cauchy problem/initial value problem 
for wave equation is 
\begin{align*}
    u_{tt} - c^2u_{xx} = 0 \\
    t > 0 \\
    x\in \mathbb{R} \\
    u(x,0) = \phi(x) \\
    u_t(x,0) = \psi(x)
\end{align*}
e.g. $u(x,t)$ is the height of an infinite string with initial displacement $\phi(x)$ and 
initial velocity $\psi(x)$. The solution is 
\begin{align*}
    u(x,t) = F(x-ct) + G(x+ct)
\end{align*}
We plug in the initial conditions to get
\begin{align*}
    u(x,0) &= F(x) + G(x) = \phi(x) \\
    u_t(x,0) &= -cF'(x) + cG'(x) = \psi(x)
\end{align*}
These can be solved for $F,G$ and we get 
\begin{align*}
    u(x,t) = \frac 12 (\phi(x-ct) + \phi(x+ct)) + \frac{1}{2c}\int_{x-ct}^{x+ct} \psi(s)ds
\end{align*}
\boxed{\text{ex}} Plucking an infinite string. Solve 
\begin{align*}
    u_{tt} - 4 u_{xx} &= 0 \\
    x \in \mathbb{R},\ t &> 0 \\
    u(x,0) = \phi(x) &= e^{-x^2} \\
    u_t(x,0) = \psi(x) &= 0
\end{align*}
So we have 
\begin{align*}
    u(x,t) &= \frac{1}{2}(\phi(x-ct) + \phi(x+ct)) \\
    &= \frac 12 (e^{-(x-2t)^2} + e^{-(x+2t)^2})
\end{align*}
Visually, notice that $e^{-(x-2t)^2}$ is a peak at location $x = 2t$, and as $t$ increases, 
this peak moves to the right. In contrast, $e^{-(x+2t)^2}$ is a left-moving peak. \\ \\ 
\boxed{\text{ex}} $u_{tt} - c^2u_{xx} = 0$
\begin{align*}
    u(x,0) &= 0 \\
    u_t(x,0) &= \psi(x) = \frac{1}{1+x^2}
\end{align*}
We use the D'Alembert solution again.
\begin{align*}
    u(x,t) &= \frac{1}{2c}\int_{x-ct}^{x+ct}\psi(s)ds \\
    &= \frac{1}{2c} \int_{x-ct}^{x+ct} \frac{1}{1+s^2}ds \\
    &= \frac{1}{2c} (\arctan(x+ct) - \arctan(x-ct))
\end{align*}
\underline{Causality}: \\
D'Alembert solution reveals a causal structure of the wave equation.
\begin{align*}
    u(x,t) = \frac 12 (\phi(x-ct)+\phi(x+ct)) + \frac{1}{2c} \int_{x-ct}^{x+ct} \psi(s)ds
\end{align*}
Suppose $\phi(x),\ \psi = 0$ for $x < a,\ x > b$, i.e. $\phi,\psi$ are non-zero only when 
$x \in [a,b]$. Then $u(x,t)$ is non-zero only if $x-ct\in[a,b]$ or $x+ct \in [a,b]$.
\begin{itemize}
    \item Signals propagate at speed $\pm c$ along lines
    \begin{align*}
        x - ct = \text{ const} \\
        x + ct = \text{ const}
    \end{align*}
    \item The area between $x+ct = a$ and $x-ct = b$ is called the region of influence of signal.
    \item The solution at a point $(x_0,t_0)$ has a domain of dependence which comes from 
    tracing characteristics backwards.
\end{itemize}
\underline{Energy}.
Just like the heat equation has a conserved mass, the wave equation has a conserved 
energy. Suppose $u_{tt} - c^2u_{xx} = 0$ with $u_t$ or $u_x \rightarrow 0$ as $|x| \rightarrow \infty$.
We can define energy as 
\begin{align*}
    E(t) = \int_{-\infty}^{\infty} \frac{1}{2}(u_t^2 + c^2u_x^2)dx
\end{align*}
Then $\frac{dE}{dt} = 0$, i.e. $E$ is conserved. \\ \\
\underline{Remarks}
\begin{itemize}
    \item Conservation of $E(t)$ is a good check on solutions (especially for simulations)
    \item Conserved quantities tell us how solutions behaves.
\end{itemize}
\underline{Proof}
\begin{align*}
    \frac{dE}{dt} &= \int_{\infty}^{\infty} (u_tu_{xx}+c^2u_xu_{xt})dx \\
    &= \int_{-\infty}^{\infty} (c^2u_tu_{xx} + c^2u_xu_{xt})dx \\
    &= c^2 \int_{-\infty}^{\infty} \partial_x(u_tu_x)dx \text{ by product rule} \\
    \Rightarrow \frac{dE}{dt} &= c^2[u_tu_x]^\infty_{-\infty} \\
    &= 0
\end{align*}
\underline{Summary}:
\begin{itemize}
    \item Heat: parabolic, signal spreads out, diffuses
    \item Wave: hyperbolic, signal propagates at speed $c$, coherently
    \item Laplace: elliptic, ?
\end{itemize}
What about Laplace's equation?
\begin{align*}
    u_{xx} + u_{yy} = 0 \text{ for } u(x,y)
\end{align*}
Let's try to formulate a Cauchy/initial value problem as with heat/wave equations with $y$ acting 
as time. So the Cauchy problem is 
\begin{align*}
    u_{xx} + u_{yy} &= 0\\
    x\in \mathbb{R},\ y &> 0\\
    u(x,0) &= \phi(x) \\
    u_y(x,0) &= \psi(x)
\end{align*}
As with the wave equation, we have $u(x,0)$ as the initial profile and $u_y(x,0)$ as the 
initial velocity.

\subsection{Lecture 11 - Heat Equation on Half-line}
\underline{Recap}:
\begin{itemize}
    \item Heat equation Cauchy problem
    \begin{align*}
        u_t - ku_{xx} = 0 \\
        x \in \mathbb{R} \\
        k, t > 0 \\
        u(x,0) = \phi(x)
    \end{align*}
    \item Wave equation Cauchy problem 
    \begin{align*}
        u_{tt} - c^2u_{xx} = 0 \\
        x \in \mathbb{R} \\
        t > 0 \\
        u(x,0) = \phi(x) \\
        u_t(x,0) = \psi(x)
    \end{align*}
\end{itemize}
Following from last lecture, we have the following Cauchy problem for Laplace's equation.
\begin{align*}
    u_{xx} + u_{yy} &= 0\\
    x\in \mathbb{R},\ y &> 0\\
    u(x,0) &= \phi(x) \\
    u_y(x,0) &= \psi(x)
\end{align*}
where $u(x,0)$ is the initial profile and $u_y(x,0)$ is the 
initial velocity. This is fine for zero IC,
\begin{align*}
    \text{Take }\ \psi(x) &= 0 \\
    \phi(x) &= \phi^0(x) = 0
\end{align*}
Note here that superscripts are labels, not exponents. We have 
\begin{align*}
    u(x,y) = u^0(x,y) = 0
\end{align*}
So zero IC gives zero solution, now we try 'almost zero' ICs.
\begin{align*}
    \psi(x) &= 0\\
    \phi(x) &= \phi^\lambda(x) = \lambda\cos(\frac x\lambda)
\end{align*}
Let $u^\lambda(x,y)$ be the solution for these ICs. We have $\lim_{\lambda \rightarrow 0} \phi^\lambda(x) = 0 = \phi^0(x)$.
So does $\lim_{\lambda \rightarrow 0} u^\lambda(x,y) = 0$? No! But why?
\begin{align*}
    u^\lambda(x,y) = \lambda\cos(\frac x\lambda)\cosh(\frac y\lambda)
\end{align*}
This solution blows up as $\lambda \rightarrow 0$. \\ \\
\underline{Remarks}:
\begin{itemize}
    \item The Cauchy problem (IVP) for Laplace's equation is ill-posed.
    \item Elliptic PDEs (like Laplace's equation) describe static, time-independent processes.
    \item A PDE with ICs or BCs is well-posed if solutions have 
    \begin{itemize}
        \item Existence
        \item Uniqueness
        \item Stability (solution depends continuously on ICs and BCs)
    \end{itemize}
    \item The Cauchy problems for the heat and wave equation is well-posed.
    \item Next: The solutions to the wave/heat equation let us solve more general problems
    \begin{itemize}
        \item Semi-infinite domains
        \item Inhomogeneous problems
    \end{itemize}
\end{itemize}
\textbf{Semi-infinite domains} (Logan 2.4). We've solved the wave/heat equations on domain $x \in \mathbb{R}$. We now solve PDEs on
the half-line $x \in (0,\infty)$.  \\
\boxed{\text{ex}} Consider the heat equation on the half-line
\begin{align*}
    u_t - ku_{xx} = 0 \\
    k,x,t > 0 \\
    u(x,0) = \phi(x) = xe^{-x^2} \\
    u(0,t) = 0 \\
\end{align*}
\underline{Method}: extend to an IVP on $\mathbb{R}$ using symmetry, whose solution on $x > 0$
is the desired solution. \\ \\ 
Consider the heat equation on $\mathbb{R}$
\begin{align*}
    v_t - kv_{xx} = 0 \\
    x\in \mathbb{R}, t > 0 \\
    u(x,0) = \begin{cases}
        \phi(x) & x > 0 \\
        -\phi(-x) & x < 0
    \end{cases} = p(x)
\end{align*} 
Here $p(x)$ is called the odd-extension of $\phi(X)$. The solution for $v(x,t)$ is
\begin{align*}
    v(x,t) = \frac{1}{\sqrt{\pi}} \int e^{-s^2}p(x-s\sqrt{4kt})ds
\end{align*}
The solution $v(x,t)$ satisfies the half-line PDE for $x > 0$
\begin{align*}
    \text{Check }\ &v_t - kv_{xx} = 0 \text{ for } x > 0\\
    &v(x,0) = p(x) = \phi(x) \text{ for }x > 0 \\
    &v(0,t) = 0 \ \forall t\ \text{ 'by symmetry'}
\end{align*}
So the solution to the half-line PDE is 
\begin{align*}
    u(x,t) = \frac{1}{\sqrt{\pi}} \int_{-\infty}^{\infty} e^{-s^2}p(x-s\sqrt{4kt})ds \\
    \text{for } x > 0
\end{align*}
\underline{Remark}: This works even for $\phi(0) \ne 0$ \\ \\
\textbf{Odd-even functions}. Let $f:\mathbb{R} \rightarrow \mathbb{R}$ be a function 
\begin{itemize}
    \item $f$ is an even function if 
    \begin{align*}
        f(x) = f(-x)
    \end{align*} 
    \item $f$ is an odd function if 
    \begin{align*}
        f(x) = -f(-x)
    \end{align*}
\end{itemize}
\underline{Properties}.
\begin{itemize}
    \item If $f$ is odd and continuous, then $f(0) = 0$.
    \item If $f$ is even and smooth, then $f'(0) = 0$
\end{itemize}
\underline{Definition}. Let $g:(0,\infty) \rightarrow \mathbb{R}$
\begin{itemize}
    \item The odd extension of $g$ is 
    \begin{align*}
        G_{\text{odd}}(x) = \begin{cases}
            g(x) & x > 0 \\
            -g(-x) & x < 0
        \end{cases}
    \end{align*}
    \item The even extension of $g$ is 
    \begin{align*}
        G_{\text{even}}(x) =\begin{cases}
            g(x) & x > 0 \\
            g(-x) & x < 0
        \end{cases}
    \end{align*}
\end{itemize}
\underline{Symmetry of PDEs}. \\
Suppose $u(x,t)$ is a solution of the heat or wave equation for $x \in \mathbb{R},\ t > 0$. 
If the initial data are odd functions, then $u(x,t)$ is odd wrt to $x$
\begin{align*}
    u(x,t) &= -u(-x,t) \\
    u(0,t) &= 0\ \text{ if } u \text{ is continuous}
\end{align*}
Similarly, if the initial data are even functions, then $u(x,t)$ is even wrt to $x$
\begin{align*}
    u_x(0,t) = 0 \ \forall t\ \text{ if } u \text{ is smooth}
\end{align*}
\underline{Solutions of half-line PDEs}. $x \in (0,\infty)$. \\
Idea: Extend PDE to $x \in \mathbb{R}$ by taking an odd or even extension of initial data, 
depending on BCs.
\begin{align*}
    \begin{matrix}
        \text{Dirichlet BCs} \\ u(0,t) = 0
    \end{matrix} 
    \rightarrow 
    \begin{matrix}
        \text{odd extension} \\ \text{of ICs}
    \end{matrix}
    \rightarrow
    \begin{matrix}
        \text{solve} \\ \text{take } x > 0
    \end{matrix} \\
    \begin{matrix}
        \text{Neumann BCs} \\ u_x(0,t) = 0
    \end{matrix} 
    \rightarrow 
    \begin{matrix}
        \text{even extension} \\ \text{of ICs}
    \end{matrix}
    \rightarrow
    \begin{matrix}
        \text{solve} \\ \text{take } x > 0
    \end{matrix}
\end{align*}
\boxed{\text{ex}} Consider the wave equation.
\begin{align*}
    u_{tt} - c^2u_{xx} = 0 \\
    x > 0,\ t > 0 \\
    u(x,0) = \phi(x) = \cos x, x > 0 \\
    u_t(x,0) = \psi(x) = 0, x > 0 \\
    u_x(0,t) = 0, t > 0
\end{align*}
The BC here is Neumann, so we use even extension on the ICs. Let $p(x),q(x)$ be the even 
extensions of $\phi(x), \psi(x)$. Then
\begin{align*}
    p(x) &= \begin{cases}
        \cos(x) & x > 0 \\
        \cos(-x) & x < 0
    \end{cases} \\ 
    &= \cos x \quad \forall x \in \mathbb{R} \\
    q(x) &= 0\quad x \in \mathbb{R}
\end{align*}
We use D'Alembert's solution 
\begin{align*}
    u(x,t) &= \frac 12 (p(x-ct) + p(x + ct)) \\
    &= \frac 12 (\cos(x-ct) + \cos(x+ct)) \\ 
    &= \cos(x)\cos(ct),\quad x \in \mathbb{R}
\end{align*}
The half-line solution is 
\begin{align*}
    u(x,t) = \cos x \cos ct \\
    x > 0
\end{align*}
Our BC is satisfied 
\begin{align*}
    u_x &= -\sin x \cos ct \\
    &= 0 \quad \text{for x = 0}
\end{align*}

\subsection{Lecture 12 - Inhomogeneous Wave/Heat Equation} 
Note on HW4, Q2a:
\begin{align*}
    u_t - ku_{xx} = 0 \\
    x \in \mathbb{R} \\
    k > 0, t > 0 \\
    u(x,0) = \exp\left(-\frac{(x-a)^2}{b^2}\right)
\end{align*}
Method 1: Poisson integral representation. \\
Method 2: Transform the heat kernel. \\
The heat kernel $G(x,t)$ satisfies heat equation
\begin{align*}
    G(x,t) &= \frac{1}{\sqrt{4\pi kt}}e^{\frac{-x^2}{4kt}} \\
    G_t - kG_{xx} &= 0
\end{align*}
At time $t = 0$, $G(x,t)$ is a point source. For $t > 0$, $G(x,t)$ is Gaussian. Since $G(x,t)$
satisfies the heat equation, we have 
\begin{itemize}
    \item $G(x+A,t)$ satisfies heat equation
    \item $G(x+A,t+B)$ satisfies heat equation
    \item $\lambda G(x+A,t+B)$ satisfies heat equation
\end{itemize}
This problem has a Gaussian IC, and the heat kernel describes a sequence of Gaussians.
To solve this problem, try the candidate solution
\begin{align*}
    u(x,t) &= \lambda G(x+A,t+B) \\
    &= \frac{\lambda}{\sqrt{4\pi k (t+B)}}\exp\left( -\frac{(x+A)^2}{4k(t+B)} \right)
\end{align*}
We need $u(x,0) = \exp\left(-\frac{(x-a)^2}{b^2}\right)$.
\begin{align*}
    \Rightarrow A = -a,\ B = \frac{b^2}{4k},\ \lambda &= \sqrt{4\pi k B} \\
    &= \sqrt{\pi}b
\end{align*}
\textbf{Inhomogeneous PDEs} \\
Consider the inhomogeneous heat equation with source $f(x,t)$
\begin{align*}
    u_t - ku_{xx} = f(x,t) \\
    x\in \mathbb{R},\ t,k > 0 \\
    u(x,0) = 0
\end{align*}
\underline{ODE Case}. Consider an ODE version of the above example, for $y(t)$ 
\begin{align*}
    y' - ky = t \\
    t > 0 \\
    y(0) = 0
\end{align*}
Note here that the source is $f(t) = t$. We can solve this ODE in many ways, e.g. integrating
factor.
\begin{align*}
    e^{-kt}y' - ke^{-kt}y &= te^{-kt} \\
    (e^{-kt}y)' &= te^{-kt} \\
    e^{-kt}y &= \int_{0}^{t}e^{-k\tau}\tau d\tau + c
\end{align*}
Note that $y(0) = 0 \Rightarrow C = 0$
\begin{align*}
    y(t) &= e^{kt} \int_{0}^{t}e^{-k\tau}\tau d\tau \\
    &= \int_{0}^{t} e^{k(t-\tau)}\tau d\tau
\end{align*}
The solution at time $t$ is the response to a source $f(\tau) = \tau$ for all $0 < \tau < t$.
This is equivalent to taking solutions of the homogeneous ODE ($y'-ky=0$) with IC $f(\tau) = \tau$
for $0 < \tau < t$ and superposing them. \\ \\
E.g. define $w(t;\tau)$ by 
\begin{align*}
    w' - kw = 0 \\
    \text{with } w(0;\tau) = \tau
\end{align*}
We solve for $w$
\begin{align*}
    \Rightarrow w(t;\tau) = Ae^{kt} = \tau e^{kt}
\end{align*}
The value $w(t-\tau;\tau)$ gives the effect of a source $f(\tau) = \tau$ at a time $t > \tau$. \\ \\
Superpose to find $y(t)$
\begin{align*}
    y(t) &= \int_{0}^{t} w(t -\tau;\tau) d\tau \\
    &= \int_{0}^{t}e^{k(t-\tau)}\tau d\tau
\end{align*}
This is Duhamel's principle, which converts inhomogeneous problem with zero IC to homogeneous
problem with non-zero IC. \\
\underline{PDE Case}. Consider the inhomogeneous heat equation.
\begin{align*}
    u_t - ku_{xx} = f(x,t) \\
    x \in \mathbb{R},\ k,t > 0 \\
    u(x,0) = 0
\end{align*}
First find solutions $w(x,t;\tau)$ of heat equation with IC $f(x,\tau)$
\begin{align*}
    w_t - kw_{xx} = 0 \\
    x \in \mathbb{R},\ k, t > 0 \\
    w(x,0; \tau) = f(x,\tau)
\end{align*}
Then solution is 
\begin{align*}
    u(x,t) = \int_{0}^{t} w(x, t-\tau;\tau)d\tau \tag{Duhamel's principle}
\end{align*}
We can write $w(x,t;\tau)$ in terms of the heat kernel $G(x,t) = \displaystyle\frac{1}{\sqrt{4\pi kt}} e^{\frac{-x^2}{4kt}}$.
\begin{align*}
    w(x,t;\tau)&= \int_{-\infty}^{\infty} G(x-y,t) f(y,\tau)dy \\ 
    \Rightarrow u(x,t) &= \int_{0}^{t}d\tau \int_{-\infty}^{\infty} G(x-y,t-\tau)f(y,\tau)dy
\end{align*}
\underline{Remarks}:
\begin{itemize}
    \item Rarely possible to evaluate this integral.
    \item Expression is useful for analyzing solution properties, or numerics.
    \item Intuition: point heat source at position $y$, time $\tau$, spreads out as $G(x-y,t-\tau)$
    \item Duhamel's solution due to source $f(x,t)$ = solution due to point sources of strength 
    $f(y,\tau)$ at $y,\tau$
\end{itemize}

\subsection{Lecture 12 - Series Expansion} 
\boxed{\text{ex}} Heat equation on a bounded domain in 1d.
\begin{align*}
    u_t = u_{xx} \tag{I}\\
    0 < x < \pi \\
    t > 0  \\
    \text{BC: } u(0,t) = u(\pi,t) = 0 \tag{II} \\
    \text{IC: } u(x,0) = f(x) \tag{III}
\end{align*}
Strategy:
\begin{enumerate}
    \item Observe that 
    \begin{align*}
        u_n(x,t) = e^{-n^2t}\sin(nx)
    \end{align*}
    Satisfies (I), (II) for $n = 1,2,3,\dots$
    \item Every solution to (I) and (II) is a superposition of $u_n$'s
    \begin{align*}
        u(x,t) = \sum_{n=1}^{\infty} a_n e^{-n^2t} \sin(nx) \\
        \text{for } a_n \in \mathbb{R}
    \end{align*}
    \item The IC determines the constants $a_n$
    \begin{align*}
        u(x,0) = f(x) = \sum_{n=1}^{\infty} a_n \sin(nx)
    \end{align*}
    All suitable functions $g(x)$ on $x \in [0,\pi]$ (domain) can be uniquely written as a sum of
    $\sin(nx)$ terms. 
\end{enumerate}
\underline{Remarks}:
\begin{itemize}
    \item We will flesh this out in the phase of class.
    \item A key method is expanding functions in terms of 'simpler' functions (e.g. steps 2 \& 3)
\end{itemize}
\textbf{Series expansions}. \\
\underline{Warm-up}: Taylor series. \\
The Taylor series of a function $f(x)$ about $x = 0$ is 
\begin{align*}
    f(x) = \sum_{n=0}^{\infty}c_nx^n = c_0 + c_1x + c_2x^2 + \dots
\end{align*}
\begin{itemize}
    \item The basis functions here are the functions in the expansion, i.e. $1,x,x^2,\dots$.
    \item The coefficients $c_n$ can be calculated 
    \begin{align*}
        c_n = \frac{f^{(n)}(0)}{n!}
    \end{align*}
    \item $f$ is any sufficiently smooth functions defined at 0.
\end{itemize}
\underline{Fourier series}. (Logan 3.2) \\
Let $f:[-\pi,\pi] \rightarrow \mathbb{R}$ (we will change domain of $f$ later). The Fourier 
series of $f$ is an expansion in terms of the following basis functions:
\begin{align*}
    1,\ \cos(x),\ \cos(2x),\dots,\ \sin(x),\ \sin(2x),\dots
\end{align*}
Fourier series for $f(x)$ is 
\begin{align*}
    f(x) = \frac{a_0}{2} + \sum_{n=1}^{\infty} (a_n\cos(nx) + b_n \sin(nx))
\end{align*}
The coefficients are given by 
\begin{align*}
    a_n &= \frac{1}{\pi} \int_{-\pi}^{\pi} f(x)\cos(nx)dx,\quad n = 0,1,2,\dots \\
    b_n &= \frac{1}{\pi} \int_{-\pi}^{\pi} f(x)\sin(nx)dx,\quad n = 1,2,3,\dots
\end{align*}
\subsection{Lecture 13 - Series Expansion cont.}
\underline{Fourier series}. \\
Let $f:[-\pi,\pi]\rightarrow \mathbb{R}$. The Fourier series for $f$ is an expansion in terms of 
the following basis functions.
\begin{align*}
    1 \text{ or } \cos(0\cdot x),\ \cos x,\ \cos 2x,\ \dots,\ \sin x,\ \sin 2x
\end{align*}
The Fourier series for $f(x)$ is 
\begin{align*}
    f(x) = \frac{a_0}{2} + \sum_{n=1}^{\infty}(a_n\cos nx + b_n \sin nx)
\end{align*}
\underline{Remark}: 
\begin{itemize}
    \item We can write Fourier series for more general domains (not just $[-\pi, \pi]$)
    \item We haven't yet said anything about when the series is valid.
\end{itemize}
The Fourier coefficients are given by
\begin{align*}
    a_n &= \frac{1}{\pi} \int_{-\pi}^{\pi} f(x)\cos(nx)dx,\quad n = 0,1,2,\dots \\
    b_n &= \frac{1}{\pi} \int_{-\pi}^{\pi} f(x)\sin(nx)dx,\quad n = 1,2,3,\dots
\end{align*}
\underline{Proof}: \\
We will first check the expression for $b_n$
\begin{align*}
    b_n = \frac 1\pi \int_{-\pi}^{\pi} f(x)\sin(nx)dx
\end{align*}
We can calculate this using Fourier series for $f(x)$. Calculating $\int_{-\pi}^{\pi}f(x)\sin(nx)dx$
requires integrals of the form
\begin{align*}
    \int_{-\pi}^{\pi}\cos(mx)\sin(nx)dx,\quad \int_{-\pi}^{\pi} \sin(mx)\sin(nx)dx \\
    \forall m,n
\end{align*}
First note that
\begin{align*}
    \int_{-\pi}^{\pi} \sin(mx)dx = 0 \quad \forall m \in \mathbb{Z}
\end{align*}
Check:
\begin{align*}
    \int_{-\pi}^{\pi}\sin(mx)dx &= -\frac{1}{m}\cos(mx) \Big|^\pi_{-\pi} \\
    &= -\frac{1}{m}[\cos(\pi m) - \cos(-\pi m)] \\
    &= 0\quad \text{ (since } \cos(x) = \cos (-x))
\end{align*}
Similarly
\begin{align*}
    \int_{-\pi}^{\pi} \cos(mx)dx = \begin{cases}
        0 & m \in \mathbb{Z},\ m > 0 \\
        2\pi & m = 0
    \end{cases}
\end{align*}
Also 
\begin{align*}
    \int_{-\pi}^{\pi} \cos(mx)\sin(nx) dx &= \int_{\pi}^{-\pi} \frac{1}{2} (\sin((n+m)x)-\sin((n-m)x))dx \\
    &= 0
\end{align*}
Similarly
\begin{align*}
    \int_{-\pi}^{\pi} \sin(mx)\sin(nx)dx &= 0 \ \text{ for } m\ne n\\
    &= \frac 12 [\cos((m-n)x) - \cos((m+n)x)] \\
    & m-n,\ m+n \ne 0,\ \text{if } m\ne n
\end{align*}
And 
\begin{align*}
    \int_{-\pi}^{\pi} \sin^2(nx)dx = \int_{-\pi}^{\pi} \cos^2(nx)dx = \pi
\end{align*}
We use these integrals to calculate $\int_{-\pi}^{\pi} f(x) \sin(nx)dx$
\begin{align*}
    &\int_{-\pi}^{\pi}f(x) \sin(nx)dx \\
    &= \int_{-\pi}^{\pi} \left[\frac{a_0}{2} + \sum_{m=1}^{\infty} (a_m \cos(mx) + b_m \sin(mx))\right]\sin(nx)dx \\
    &= \int_{-\pi}^{\pi} \frac{a_0}{2}\sin(nx)dx + \sum_{m=1}^{\infty} \left(\int_{-\pi}^{\pi}a_m \cos(mx)\sin(nx)dx + \int_{-\pi}^{\pi}b_m \sin(mx)\sin(nx)\right) \\
    &= 0 + \sum_{m=1}^{\infty} \left(0 + \int_{-\pi}^{\pi}b_m \sin(mx)\sin(nx)\right) \\
    &= \int_{-\pi}^{\pi} b_n\sin^2(nx) dx \\
    &= b_n \pi
\end{align*}
\underline{Remarks}:
\begin{itemize}
    \item Key properties
    \begin{align*}
        \int_{-\pi}^{\pi} \cos(mx)\sin(nx)dx &= 0 \\
        \int_{-\pi}^{\pi} \sin(nx)dx &= 0\\
        \int_{-\pi}^{\pi} \sin(mx)\sin(nx)dx &= \begin{cases}
            0 & m\ne n \\
            \pi & m = n
        \end{cases}
    \end{align*}
    \item Functions whose product integrate to 0 are called \underline{orthogonal}
\end{itemize}
\boxed{\text{ex}} Fourier series of a step-function on $[-\pi,\pi]$
\begin{align*}
    f(x) = \begin{cases}
        1 & 0 < x < \pi \\
        0 & -\pi < x < 0
    \end{cases}
\end{align*}
Goal: find $a_n, b_n$ such that
\begin{align*}
    f(x) &= \frac{a_0}{2} + \sum_{n=1}^{\infty} (a_n \cos(nx) + b_n\sin(nx)) \\
    a_n &= \frac 1\pi \int_{-\pi}^{\pi} f(x)\cos(nx)dx \quad n=0,1,2,\dots \\
    b_n &= \frac 1\pi \int_{-\pi}^{\pi} f(x)\sin(nx)dx \quad n=1,2,3,\dots 
\end{align*}
Calculate $a_0$
\begin{align*}
    a_0 &= \frac 1\pi \int_{-\pi}^{\pi}f(x)dx = \frac 1\pi \int_{0}^{\pi}1dx = 1
\end{align*}
For $n > 0$:
\begin{align*}
    a_n &= \frac 1\pi \int_{-\pi}^{\pi} f(x)\cos(nx) dx \\
    &= \frac 1\pi \int_{0}^{\pi}\cos(nx)dx \\
    &= \frac 1\pi \left[\frac 1n \sin(nx)\right]^\pi_0 \\
    &= 0 
\end{align*}
\begin{align*}
    b_n &= \frac 1\pi \int_{-\pi}^{\pi} f(x)\sin(nx)dx \\
    &= \frac 1\pi \int_{0}^{\pi} \sin(nx)dx \\
    &= \frac 1\pi \left[-\frac{1}{n}\cos(nx)\right]^\pi_0 \\
    &= \frac{1}{n\pi}(1-\cos(n\pi)) \\
    &= \frac{1}{n\pi}(1-(-1)^n) \\
    &= \begin{cases}
        0 & n \text{ even} \\
        \frac{2}{n\pi} & n \text{ odd}
    \end{cases}
\end{align*}
$\Rightarrow$ Fourier series for $f(x)$ is 
\begin{align*}
    f(x) = \frac 12 + \frac 2\pi(\sin x + \frac 13 \sin 3x + \frac 15 \sin 5x + \dots)
\end{align*}
\underline{Remarks}. As we take more terms, we get a better approximation for $f(x)$. \\ \\
\textbf{Function spaces}. (Logan 3.2). Writing $f(x)$ in terms of Fourier basis is analogous
to writing a vector in components. \\ \\
\boxed{\text{ex}} Consider $\mathbb{R}^3$. Let $e_1, e_2, e_3$ be the standard basis
\begin{align*}
    e_1 &= (1,0,0) \\
    e_2 &= (0,1,0) \\
    e_3 &= (0,0,1)
\end{align*}
This basis is \underline{orthogonal} since 
\begin{align*}
    e_1 \cdot e_2 = e_2 \cdot e_3 = e_1 \cdot e_3 = 0
\end{align*}
This basis is \underline{normalized} since 
\begin{align*}
    e_1 \cdot e_1 = e_2 \cdot e_2 = e_3 \cdot e_3 = 1
\end{align*}
We can 'expand' vector $v \in \mathbb{R}^3$ in terms of $e_1,e_2,e_3$
\begin{align*}
    v = a_1e_1 + a_2e_2 + a_3e_3 \\
    \text{where } a_1 = v\cdot e_1,\text{ etc.}
\end{align*}
Connection with Fourier series 
\begin{center}
    \begin{tabular}{c c c}
        vector & $\rightarrow$ & functions \\
        $v,w \in \mathbb{R}^3$ & & $f,g:[-\pi,\pi] \in \mathbb{R}$ \\ \\
        inner product & $\rightarrow$ & integral \\
        $v\cdot w$ & & $\int_{-\pi}^{\pi} f(x)g(x)dx$ \\ \\
       basis & $\rightarrow$ & Fourier basis \\
       $e_1,e_2,e_3$ & & $1,\cos x,\dots,\sin x,\dots$ \\ \\
       expansion & $\rightarrow$ & Fourier series \\
       $v = \sum a_i e_i$ & & $f(x) = \frac{a_0}{2} + \sum(a_n\cos nx + b_n \sin nx)$ \\
       $a_i = v\cdot e_i$ & & $a_n = \frac 1\pi(f,\cos nx),\ b_n = \frac 1\pi (f,\sin nx)$
    \end{tabular}
\end{center}
Notation: $(f,g) = \int_{-\pi}^{\pi}f(x)g(x)dx$ \\ \\ 
\underline{$L^2$-space}. The correspondence above works because functions and vectors obey the 
same properties - in fact, we say that functions are vectors. To fix context, consider functions
on a fixed domain $[a,b]$.  \\ \\
The function space $L^2[a,b]$ is the set of all functions $f:[a,b] \rightarrow \mathbb{R}$ 
such that $\int_{a}^{b}f(x)^2 dx < \infty$. This is called a square integrable function. \\ \\
Fourier series, and other series expansions, work for $L^2$ functions. $L^2$ includes smooth 
functions but also some non-continuous functions like step-functions, etc. Typically, we only 
see $L^2$ functions. $L^2$ has the structure to understand series expansions.

\subsection{Lecture 14 - Function Spaces}
\underline{Recap}:
\begin{itemize}
    \item Fourier series for functions $f:[-\pi,\pi] \rightarrow \mathbb{R}$
    \item $L^2$ function spaces
    \item Fourier series are analogous to expanding vectors in terms of basis vectors.
    The Fourier basis for functions on $[-\pi,\pi]$ is $1,\sin(nx),\cos(nx),\ n = 1,2,$
    \item Consider functions on a fixed domain $[a,b]$, the space of functions $L^2[a,b]$
    is the set of all $f:[a,b]\rightarrow\mathbb{R}$ such that 
    \begin{align*}
        \int_{a}^{b}f(x)^2dx < \infty
    \end{align*}
    This includes all 'typical' functions.
    \item Fourier series 'work' on $L^2$ spaces.
\end{itemize}
\textbf{Vector operations}. $L^2[a,b]$ is a vector space which means that
\begin{itemize}
    \item $\forall f,g \in L^2[a,b]$ and 
    $\forall\lambda \in \mathbb{R}$, we have 
    \begin{align*}
       f+g &\in L^2[a,b]\quad (\text{addition}) \\
        \lambda f &\in L^2[a,b]\quad (\text{scalar mult.})
    \end{align*}
    Intuition: addition, scalar multiplication for functions is same as vectors.
    \item $L^2[a,b]$ has an inner product
    \begin{align*}
        (f,g) = \int_{a}^{b}f(x)g(x)dx \in \mathbb{R}
    \end{align*}
    Note:
    \begin{align*}
        (f,\lambda g) &= \lambda(f,g),\quad \forall \lambda \in \mathbb{R} \\
        (f,g_1+g_2) &= (f,g_1) + (f,g_2),\quad \forall g_1,g_2,f \in L^2[a,b]
    \end{align*}
    \item The \underline{norm} of $f$ is 
    \begin{align*}
        ||f|| = \sqrt{(f,f)} = \left(\int_{a}^{b}f(x)^2\right)^\frac{1}{2}
    \end{align*}
    \item $f,g$ are \underline{orthogonal} if 
    \begin{align*}
        (f,g) = 0
    \end{align*}
\end{itemize}
\boxed{\text{ex}} Consider $f(x) = x, g(x) = 1+x^2$ for $x \in [-1,1]$
\begin{align*}
    \Rightarrow f,g &\in L^2[-1,1] \\
    ||f||^2 &= (f,f) = \int_{-1}^{+1}x^2dx = \left[\frac 13 x^3\right]^{1}_{-1} = \frac{2}{3} \\
    (f,g) &= \int_{-1}^{1}x(1+x^2)dx = \left[\frac 12 x^2 + \frac 14 x^4\right]^1_{-1} \\
    &= 0 \\
    \Rightarrow f,g &\text{ are orthogonal}
\end{align*}
\underline{Remark}:
\begin{itemize}
    \item $||f|| = 0 \Leftrightarrow f = 0$
    \item Fourier series are nice because the basis functions are orthogonal. \\
    Fourier basis on $[-\pi,\pi]$
    \begin{align*}
        1,\cos x, \cos 2x, \dots, \sin x, \sin 2x, \dots\ \in L^2[-\pi,\pi]
    \end{align*}
    Then 
    \begin{align*}
        (1,\cos nx) = (1,\sin nx) &= 0,\ \forall n > 0 \\
        (\cos nx, \sin mx) &= 0,\ \forall m,n > 0 \\
        (\cos nx, \cos mx) &= 0,\ \forall m\ne n \\
        (\sin nx, \sin mx) &= 0,\ \forall m\ne n
    \end{align*}
    $\Rightarrow$ Every pair is orthogonal
\end{itemize}
\textbf{Orthogonal expansions} (Logan 3.2, 3.3). \\
Define a set of functions 
\begin{align*}
    f_1, f_2,\dots \in L^2[a,b] 
\end{align*}
is an orthogonal set if 
\begin{align*}
    (f_i,f_j) &= 0 \quad \forall i\ne j \\
    (f_i,f_i) &\ne 0 \quad \forall i\ (\text{i.e. } f_i \ne 0)
\end{align*}
For example, the Fourier basis is an orthogonal set. \\ \\
\underline{Definition}: An orthogonal set $f_1,f_2,\dots \in L^2[a,b]$ is \underline{complete} if 
every $g \in L^2[a,b]$ can be written 
\begin{align*}
    g(x) &= c_1f_1(x) + c_2f_2(x) + \dots \\
    &= \sum_{n=1}^{\infty} c_n f_n(x) \quad c_n \in \mathbb{R} \
\end{align*}
The $c_n$'s can then be found with inner products
\begin{align*}
    (g,f_n) &= \left(\sum_{i=1}^{\infty} c_i f_i, f_n\right) \\
    &= \sum_{i=1}^{\infty} (c_i f_i, f_n) \\
    \Bigg[\text{since } \int_{a}^{b} f(g_1+g_2)dx &= \int_{a}^{b} (fg_1 + fg_2)dx \ \forall f,g_1,g_2 \in L^2[a,b]\Bigg] \\
    &= \sum_{i=1}^{\infty} c_i(f_i,f_n) \\
    \big[\text{since } (f_i,f_n) &= 0 \text{ for } i\ne n\big] \\
    &= c_n (f_n,f_n) \\
    \Rightarrow (g,f_n) &= c_n ||f_n||^2 \\
    c_n &= \frac{(g,f_n)}{||f_n||^2}\ \forall n
\end{align*}
\underline{Remarks}:
\begin{itemize}
    \item Complete means that the series is valid
    \item Orthogonal means that the coefficients $c_n$'s can be calculated with 
    inner products.
    \item An orthogonal set which is complete is called a complete orthogonal set (c.o.s)
    (same as orthogonal basis for our purposes).
\end{itemize}
\boxed{\text{ex}} The Fourier basis on $[-\pi,\pi]$ is an orthogonal basis. For $f\in[-\pi,\pi]$
\begin{align*}
    f &= \frac{a_0}{2} + \sum_{n=1}^{\infty}(a_n\cos nx + b_n \sin nx) \\
    \text{with } (f,1) &= (\frac{a_0}{2},1) = \int_{-\pi}^{\pi}\frac{a_0}{2}\cdot 1 dx = a_0 \pi
\end{align*}
For $n \ge 1$ 
\begin{align*}
    (f,\cos nx) &= a_n (\cos nx, \cos nx) = a_n \pi \\
    (f,\sin nx) &= b_n (\sin nx, \sin nx) = b_n \pi
\end{align*}
\underline{Fourier series on $[-L,L]$} \\
The Fourier basis on $[-L,L]$ is 
\begin{align*}
    1, \cos\left(\frac{n\pi x}{L}\right), \sin\left(\frac{n\pi x}{L}\right)\quad n = 1,2,\dots
\end{align*}
This is an orthogonal basis (c.o.s). The Fourier series for $f:[-L,L] \rightarrow \mathbb{R}$ is 
\begin{align*}
    f(x) &= \frac{a_0}{2} + \sum_{n=1}^{\infty}\left(a_n\cos\left(\frac{n\pi x}{L}\right) + b_n \sin\left(\frac{n\pi x}{L}\right)\right) \\
    a_n &= \frac 1L\left(f,\cos\left(\frac{n\pi x}{L}\right)\right) \quad n = 0,1,2,\dots \\
    b_n &= \frac 1L\left(f,\sin\left(\frac{n\pi x}{L}\right)\right) \quad n = 1,2,3,\dots
\end{align*}
\boxed{\text{ex}} Find Fourier series of $f(x) = x,\ x \in [-L,L]$. We have 
\begin{align*}
    a_0 = \frac 1L (f,1) = \frac 1L \int_{-L}^{L} xdx = 0
\end{align*}
For $n > 0$
\begin{align*}
    a_n &= \frac 1L \int_{-L}^{L}x\cos\left(n\pi x\right)dx \\
    &= \frac 1L \left[x\cdot \frac{L}{n\pi}\sin \left(\frac{n\pi x}{L}\right)\right]^L_{-L} - \frac{1}{L}\int_{-L}^{L} \frac{L}{n\pi} \sin\left(\frac{n\pi x}{L}\right) dx \\
    &= \frac{L}{n\pi} \left(\sin(n\pi) - \sin(-n\pi)\right) \\
    &= 0 \\
\end{align*}
\begin{align*}
    b_n &= \frac 1L \int_{-L}^{L} x\sin \left(\frac{n\pi x}{L}\right)dx \\
    &= \frac 1L \left[x \cdot \frac{-L}{n\pi}\cos\left(\frac{n\pi x}{L}\right)\right] - \frac{1}{L} \int_{-L}^{L} \frac{-L}{n\pi} \cos\left(\frac{n\pi x}{L}\right)dx \\
    &= \frac{-1}{n\pi} \left[L\cos(n\pi) + L \cos (-n\pi)\right] \\
    &= \frac{-2L}{n\pi} \cos(n\pi) \leftarrow (-1)^n \\
    &= \frac{2L}{n\pi}(-1)^{n+1} \\
    \Rightarrow f(x) &= \frac{2L}{\pi}\left(\sin\left(\frac{\pi x}{L}\right)-\frac 12 \sin\left(\frac{2\pi x}{L}\right)+\dots\right) \\
    &= x
\end{align*}
\underline{Remarks}:
\begin{itemize}
    \item $\cos$ is even, $\sin$ is odd
    \item Odd $\times$ even is an odd function 
    \item Odd functions integrate to 0 over $[-L,L]$
    \item $f$ is odd $\Leftrightarrow a_n = 0\ \forall n$ (no $\cos$)
    \item $f$ is even $\Leftrightarrow b_n = 0\ \forall n$ (no $\sin$)
    \item Above $f(x)=x$ is odd $\Rightarrow a_n = 0$.
    \item The Fourier series extends to periodic functions on $\mathbb{R}$ with period $2L$
\end{itemize}
Any functions $g:[0,2L]\rightarrow \mathbb{R}$ also extends to a $2L$-periodic function on 
$\mathbb{R}$ and so can be expanded with the same Fourier basis. \\ \\
\underline{Fourier series on $[0,2L]$} \\
Fourier basis 
\begin{align*}
    1,\cos\left(\frac{n\pi x}{L}\right),\sin\left(\frac{n\pi x}{L}\right)\quad n =1,2,\dots
\end{align*}
A function $g:[0,2L]\rightarrow \mathbb{R}$ has Fourier series 
\begin{align*}
    g(x) = \frac{a_0}{2} + \sum_{n=1}^{\infty}\left(a_n\cos\left(\frac{n\pi x}{L}\right)+b_n\sin\left(\frac{n\pi x}{L}\right)\right)
\end{align*}
where 
\begin{align*}
    a_n &= \frac 1L\left(g,\cos\left(\frac{n\pi x}{L}\right)\right) = \frac 1L \int_{0}^{2L} g(x)\cos\left(\frac{n\pi x}{L}\right)dx \quad n = 0,1,2,\dots \\
    b_n &= \frac 1L\left(g,\sin\left(\frac{n\pi x}{L}\right)\right) \quad n = 1,2,3,\dots
\end{align*}
\boxed{\text{ex}} Consider $g(x) = \begin{cases}
    x & 0 < x < L \\
    x - 2L & L < x < 2L
\end{cases}$
The period extension of $g$ to $\mathbb{R}$ is the same as $f(x)$ above, $g$ has same Fourier series
as $f$.
\begin{align*}
    g(x) = \frac{2L}{\pi} \left(\sin\left(\frac{\pi x}{L}\right)-\frac 12\sin\left(\frac{2\pi x}{L}\right)+\dots\right)
\end{align*}
