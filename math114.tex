\end{align*}

\section{Markov Chains}
\subsection{Lecture 9}
\underline{Def:} A stochastic process is a family of RVs, $X_\epsilon \in S$, 
indexed by $t$, where $S \subseteq \mathbb{R}^d$ is a state space. Typically, 
$t \in [0,\infty)$, or $t \ in [a,b]$, or $t\in\{0,1,2,3,...\}$, etc.
\begin{itemize}
    \item A discrete time stochastic process, is a sequence of RVs, $X_n \in S$,
        $n = 0,1,2,...$ or $n = 1,2,3,...$
    \item If S is finite or countably infinite then the stochastic process is a 
        discrete state process.
\end{itemize}
First, we will examine discrete-time, discrete pace, stochastic process. \\
Two basic questions:
\begin{itemize}
    \item Find the joint distribution
    \begin{align*}
        &P(X_0 = i_0, X_1 = i_1, ..., X_n = i_n) \\
        &i_0, i_1, ..., i_n \in S
    \end{align*}
    \item Find the long-time behavior of the distribution of $X_n$:
    \begin{align*}
        P(x_n = i), i\in S, n>>1
    \end{align*}
\end{itemize}

\noindent \underline{Def:} Markov processes
\begin{itemize}
    \item A stochastic process $\{X_n\}^\infty_{n=0}$ is a Markov process if 
    \begin{align*}
        &P(X_{n + 1} = i_{n+1} | x_0 = i_0, ..., x_n = i_n) \\
        &= P(X_{n+1} i_{n+1} | X_n = i_n) \\
        & \forall n\ge 0, \forall i_0, ..., i_{n+1} \in S
    \end{align*}
    \item A Markov process is time homogeneous if: 
    \begin{align*}
        &P(x_{n+1} = j | x_n = i) \\ 
        &= P(X_1 = j | X_0 = i) \\
        &\forall n\ge 0, \forall i,j \in S
    \end{align*}
    Here we consider discrete time/state, time homogeneous Markov Chains (MC).
\end{itemize}

\noindent \underline{Def:} Transition probability
\begin{align*}
    P(i,j) &= P(X_1 = j | X_0 = i)\\
    &= P(i\rightarrow j ) \quad \forall i,j \in S
\end{align*}

\noindent \underline{Def:} Transition matrix
\begin{align*}
    &P=[p(i,j)] \text{ if } S = \{1,2,3,...\} \text{ or } \{0,1,2,...\} \\
    &\text{Here i,j are states and also are row/column labels}
\end{align*}

\noindent For any $n\ge 1, i_0, ..., i_n \in S$, we have:
\begin{align*}
    &P(X_0 = i_0, ..., X_n = i_n) \\
    &= P(X_0 = i_0) P(X_1 = i_1 | X_0 = i_0) P(X_2 = i_2 | X_1 = i_1, X_0 = i_0) \\
    &\ ...\ P(X_n = i_n | X_{n-1} = i_{n-1},..., X_0 = i_0)\qquad [\text{conditional probability}]
\end{align*}

\noindent \boxed{ex} \ Let $X_0, X_1,...,X_n \in S$\ iid. Then $\{X_n\}^\infty_{n=0}$ 
is a time homogeneous MC. \\
Let $S = \{1,2,...,N\}$. Then $p(i,j)=\mathbb{P}(X_1 = j | X_0 = i) = \mathbb{P}(X_1=j)=p_j$ \\
$\implies p = \begin{bmatrix}
    p_1 & \dots & p_n \\
    \vdots & \ddots & \vdots \\
    p_1 & \dots & p_n
\end{bmatrix}$, $\displaystyle\sum_{i=1}^{n} p_i = 1$

\subsection{Lecture 10}
Recap: \begin{itemize}
    \item Discrete time/state, time homogeneous
    \item MC: $\{X_n\}^\infty_{n=0} \in S$
    \item $\mathbb{P}(X_{n+1}=i_{n+1}|X_0=i_0,...,X_n=i_n)=\mathbb{P}(X_{n+1}=i_{n+1} | X_n=i_n)$ \\
    $= p (i_n, i_{n+1})$
    \item Transition probability $\mathbb{P}(X_1=j | X_0 = i)$ \\
    $=p(i,j) = p_{ij} \quad \forall i,j \in S$
    \item Transition matrix $\mathbb{P}[p(i,j)]$
\end{itemize}
\boxed{ex} Two-state MC, e.g. state of phone $X_n \in S = \{0,1\}$ \\
$\begin{cases}
    X_n = 0 & \text{phone free at time n} \\
    X_n = 1 & \text{phone busy at time n} 
\end{cases}$ \\ \\
$\mathbb{P}(\text{free} \rightarrow \text{busy in time step}) = p$ \\
$\mathbb{P}(\text{busy} \rightarrow \text{free in time step}) = q$ \\
$\mathbb{P}(\text{free} \rightarrow \text{free in time step}) = 1-p$ \\
$\mathbb{P}(\text{busy} \rightarrow \text{busy in time step}) = 1-q$ \\

\noindent Transition matrix:
\begin{align*}
    P = \begin{bmatrix}
        1-p & p \\
        q & 1-q
    \end{bmatrix}
\end{align*}
e.g. $p = \frac 12, q = \frac 13$
\begin{align*}
    P = \begin{bmatrix}
        \frac 12 & \frac 12 \\
        \frac 13 & \frac 23
    \end{bmatrix}
\end{align*}
Let $\mathbb{P}(X_0 = 0) = \frac 15, \mathbb{P}(X_0 = 1) = \frac 45$ \\
Then $\mathbb{P}(X_0 = 0, X_1 = 1, X_2 = 0)$ \\
$= \mathbb{P}(X_0 = 0) \cdot p(0,1) \cdot p(1,0)$ \\
$= \frac 15 \cdot \frac 12 \cdot \frac 13 = \frac 1{30}$ \\ 
And $\mathbb{P}(X_0 = 0, X_2 = 0) = \sum_{i=0}^{1}\mathbb{P}(X_0 = 0, X_1 = i, X_2 = 0)$ \\

\noindent \boxed{ex} $P_0 = I$, $p_0(i,j) = \delta_{ij} = \begin{cases}
    1 & i = j \\
    0 & i \ne j
\end{cases}$ \\ 
$P_1 =P$, $p_1(i,j) = p(i,j)$ \\

\noindent The distribution vector $\pi_n$ for $X_n$ is 
\begin{align*}
    \pi_n &= (\pi_n(0), \pi_n(1), ...) \\ 
    \text{where } \pi_n(i) &= \mathbb{P}(X_n = i), i \in S \\
    \text{Note that } \pi_n \text{ is a} & \text{ probability vector}
\end{align*}

\noindent \underline{Property:} Let $X_n \in S = \{1,2,3,...\}, n \ge 0$ be the time 
homogeneous Markov Chain with transition matrix $P$. Then 
\begin{itemize}
    \item $P$ is a stochastic matrix
    \item $P \cdot \mathbb{I} = \mathbb{I}$ where $\mathbb{I} = \begin{bmatrix}
    1 \\ 1 \\ \vdots \\ 1
    \end{bmatrix}$
    \item 1 is an eigenvalue of $P$ with eigenvector $\mathbb{I}$
    \item $P_{m+n} = P_m \cdot P_n$ (Chapman-Kolmogorov equation)
    \item $P_n = P^n$
    \item $\pi_n = \pi_0 P_n = \pi_0 P^n$
    \item $\pi_{n+1} = \pi_n P,\ \forall n \ge 0$
\end{itemize}

\noindent \underline{Proof:} direct verification \\
e.g. $\mathbb{P}(X_1 = j) = \sum_{i \in S} \mathbb{P}(X_0 = i) \mathbb{P}(X_1 = j | X_0 = i)$ \\
$\pi_1(j) = \sum \pi_0 (j)P_{ij}$

Long-run behavior, invariant distribution\\\\
\underline{Def:} Let $P = [p(i,j)]$ be the transition matrix for Markov Chain $\{X_n\}_{n=0}^\infty$ with $S = \{1,2,..., N\}$.
A probability vector $\pi \in \mathbb{R}^n$ is a stationary or invariant distribution if $\pi^p = \pi$

\subsection{Lecture 11 - Invariant Distribution}
\begin{itemize}
    \item Recap: Discrete time/state, time homogeneous Markov Chains. Transition matrix $P = [p(i,j)],\ p(i,j) = \mathbb{P}(X_1 = j | X_j = i),\ p_n(i,j) = \mathbb{P}(X_n=j | X_0 = i)$
    \item Markov Chains $\rightarrow$ MCMC 
\end{itemize}
\boxed{ex}\ 3 state MC
\begin{align*}
    &P = \begin{bmatrix}
        \frac 34 & \frac 14 & 0 \\
        \frac 18 & \frac 23 & \frac 5{24} \\
        0 & \frac 56 & \frac 16
    &\end{bmatrix} \\
    &S = \{1,2,3\} \\
    &\mathbb{P}(X_1=1|X_0=1) = \frac 34, \text{etc.} \\
    &\lim_{n\rightarrow \infty}P^n = \begin{bmatrix}
        \pi \\ \pi \\ \pi
    \end{bmatrix},\ \pi = (\frac 2{11}, \frac 4{11}, \frac 5{11}) \\
    & \text{Set } \pi_0 = (\mathbb{P}(X_0 =1), \mathbb{P}(X_0 =2), \mathbb{P}(X_0 =3)) \\
    & = \mathbb{P}(X_0 = i)
\end{align*}
Then 
\begin{align*}
    \pi_n &= \pi_0 p^n \\
    \lim_{n\rightarrow\infty} \pi_n &= \pi_0 \lim_{n\rightarrow \infty}p^n \\
    &= \pi_0 \begin{bmatrix}
        \pi \\ \pi \\ \pi
    \end{bmatrix} \\
    &= \Big(\pi(1) \cdot \sum_{i} \pi_0 (i),\ \pi(2) \cdot \sum_{i} \pi_0 (i), ... \Big) \\
    &= \pi
\end{align*}
\underline{Remarks}
\begin{itemize}
    \item $\lim_{n\rightarrow \infty} \mathbb{P} (X_n = j) - \pi(j) \forall j$
    \item \begin{align*}
        \pi_n &= \pi_{n-1}P \\
        \lim_{n\rightarrow\infty} \pi_n &= \lim_{n\rightarrow\infty}\pi_{n-1}P \\
        \pi &= \pi P
    \end{align*}
    \item $\pi$ is a left eigenvector of P with eigenvalue 1.
\end{itemize}
\underline{Def:} Let $P = [p(i,j)]$ be the transition matrix for MC $\{X_n\}^\infty_{n=0}$ with 
$S = \{1,...,N\}$. A probability vector $\pi \in \mathbb{R}^n$ is a stationary or 
invariant distribution for $X_n$ (or P) if: $\pi = \pi P$ \\ \\
\underline{Remarks} \\
Above example shows $\lim_{n\rightarrow\infty} \pi_n = $ invariant distribution 
(with some conditions). \\ 

\noindent Questions: invariant distribution.
\begin{itemize}
    \item When do they exist?
    \item Are they unique?
    \item Roughly: typically have unique invariant distribution unless reducible or periodic.
\end{itemize}
\noindent \underline{Def:} A finite state time-homogeneous MC $X_n (n \ge 0)$ with transition matrix $P$ is
regular if there exists an integer $k\ge 1$, such that $P^k > 0$, (i.e. all entries of $P^k$ are 
strictly $>0$).  \\ \\
\underline{Thm} \\
Let $X_n (n\ge 0)$ be a regular N-state Markov Chain with transition matrix P. Then:
\begin{enumerate}
    \item $\{X_n\}_{n=0}^\infty$ has a unique stationary distribution $\pi,\ \pi > 0$
    \item $\lim_{n \rightarrow\infty} P^n = \begin{bmatrix}
        \pi \\ \vdots \\ \pi
    \end{bmatrix}$, i.e. $\lim_{n\rightarrow\infty}p_n(i,j) = \pi(j)$
    \item For any initial distribution $\pi_0,\ \lim_{n\rightarrow\infty} \pi_0 P^n = \pi$
    i.e. $\lim_{n\rightarrow\infty} \mathbb{P}(X_n=j)= \pi(j)\ \forall j$
\end{enumerate}
\underline{Remarks}
\begin{itemize}
    \item 2. $\implies \displaystyle\lim_{n\rightarrow\infty} \mathbb{P}(X_n = j | X_0 = i)=\pi(j)$ independent of $i$
    \item From 2.:
    \begin{align*}
        \lim_{n\rightarrow\infty} \frac 1n \sum_{k=0}^{n-1}p_k(i,j) &= \lim_{n\rightarrow\infty} p_n(i,j) = \pi(j)\\
        \text{And } \frac 1n \sum_{k=0}^{n-1} p_k(i,j) &= \frac 1n \sum_{k=0}^{n-1}\mathbb{P}(X_k = j | X_0 = i) \\
        &= \frac 1n \sum_{k=0}^{n-1}\mathbb{E}[\mathbb{I}[X_k=j]|X_0 =i] \\
        &= \mathbb{E} \Big[\frac 1n \sum_{k=0}^{n-1} \mathbb{I}(X_k = j) | X_0 = i\Big] \\
        &\rightarrow \pi(j) \text{ as } n \rightarrow \infty 
    \end{align*}
    \begin{center}
        Which means the mean fraction of time spent in $J \rightarrow \pi (j)$
    \end{center}
\end{itemize}  
How do we find $\pi$?
\begin{itemize}
    \item Solve $\pi P = \pi$
    \item Diagonalize $P = U\Lambda U^{-1}$
    \begin{align*}
        \Rightarrow \lim_{n\rightarrow\infty} P^n &= \lim_{n\rightarrow\infty}(U\Lambda U^{-1})^n \\
        &= U(\lim_{n\rightarrow\infty}\Lambda ^n)U^{-1} \\
        &= \begin{bmatrix}
            \pi \\ \vdots \\ \pi
        \end{bmatrix}
    \end{align*}
\end{itemize}

\subsection{Lecture 12 - MCMC}
Recap:
\begin{itemize}
    \item Discrete time/space MC $\{X_n\}_{n=0}^\infty, X_n \in S$
    \item $P = [p(i,j)], p(i,j) = \mathbb{P}(X_1 = j | X_0 = i)$
    \item $\pi_0 = \mathbb{P}(X_0 = i),\ \pi_n = \mathbb{P}(X_n=i)\quad (\text{pmf }X_n)$
    \item Stationary distribution $\pi$ s.t. $\pi P = \pi$ 
    \item Regular: $\exists k\ge 1 $ s.t. $P^k > 0 \Rightarrow$ unique stationary distribution $\pi >0$ \\
        And $\lim_{n\rightarrow \infty} P^n = \begin{bmatrix}
            \pi \\ \vdots \\ \pi
        \end{bmatrix}$, $\lim_{n\rightarrow\infty} \pi_n = \pi$
    \item Detailed balance: probability vector $\lambda$ s.t. $\lambda_i p(i,j)=\lambda_j p(j,i)\ \forall i,j \in S$\\
    $\Rightarrow \lambda P = \lambda \Rightarrow \lambda $ stationary with respect to $P$.
\end{itemize}

\noindent\underline{Remarks}
\begin{enumerate}
    \item Can view MC as evolution of probability vectors (pmfs): $\pi_0, \pi_1, ...$ with $\pi_{n+1} = \pi_n P$.
    This is hard to compute for $|S|$ large.
    \item MC is a random sequence of states $X_0,X_1,... \in S$ which is often easier to compute.
\end{enumerate}

\noindent Recall: Discrete Inversion (Lecture 6) \\
e.g.: Generate discrete RV X  
\begin{align*}
    &P(X=i) = p_i,\ i=1,2,3 \\
    &p_1 + p_2 +p_3 = 1
\end{align*}
\begin{enumerate}
    \item $U \sim U[0,1]$
    \item $k = min\{m | \sum_{i=1}^{m} p_i \ge U\}$
    \item Set $X \leftarrow k$
\end{enumerate}

\noindent \underline{Algorithm:} Generate / simulate MC \\
Input: transition matrix $P$, initial distribution $\pi_0$ \\
Output: MC $X_0, X_1,... \in S, X_n \sim \pi_k = \pi_0P^k$
\begin{enumerate}
    \item Generate $X_0 \sim \pi_0$
    \item For $k=1,...,n$, generate $X_{n+1}$ from $p(X_k)$. i.e. pmf over $S$, given by the $X_k$'th row of P.
    \item Repeat $n$ times to get samples from $\pi_k$
\end{enumerate}

\noindent\underline{MCMC:} (R$\&$K ch.6)
\begin{itemize}
    \item Key idea: generate distribution $\pi$ by simulating MC with stationary distribution $\pi$
    \item Often $\pi$ is known up to a normalizing constant.
    \item $\lim_{n\rightarrow\infty} \frac 1n \sum_{k=1}^{n} \mathbb{I}\{X_k=i\} = \pi(i)\ \forall i \in S$
    \item $\lim_{n\rightarrow\infty} \frac 1n \sum_{k=1}^{n} g(X_k) = \mathbb{E}_\pi (g(x))$
\end{itemize}
We assume that 
\begin{itemize}
    \item RV $X\in S$ is finite, e.g. $S = \{1,2,...,N\}$
    \item Target pmf of X is $\pi = (\pi(1), ...,\pi_N), \pi(i) = \mathbb{P}(X=i)\ \forall i\in S$
    \item $\pi$ has special form $\pi (i) = \frac 1z b(i) > 0 \ \forall i \in S$. \\
    Where $b(i)>0$ are known and $z > 0$ is unknown norm.
    \begin{align*}
        \sum_{i\in S} \pi(i) = 1 \iff \sum_{i \in S} b(i) = z
    \end{align*}
\end{itemize}

\noindent\underline{Remarks}
\begin{itemize}
    \item $z$ is hard to compute, (sum of $N  = |S|$ terms)
    \item $z$ is called a partition function 
\end{itemize}

\noindent\underline{Algorithm:} Metropolis 
\begin{itemize}
    \item First construct proposal transition matrix $Q = [q(i,j)]$, where $Q$ is symmetric.
    \item Choose initial state $X_0 \in S$ (e.g. randomly).
\end{itemize}
Given $X_k = i \in S$
\begin{enumerate}
    \item Sample $Y\in S$ from $q(i,\cdot)$, i.e. $\mathbb{P}(Y=y | X_k =i) = q(i,y)\ \forall y \in S$ \\
    Assume $Y = j$ is observed.
    \item Let $\alpha = \alpha_{ij} = \min\left(1, \frac{\pi(j)}{\pi(i)}\right) \in [0,1]$ and generate $U \sim U[0,1]$. \\
    If $U\le \alpha$, accept $Y$, $X_{k+1} \leftarrow Y$ \\
    If $U > \alpha$, reject $Y$, $X_{k+1} \leftarrow X_k$ \\
    (Accept $Y$ with probability $\alpha$) \\
    Output: $X_0, X_1, X_2, ...$
\end{enumerate}

\noindent\underline{Remarks}
\begin{itemize}
    \item $\{X_n\}_{n=0}^\infty$ is MC with $\lim_{n\rightarrow\infty}X_n \sim \pi$
    \item Can choose $Q = [q(i,j)]$ to be simple, e.g. uniform jumps, $q(i,j) = \frac 1N$
    \item Only need the ratios $\frac{\pi(j)}{\pi(i)}$, so we don't need $z$.
    \item Call $Y$ a proposed state.
    \item Call $\alpha$ the acceptance probability.
    \item In practice, $\alpha = a_{ij}$ is easy to compute.
    \item Can be extended to non-symmetric $Q$
\end{itemize}
\boxed{ex} 1D Ising model
\begin{align*}
    S &= \{\sigma = (\sigma_1, ... , \sigma_m) | \sigma_i = -1 \text{ or } +1,\ 1\le i \le m\} \\
    &= \{-1,+1\}^m \\
    |S| &= 2^m \text{ states}
\end{align*}

\subsection{Lecture 13 - Metropolis}
\underline{Metropolis Algorithm} 
\begin{align*}
    S = \{1,2,...,N\}
\end{align*}
Goal: Sample random variable $X$ from some target distribution $\Pi(i) = \mathbb{P}(X=i)\implies$
$\Pi_i (i) = \frac 1z b(i)$ \\ \\
We have $Q = [q(i,j)]$ which is our transition matrix (symmetric).
\begin{enumerate}
    \item Choose $X_0 \in S$. Given $X_n \in S$, sample $Y \in S$ from $q(X_n, i)$, i.e. $\mathbb{P}(Y=y | X_n = i)$
    \item Set $\alpha = \min(1, \frac{\pi(y)}{\pi(X_n)}) \in [0,1]$. \\
    Generate $U \sim U[0,1]$ \\
    If $U\le \alpha$, accept $Y:X_{n+1} \leftarrow Y$ \\
    Otherwise, $U > \alpha$, reject $Y$, $X_{n+1}\leftarrow X_n$
\end{enumerate}

\noindent \underline{1D Ising Model} \\
$S = \{\sigma = \{\sigma_1, \sigma_2, ... , \sigma_n\} : \sigma_j = \pm 1\}$ \\
$\Rightarrow 2^m$ states.
\begin{align*}
    &\Pi (\sigma) = \frac 1z e^{-\beta \mathcal{H}(\sigma)} \\
    &\mathcal{H} = - J \sum_{j=1}^{m-1} \sigma_j \sigma_{j+1} - h\sum_{j=1}^{m}\sigma_j \\
    & J, h > 0 
\end{align*}

Large $\Pi (\sigma) \longleftrightarrow $ Low energy states.

\noindent \underline{Calculate physical quantities.} \\
\begin{align*}
    m(\sigma) &= <\sigma> = \frac 1m (N_{+} - N_{-}) \\ 
    &= \frac 1m \sum (N_+ - N_-) \Pi_p (\sigma)
\end{align*}
\begin{align*}
    \frac{\Pi_\beta (y)}{\Pi_\beta (X_n)} = e^{-\beta ( \Delta \mathcal{H})}
\end{align*}
\begin{enumerate}[a.]
    \item $1 < i < m$ \\
        $\Delta \mathcal{H} = 2J \sigma_i(\sigma_{i+1} + \sigma_{i-1}) + 2h\sigma_i$
\end{enumerate}

\subsection{Lecture 14}
\underline{Recap: Metropolis Chain} \\
We have 
\begin{itemize}
    \item State space $S = \{1,2,...,N\}$
    \item Proposed transition matrix $Q = [q(i,j)]$ which is symmetric
    \item Target distribution $\pi > 0 $
\end{itemize}
Metropolis summary
\begin{itemize}
    \item Given $X_k \in S$, choose $y$ sampled from $q(X_k, \cdot)$, i.e. $X_k$'th 
    row of $Q$
    \item Accept $Y$ with probability $\min(1, \frac{\pi(y)}{\pi(X_k)})$
    \item Output: MC $X_0, X_1, ..., \in S$ with $X_n \sim \pi$ as $n \rightarrow \infty$
\end{itemize}

\noindent \underline{Convergence}
\begin{itemize}
    \item How big should $n$ be to get $X_n \sim \pi$ 'roughly'?
    \item Hard to compute!
    \item Heuristic: Consider $R(n) = \text{Cov}(X_n, X_0)$
\end{itemize}

\noindent \underline{Thm}
\begin{enumerate}
    \item The transition matrix $P = [p(i,j)]$ of the Metropolis chain is 
    \begin{align*}
        p(i,j) = \begin{cases}
            q(i,j) \min(1, \frac{\pi(j)}{\pi(i)}) & i \ne j \\
            1 - \sum_{k\in S, k\ne i} q(i,k) \min(\frac{\pi(k)}{\pi(i)}) & i = j
        \end{cases}
        \forall i,j \in S
    \end{align*}
    \item $\pi$ and $P$ satisfy detailed balance (DB)
    \begin{align*}
        &\pi(i)p(i,j) = \pi(j)p(j,i)\ \forall i,j \in S \\
        &\text{Hence } \pi \text{ is stationary}
    \end{align*}
    \item If $q(i,j) > 0 \forall i,j \in S$ and $p(i,i) > 0\ \forall i \in S$
    Then $\pi$ is a unique stationary distribution and $\lim_{n\rightarrow\infty}
    \pi_0 P^n = \pi$
\end{enumerate}
\underline{Proof}
\begin{enumerate}
    \item For $i \ne j$
    \begin{align*}
        p(i,j) = \mathbb{P}(\text{propose } j|i)\cdot \mathbb{P}(\text{accept } j) \\
        q(i,j) \min(1, \frac{\pi(j)}{\pi(i)})
    \end{align*}
    Then 
    \begin{align*}
        &\sum_{j \in S} p(i,j) = 1\ \forall i \in S \\
        &\Rightarrow p (i,j) = 1 - \sum_{k \in S, k\ne i} q(i,k) \min(1, \frac{\pi(k)}{\pi(i)})
    \end{align*}
    \item Want to show: $\pi(i)p(i,j) = \pi(j)p(j,i)\ \forall i,j \in S$. \\
    This is always true for $i = j$ \\
    For $i \ne j$:
    \begin{align*}
        \pi(i)p(i,j) &= \pi(i) q(i,j) \min(1, \frac{\pi(j)}{\pi(i)}) \\
        &= q(i,j) \min(\pi(i), \pi(j)) \\
        &= q(j,i) \min(\pi(j), \pi(i)) \\
        &= \pi(j) p(j,i) \Rightarrow DB \checkmark \\
        & \pi \text{ is stationary (Lecture 11)}
    \end{align*}
    \item Conditions $\Rightarrow p(i,j) > 0\ \forall i,j \in S$ 
    $\Rightarrow P $ is regular \\
    $\Rightarrow \pi$ is unique and $\lim_{n\rightarrow\infty} \pi_0P^n = \pi$
\end{enumerate}

\underline{Remarks}
\begin{itemize}
    \item For Metropolis Algorithm applied to Ising model, we had:

    \end{align*}

\noindent\underline{Glauber dynamics} \\
(Random-scan Gibbs sampler for Ising model)\\
Goal: Construct MC, $X_0, X_1,... \in S$ s.t. $X_n \sim \pi$ as $n\rightarrow \infty$ \\
Assume $X_k = \sigma \in S$
\begin{enumerate}
    \item Choose $J$ uniformly at random from 1,2,..., m 
    \item Set $\alpha_+ = \frac{\pi(\sigma^{J, +1})}{\pi(\sigma^{J, -1}+ \sigma^{J, +1})}$
    and $\alpha_- = \frac{\pi(\sigma^{J, -1})}{\pi(\sigma^{J, -1}+ \sigma^{J, +1})}$
\end{enumerate}
Note that $\alpha_+ + \alpha_-  = 1$ \\
Set $X_{k+1} \leftarrow \sigma^{J, +1}$, with probability $\alpha_+$ \\
$X_{k+1} \leftarrow \sigma^{J, -1}$, with probability $\alpha_-$ \\

\noindent\underline{Remarks}
\begin{itemize}
    \item Each 'spin' is in $\{-1, +1\}$
    \item $\{\alpha_+, \alpha_-\}$ is pmf over $\{-1,+1\}$
    \item Set $J$'th spin conditional on the value of every other spin.
\end{itemize}

\noindent\boxed{ex} $m = 2$ spins
\begin{align*}
    \text{States}: &\ S = \{(-1, -1),(-1, +1),(+1, -1),(+1, +1), \} \\
    \text{Set}: & \ \pi(-1,-1) = \frac 1z, \pi(-1,+1) = \frac 2z \\
    & \pi(+1, -1) = \frac 3z, \pi(+1, +1) = \frac 5z \\
    (\text{Here } z = 11)
\end{align*}
Suppose $X_k = (-1,-1), J=2$ (Glauber dynamics) \\
Then $\sigma^{J, +1} = (-1,+1)$, $\sigma^{J, -1}=(-1,-1)$ \\
$\alpha_+ = \frac{\frac 3z}{ \frac 1z + \frac 3z} = \frac 34$, $\alpha_- = \frac{1}{1+3} = \frac 14$ \\
Then $\mathbb{P}(X_{k+1}=(-1,+1)\ |\ X_k = (-1,-1))$
$= \frac 34 \cdot \frac 12 = \alpha_+ \cdot \mathbb{P}(J=2)$

\textbf{MC Integration} \\
\underline{Goal:} Calculate 
\begin{align*}
    I = \int_D g(x)f(x)dx \quad D \subseteq \mathbb{R}^d
\end{align*}
\begin{align*}
    \text{where } &D \text{ is bounded} \\
    &f,g: D \rightarrow \mathbb{R} \\
    &f \text{ is a pdf on } D
\end{align*}
For $X \sim f$: 
\begin{align*}
    I &= \mathbb{E}_f(g(x)) \\
    \text{Var}_f(g(x)) &= \mathbb{E}_f(g(x)^2) - [\mathbb{E}_f(g(x))]^2 \\
    &= \int_D g(x)^2 f(x) dx - I^2
\end{align*}
\underline{Crude MC estimator:} \\
\begin{align*}
    I_N = \frac 1N \sum_{i=1}^{N}g(x_i) && X_1, ...,X_N \sim f, iid
\end{align*}
Then (check):
\begin{itemize}
    \item $\mathbb{E}(I_N) = I$ (unbiased)
    \item Var$I_N = \frac 1N $Var$_f (g(x_i))$
    \item $I_N \rightarrow I$ with probability 1
\end{itemize}
\underline{Importance sampling:} \\
The importance function $\phi$ is a pdf on $D$. \\
Then 
\begin{align*}
    I &= \int_D \frac{g(x)f(x)}{\phi(x)}\phi(x)dx \\
    &= \mathbb{E}_\phi \left[\frac{g(y)f(y)}{\phi(y)}\right], \quad y \sim \phi
\end{align*}
and we have our estimator 
\begin{align*}
    J_N = \frac 1N \sum_{i=1}^{N} \frac{g(y_i)f(y_i)}{\phi(y_i)} && y_1, ..., y_N \sim \phi iid
\end{align*}
Then (check):
\begin{itemize}
    \item $\mathbb{E}(J_N) = I$ (unbiased)
    \item Var$J_N - \frac 1N \left[\int_D \frac{g(x)^2f(x)^2}{\phi(x)}dx - I^2\right]$
    \item Choose $\phi$ to be easy to compute and to give small Var$J_N$
\end{itemize}
\textbf{Variance Reduction} \\
Let $U, V \sim U[0,1],\ \alpha, \beta: \mathbb{R} \rightarrow \mathbb{R}$ \\
Set $W = \alpha(U) + \beta(V),\ W$ is an estimator for $\mu = \mathbb{E}(\alpha(U)+\beta(V)) = \mathbb{E} W$\\
\underline{Goal:} Find relationship between $U,V$ to reduce Var $W$  \\
Note: Var $W$ = Var$(\alpha(U))$ + Var$\beta(V)$ + 2Cov$(\alpha(U),\beta(V))$ \\
Note: Cov$(X,Y) = \mathbb{E}(XY) - \mathbb{E}(X)\mathbb{E}(Y)$ which $= 0 $ if $X,Y$ are independent.
\begin{enumerate}
    \item $\alpha, \beta$ increasing \\
    $\rightarrow$ Use antithetic RVs: $V = 1-U$ \\
    Cov$(\alpha(U),\beta(1-U)) \le 0, \quad \alpha, \beta \uparrow$
    \item $\alpha \uparrow, \beta \downarrow$ \\
    $\rightarrow$ Use common RVs: $V = U$ \\
    Cov$(\alpha(U), \beta(U)) \le 0, \quad \alpha \uparrow, \beta \downarrow$ \\
    See practice midterm Q4, HW2 Q4.
\end{enumerate}
\textbf{Markov Chain}
\begin{itemize}
    \item Discrete time/state, time homogeneous MC \\
    $X_0, X_1, ... \in S = \{1,2,3,...,N\}$
    \item Transition matrix \\
    $P = [p(i,j)],\ p(i,j) = \mathbb{P}(X_1 = j | X_0 = i)\ \forall i,j \in S$
    \item Distribution vectors:
    $\pi_n = [\mathbb{P}(X_n = i)]\ \forall i \in S, $ pmf $X_n$
    \item $\pi_{n+1} = \pi_n P$ (HW3)
    \item Stationary/invariant distribution \\
    $\pi $ s.t. $\pi P = \pi$
    \item Regular MC or transition matrix means that $\exists\ k \ge 1$ s.t. $P^k > 0$ (Lecture 10, 11) \\
    Regular $\Rightarrow$ unique stationary distribution $\pi > 0$ and $\pi_n \rightarrow \pi$ as $n \rightarrow\infty$
\end{itemize}

Note that this matrix is regular, with a unique stationary distribution $\Rightarrow \pi_0 P^n \rightarrow\pi$ 
for any $\pi_0$. \\
Solve $\pi P = \pi$, let $\pi = (a,b)$ (HW3, practice midterm)

\end{align*}

\textbf{Acceptance-Rejection} \\
We have 
\begin{align*}
    f &: \text{target pdf of } Z \in \mathbb{R}^d \\
    \alpha g &: \text{proposed pdf},\ \alpha > 0 \text{ s.t. } f(x) \le \alpha g(x)\ \forall x
\end{align*}
\underline{Algorithm:} sampling $Z \sim f$
\begin{enumerate}
    \item Generate $X \sim g$ 
    \item Generate $U \sim U[0,1]$
    \item Set $Y = \alpha g(x) \cdot U$
    \item If $U \le \frac{f(x)}{\alpha g(x)}$, then accept $X$
    \item Return $Z \leftarrow X$
\end{enumerate}
\underline{Efficiency:} $\frac 1\alpha = \frac{\text{vol B}}{\text{Vol A}} = \text{acceptance rate}$

\underline{Recap} - MCMC
\begin{itemize}
    \item Goal: sample from target distribution $f$ on a state space $S$, by constructing
    MC $X_0, X_1,... \in S$ with $X_n \sim f$ as $n \rightarrow \infty$
    \item Output: $X_0, X_1, X_2, ... \in S$
    \item For large enough $n$, $X_n, X_{n+1}, X_{n+2}, ...$ is approximately a sample 
    from $f$, but $X_n, X_{n+1}, X_{n+2}, ...$ are dependent, not iid.
    \item For independent sample, we would need to run MCMC many times.
    \item Estimator: Suppose $X\sim f$ and $Y = g(X)$, we can use MCMC to estimate $\mathbb{E}Y$
    \begin{align*}
        \text{Let }\ Y_n = \frac 1n \sum_{i=1}^{n}g(X_i)
    \end{align*}
    Typically, $Y_n \rightarrow \mathbb{E}Y$ (HW4)
    \item $Y_n$ can be used to assess convergence (often)
    \item Can also assess convergence with sample autocovariance. (HW4)
\end{itemize}
\textbf{Gibbs Sampler}
\begin{itemize}
    \item Goal: sample a target pdf
    \begin{align*}
        f(x),\ x \in \mathbb{R}^n
    \end{align*}
    \item Conditional distribution:
    \begin{align*}
        f_j (y | x_1, ..., x_n) = \frac{f(x_1,...,x_{j-1},y,x_{j},...,x_{m+1})}{\int dy f(x_1,...,x_{j-1},y,x_{j},...,x_{m+1})}
    \end{align*}
    for $1 \le j \le m$
\end{itemize}
So $f_j (\ \cdot\ |\ x_1,..., x_{m-1})$ is a pdf on $\mathbb{R}$ \\ \\
\underline{Algorithm:} Random-scan Gibbs Sampler \\
Given $X_k = (X_{k,1},..., X_{k,m}) \in \mathbb{R}^m$
\begin{enumerate}
    \item Choose $J$ in $\{1,...,m\}$ randomly (e.g. from uniform distribution on $\{1,...,m\}$)
    \item Sample $Y$ from distribution: 
    \begin{align*}
        f_J (\ \cdot\ |\ \underbrace{X_{k,1},..., X_{k,J-1}, X_{k,J+1}, ..., X_{k,m}}_{\in \mathbb{R}^{m-1}})
    \end{align*}
    Set: 
    \begin{align*}
        X_{k+1} \leftarrow (\underbrace{X_{k,1},..., X_{k,J-1}, Y, X_{k,J+1},..., X_{k,m}}_{\in \mathbb{R}^m})
    \end{align*}
\end{enumerate}
\boxed{ex} \\
Target: $f(x,y) = c\exp(-(x^2+1)(y^2+1)),\ (x,y) \in \mathbb{R}^2$ \\
$c$ is normalization so
\begin{align*}
    \int_{\mathbb{R}^2} c \exp (-(x^2+1)(y^2+1)) dxdy = 1
\end{align*}
What are the conditional distributions?
\begin{align*}
    f_1(x | y) &= \frac{1}{\int_\mathbb{R}f(x,y)dx} \cdot c \exp(-(x^2+1)(y^2+1)) \\
    &= c_1(y)\cdot \exp (-(y^2+1)x^2)
\end{align*}
where $\int_\mathbb{R}f(x,y)dx$ is a function of $y$ and $c_1(y)$ is such that 
\begin{align*}
    \int_{-\infty}^{+\infty} c_1(y) \exp(-(y^2+1)x^2)dx=1
\end{align*}
i.e.: $c_1(y)$ is the normalization of the pdf 
\begin{align*}
    f_1(x|y) \propto \exp(-(y^2+1)x^2)
\end{align*}
Recall that $W \sim N(0, \sigma^2)$ has pdf $f_w(w) \propto \exp(\frac{-1}{2\sigma^2}w^2)$
\begin{align*}
    \Rightarrow f_1 (x|y) \text{ is the pdf of } N\left(0,\frac{1}{2(y^2+1)}\right)
\end{align*}
Similarly, $f_2 (y|x)$ is pdf of $N(0, \frac{1}{2(x^2+1)})$\\
\underline{Algorithm} (Random-scan Gibbs) \\
Given $(x_k, y_k) \in \mathbb{R}^2$
\begin{enumerate}
    \item Choose $J$ uniformly at random from $\{1,2\}$
    \item Generate $Z \sim N(0,1)$ \\
    If $J = 1$  \\
    Set $W = \frac{1}{\sqrt{2(y_k^2+1)}}z \quad (\text{i.e. } W \sim  f_1(\ \cdot\ |\ y_k))$ \\
    Set $(x_{k+1}, y_{k+1}) \leftarrow (w, y_k)$ \\
    If $J = 2$
    Set $W = \frac{1}{\sqrt{2(x_k^2+1)}}z \quad (\text{i.e. } W \sim  f_2(\ \cdot\ |\ x_k))$ \\
    Set $(x_{k+1}, y_{k+1}) \leftarrow (x_k, w)$
    \item Output: $(x_1, y_1), (x_2, y_2), ...$
\end{enumerate}

\underline{Preliminaries}: Non-stochastic processes \\
ODE example:
\begin{align*}
    \text{Let } x(t) \in \mathbb{R},\ t \in [0,\infty) \\
    \text{Consider } \frac{dx}{dt} = x,\ x(0) = 1 \\
    \Rightarrow \text{Solution: } x(t) = Ae^t \\
    \text{Initial condition } \Rightarrow x(t) = e^t
\end{align*}
\underline{Discretization} \\
We can simulate/solve the ODE by discretizing it. \\
\begin{align*}
    &\text{Let } \delta t > 0 \text{ be small and } \\
    &\text{Let } t_n = n\cdot \delta t,\ n = 0,1,2,...
\end{align*}
How are $x(t_n), x(t_{n+1})$ related?
\begin{align*}
    &\text{We have } x(t_n) = \frac{dx}{dt} \Big|_{t=t_n} \approx \frac{x(t_{n+1})-x(t_n)}{\delta t} \\
    &\Rightarrow x(t_{n+1}) \approx x(t_n) + x(t_n) \cdot \delta t
\end{align*}
Consider the difference equation:
\begin{align*}
    &x_{n+1} = x_n + x_n \cdot \delta t,\ x_0 = 1,\ n = 0,1,2,... \\
    &x_1 = 1 + \delta t,\ x_2 = (1+\delta t)^2,\ ...\ ,\ x_n = (1+\delta t)^n = (1+\frac{t_n}{n})^n 
\end{align*}
\begin{itemize}
    \item Does the discrete solution $x_n$ approach the continuous solution $x(t)$?
\end{itemize}
Let $t' = t_n = n\delta t$ and consider $\lim_{n\rightarrow \infty} \delta t \rightarrow 0 $ with $t'$ fixed. \\
Then $x_n = (1 + \frac{t'}{n})^n \rightarrow e^{t'} = x(t')$ as $n\rightarrow \infty,\ \delta t \rightarrow 0,\ t' = n\delta t$ fixed. \\ \\
\underline{Remark}:
\begin{itemize}
    \item We can think of an ODE $\frac{dx}{dt} = x$ as a difference equation 
    $x_{n+1} = x_n + x_n \delta t$ in the limit $\delta \rightarrow 0$
\end{itemize}
\underline{Continuous limit of difference equation} \\
Consider difference equation $x_n \in \mathbb{R}$ 
\begin{align*}
    x_{n+1} = x_n + \delta t \cdot u,\ x_0 = 0,\ n = 1,2,...
\end{align*}
Is there a continuum limit as $\delta t \rightarrow 0$? $\Leftrightarrow$ Is there an $x(t)$ such that
\begin{align*}
    x_n \rightarrow x(t) \text{ as } n &\rightarrow \infty \\
    \delta t &\rightarrow 0 \\
    \text{with } n\delta t &= t \text{ fixed}
\end{align*}
Check:
\begin{align*}
    x_1 = \delta t u,\ ...\ ,\ x_n &= n \delta t u \\  
    &= tu \rightarrow t\cdot u \\
    \Rightarrow \text{Continuum limit } x(t) &= tu,\ \frac{dx}{dt} = u
\end{align*}
\underline{Remark}:
Suppose we took
\begin{align*}
    x_{n+1} = x_n + \delta t^\alpha u 
\end{align*}
Then $x_n = n \cdot \delta t^\alpha \cdot u = t \cdot \delta t^{\alpha -1}\cdot u$ 
\begin{align*}
    x_n \rightarrow \begin{cases}
        0 & \text{if } \alpha > 1 \\
        u\cdot t & \text{if } \alpha = 1 \\
        \infty & \text{if } \alpha < 1
    \end{cases} \text{ as } 
    \begin{cases}
        \delta  t \rightarrow 0 \\
        n \rightarrow \infty \\
        n \delta t = t \text{ fixed}
    \end{cases}
\end{align*}
Sensible continuum limit only for $\alpha = 1$ \\ \\
\underline{Stochastic processes} \\
\begin{align*}
    \begin{matrix}
        \text{Discrete time} \\ \text{Continuous space} 
    \end{matrix} \rightarrow\ 
    \begin{matrix}
        \text{Continuous time} \\ \text{Continuous space}
    \end{matrix}
\end{align*}
\begin{itemize}
    \item Random walk on $\mathbb{R}$ \\
    Let $Z_0, Z_1, ..., Z_n \sim N(0,1)\ iid$ and consider 
    $X_{n+1} = X_n+ \delta t^\alpha Z_n,\ n = 0,1,2,...$ with $X_0 = 0$
    \item $X_1 = \delta t^\alpha \cdot Z_0,\ X_2 = \delta t^\alpha (Z_0 + Z_1)$
    \begin{align*}
        X_n = \sum_{i=0}^{n-1} \delta t^\alpha Z_i
    \end{align*}
    \item Sum of normals is normals \\
    $\Rightarrow \mathbb{E}X_n = 0,\ \text{Var}X_n = n \cdot \delta t^{2\alpha}$ \\
    $\Rightarrow X_n \sim N(0, n \delta t^{2\alpha})$
    \item Continuum limit: is there $X(t)$ such that $X_n \rightarrow X(t)$ as 
    $n \rightarrow \infty$, $\delta t \rightarrow 0$, with $n\delta t = t$ fixed. \\
    We have: 
    \begin{align*}
        \text{Var}X_n = n\delta t^{2\alpha} = t\cdot \delta t^{2\alpha -1} \rightarrow 
        \begin{cases}
            0 & \alpha > \frac12 \\
            \infty & \alpha < \frac 12 \\
            t & \alpha = \frac 12
        \end{cases} \\
        \text{as } 
    n \rightarrow \infty, \delta t \rightarrow 0, \text{ with } n\delta t = t \text{ fixed}.
    \end{align*}
    Sensible continuum limit only for $\alpha = \frac 12$
\end{itemize}
\subsection{Lecture 19 - Brownian Motion}
\textbf{Goals}:
\begin{itemize}
    \item Define Brownian motion
    \item SDEs
    \item Simulation
\end{itemize}
\underline{Recap}:
\begin{itemize}
    \item Discrete-time, random walk on $\mathbb{R}$ \\
    Let $Z_0, Z_1,..., \sim N(0,1)\ iid$ \\
    and consider 
    \begin{align*}
        &X_{n+1} = X_n + \delta t^\alpha Z_n,\ n = 0,1,2,\dots \\
        &X_0 = 0,\ \delta t > 0 \\
        &\Rightarrow X_n = \delta t \sum_{i=0}^{n-1} Z_i \sim N(0,n\cdot \delta t^{2\alpha})
    \end{align*}
    Take limit $n \rightarrow \infty,\ \delta t \rightarrow \infty,\ n\cdot \delta t = t\ $fixed. \\
    \begin{align*}
        \text{Var}X_n &= n \cdot \delta t^{2\alpha} \\
        &= t \cdot \delta t ^{2\alpha -1} \begin{cases}
            0 & \alpha > \frac 12 \\
            \infty & \alpha < \frac 12 \\
            t & \alpha = \frac 12
        \end{cases}
    \end{align*}
    \item For $\alpha = \frac 12$, there is finite, non-zero $X(t)$ s.t. $X_n \rightarrow X(t)$
    as $n \rightarrow \infty,\ \delta t \rightarrow \infty,\ n\cdot \delta t = t\ $fixed.
\end{itemize}
\underline{Remarks}:
\begin{itemize}
    \item Above discussion is non-rigorous
    \item Continuum limit $X(t)$ is called Brownian motion (BM) / Wiener process.
    \item Discretized BM
    \begin{align*}
        &X_{n+1} = X_n + \sqrt{\delta t} \cdot Z_n,\ X_0 = 0 \\ 
        & \text{for } Z_0, Z_1, \dots, \sim N(0,1)\ iid,\ \delta t > 0
    \end{align*}
    \item SDE notation for $X(t) \in \mathbb{R}$
    \begin{align*}
        dx &= dB \\
        \text{Infinitesimal increment of X} &= \text{Infinitesimal increment of BM} \\
        &(\text{Here } X(t) \text{ is a BM})
    \end{align*}
    \item We can think of SDEs in terms of the discrete equation, with $\delta t \rightarrow 0$ (informal)
\end{itemize}
\textbf{Brownian Motion (BM)}. \\
\underline{Def}: A stochastic process $B(t) \in \mathbb{R},\ t \ge 0$ is a BM if:
\begin{enumerate}
    \item $B(0) = 0$ with probability 1.
    \item $B(t) - B(s) \sim N(0,t-s)\ \forall\ t \ge s \ge 0$ \\
    $[B(t) \cdot B(s) = \sqrt{t-s} \cdot Z$ for $Z\sim N(0,1)]$ \\
    $[i.e.\ B(t) \sim N(0,t)]$
    \item For any $0 \le t_0 \le t_1 \le ... \le t_n$, \\ the RVs $B(t_1) - B(t_0),...\ , B(t_n) - B(t_{n-1})$
    are independent (independent increments).
    \item $B(t)$ is a continuous function of $t\ge 0$
\end{enumerate}
\underline{Remarks}: 
\begin{itemize}
    \item From 2. $B(t+\delta t) - B(t) \sim N(0,\delta t),\ \delta t > 0$ \\
    $\Rightarrow$ Have $B(t+\delta t) - B(1) = \sqrt{\delta t} \cdot Z$ for $Z\sim N(0,1)$
    \item BM is continuous, but nowhere differentiable \\
    From above:
    \begin{align*}
        \frac{B(t+ \delta t) - B(t)}{\delta t} &= \frac{\sqrt{\delta t} \cdot Z}{\delta t},\ \text{for } Z \sim N(0,1) \\
        &= \frac 1{\sqrt{\delta t}} Z
    \end{align*}
\end{itemize}
\underline{Properties of BM} \\
$B(t),\ t \ge 0$ is a Markov process. \\
For $0 \le t_1 \le ... \le t_n \le t$, we have
\begin{align*}
    \mathbb{P}(B(t) \le b\ |\  B(t_1), \dots, B(t_n)) = \mathbb{P}(B(t) \le b\ |\ B(t_n))
\end{align*}
\boxed{ex}\ The following processes are BMs:
\begin{itemize}
    \item $\tilde B(t) = -B(t), t \ge 0$ 
    \item $\tilde B(t) = B(t + t_0) - B(t_0), t \ge 0$ 
    \item $\tilde B(t) = \frac{1}{\sqrt \lambda}B(\lambda t), \lambda > 0$  
    \item (e.g. check $\tilde B(t)$ satisfies 4 properties if $B(t)$ does)
\end{itemize}
\underline{Transition probability density}: $p(y,t\ |\ x,s)$ for $s \le t$
\begin{align*}
    \int_{a}^{b} p(y,t\ |\ x,s)dy &= \mathbb{P}(a < B(t) < b\ |\ B(s) = x) \\
    &= \mathbb{P}(a-x < B(t) - B(s) < b-x) \\
    &= \mathbb{P}(a < \sqrt{t\cdot s} \cdot Z + x < b) \text{ for } Z\sim N(0,1) \\
    \Rightarrow p(y,t\ |\ x,s) = \frac{1}{\sqrt{2\pi(t\cdot s)}} \exp(-\frac{(y-x)^2}{2(t-s)^2})
\end{align*}
$B(t), B(s)$ are not independent. 
\begin{align*}
    \text{Cov}(B(t), B(s)) &= \mathbb{E}(B(t)\cdot B(s)) - \mathbb{E}(B(t))\mathbb{E}(B(s)) \\
    &= \mathbb{E}(B(t)\cdot B(s)) - 0 \text{ because } B(t) \sim N(0,t) \\
    &= \mathbb{E}(B(t)B(s))
\end{align*}
Assume $t \ge s$, then 
\begin{align*}
    B(t)B(s) &= [B(t)-B(s)][B(s)-B(0)]+ B(s)^2 \\
    \mathbb{E}(B(t)B(s)) &= \mathbb{E}(B(t) - B(s)) \mathbb{E}(B(s)B(0)) + \mathbb{E}(B(s)^2) \\
    &= s  \\
    \Rightarrow \mathbb{E}(B(s)B(t)) = \min(s,t)
\end{align*}

\subsection{Lecture 20 - SDEs}
\underline{Recap}: \\
Definition of Brownian motion, $B(t) \in \mathbb{R}, t \ge 0$
\begin{enumerate}
    \item $B(0) = 0$, with probability 1
    \item $B(t) - B(s) \sim N(0,t-s)$ for $t \ge s$
    \item Independent increments  \\
    $\forall\ 0 \le s_1 < t_1 \le s_2 < t_2$ \\
    RVs $B(t_1)-B(s_1), B(t_2)-B(s_2)$ are independent.
    \item $B(t)$ is a continuous function of $t$ (but not differentiable).
\end{enumerate}
1. and 2. $\Rightarrow B(t) \sim N(0,t)$ \\
Discrete picture: $X_0,X_1,\dots \in \mathbb{R}, \delta t >0$
\begin{align*}
    X_{n+1} = X_n + \sqrt{\delta t}\cdot Z_n,\ n=0,1,2,\dots \\
    Z_0, Z_1, \dots \sim N(0,1)\ \text{iid}, X_0 = 0
\end{align*}
Continuum limit: $n\rightarrow\infty,\ \delta t \rightarrow 0, n\cdot \delta t = t$ fixed.
\begin{align*}
    X_n \rightarrow X(t),\ X(t) \text{ is a BM}
\end{align*}
\textbf{Simulating BM} (Euler-Maruyama)\\
Let $\delta t > 0$, $n \in \mathbb{Z}$, $T = n\delta t$ \\
Let $t_k = k\delta t,\ k = 0,1,2,\dots,n,\ t_n = T = n \delta t$ \\
\underline{Idea}: Find a discrete time process on $[0,T]$
\begin{align*}
    X(t_k) = X_k \text{ with } X_k \approx B(t_k),\ \text{for BM } B(t)
\end{align*}
\underline{Algorithm}: (Euler)
\begin{enumerate}
    \item Set $X_0 = 0$ 
    \item For $i = 0$ to $n-1$
    \item Generate $Z \sim N(0,1)$
    \item Set $X_{i+1} = X_i + \sqrt{\delta t} \cdot Z$
    \item Output: $X_0, X_1, \dots, X_n = X(0), X(t_1), \dots, X(t_n)$
\end{enumerate}
\underline{Remarks}:
\begin{itemize}
    \item Can interpolate to connect $(t_k, X(t_k))$ to get approximate sample path $X(t)$
    on $[0,T]$ with $X(t) \approx B(t)$, BM
    \item Return to convergence later.
\end{itemize}
\textbf{BM with Drift} \\
\begin{align*}
    X(t) = ut + \sqrt{2D} B(t)
\end{align*}
$\text{where } u,D \in \mathbb{R},\ D > 0$ where $B(t) \text{ is a BM}$
\begin{itemize}
    \item Distribution of $X(t)$ \\
    $X(t) \sim $ normal (constant + normal)
    \item $\mathbb{E}X(t) = ut$
    \item Var$X(t) = \mathbb{E}((X(t)-ut)^2)$ \\
    $= \mathbb{E}(2D)B(t)^2$ \\
    $= 2Dt(\mathbb{E}(B(t)^2) = t)$
    \item $\Rightarrow X(t) \sim N(ut, 2Dt)$
\end{itemize}
\underline{SDE notation}:
SDE whose solution is BM with drift \\
\begin{align*}
    \underbrace{dx}_{\text{change in X(t)}} = \underbrace{udt}_{\text{change in t}} + \sqrt{2D} \cdot \underbrace{dB}_{\text{infinitesimal Brownian increment}},\ X(0) = 0
\end{align*}
Here you can 'integrate' and get the original equation $X(t) = ut + \sqrt{2D} B(t)$. However,
stochastic integration is different in general. \\ \\
\underline{Euler's method}: \\
Let $\delta t > 0,\ n \in \mathbb{Z},\ t_k = k\delta t,\ k = 0,1,\dots, n$ \\
Set $X_0 = 0$\\
For $i = 0$ to $n-1$ 

Generate $Z\sim N(0,1)$

Set $X_{i+1} = X_i+u\delta t + \sqrt{2D} \sqrt{\delta t}Z$\\
End \\ \\ 
\underline{Output}: $X_0, X_1,\dots, X_n$ \\
$X_i$ approximate true solution $X(t_i)$ \\ \\ 
\underline{Remarks}:
\begin{itemize}
    \item SDE: $dx = udt + \sqrt{2D} dB$
    \item EM: $X_{i+1} =X_i + u\delta t + \sqrt{2D\delta t} Z_i$\\
    $Z_0, Z_1, \dots \sim N(0,1)\ $iid
    \item Can think of SDE solution $X(t)$ as 'continuum limit' of solution to discrete equation.
    \item Intuitively, $dB$ is like a $\sqrt{dt} Z$ \\
    $dB(t) = B(t+dt) - B(t) \sim N(0,dt)$ \\
    $\mathbb{E}(dB) = 0,\ \mathbb{E}(dB^2) = dt$
    \item Langevin notation (physics) \\
    If we divide by dt
    \begin{align*}
        \frac{dx}{dt} = u + \sqrt{2D} \underbrace{\xi(t)}_{\text{white noise } \frac{dB}{dt}}
    \end{align*}
    Note that $\xi(t)$ is not a function! \\
    From above $\frac{dB}{dt}\sim \frac1{\sqrt{dt}}z$
\end{itemize}
\boxed{\text{ex}} Moments of BM with drift
\begin{align*}
    dX &= udt + \sqrt{2D}dB \\
    \mathbb{E}(dX) &= udt \\
    d(\mathbb{E}X) &= udt\ (\text{ODE for } \mathbb{E}X) \\
    \frac{d}{dt}(\mathbb{E}X) &= u \\
    \mathbb{E}X &= ut 
\end{align*}
From the discrete picture:
\begin{align*}
    X_{i+1} - X_i &= u\delta t + \sqrt{2D\delta t} Z\\
    \mathbb{E} X_{i+1} - \mathbb{E}X_i &= u \delta t \\
    \frac{\mathbb{E}X_{i+1}-\mathbb{E}X_i}{\delta t} &= u \\
    \frac{d}{dt} \mathbb{E}X &= u \\
    \mathbb{E}X &= ut
\end{align*}
