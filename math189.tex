\section{Statistical Learning}
\subsection{Terminology}
"Supervised Learning"  means we have both predictors and responses. \\
"Predictors/input variables/features" are arranged in a vector.
\begin{align*}
    X = (X_1,X_2,...,X_n)
\end{align*}
"Output variables/responses" is a scalar where $y \in \mathbb{R}$ \\
We think of data as coming from a model.
\begin{align*}
    y = f(x) + \epsilon
\end{align*}
$\begin{cases}
    f(x) \text{ is fixed but unknown function} \\
    \epsilon \text{ is the random error term with mean zero, independent of X}
\end{cases}$ \\ \\
Question: Why estimate $f$? \\
Prediction: Given $X$, we want to predict $y$ with $\hat{y} = \hat{f}(x)$ \\ \\ 
Mean square error is 
\begin{align*}
    \mathbb{E}(y-\hat{y})^2 &= \mathbb{E}(f(x)+\epsilon - \hat{f}(x)) \\
    &= \mathbb{E}(f(x)-\hat{f}(x))^2 + 2\mathbb{E}((f(x) - \hat{f}(x))\epsilon) + \mathbb{E}(\epsilon^2)
\end{align*}
Examine the middle term in detail 
\begin{align*}
    2\mathbb{E}((f(x) - \hat{f}(x))\epsilon)
    = \mathbb{E}(f(x)-\hat{f}(x))\cdot\mathbb{E}(\epsilon)=0
\end{align*}
We know that 
\begin{align*}
    \mathbb{E}(y-\hat{y})^2 = (f(x)-\hat{f}(x))^2 + \mathbb{E}(\epsilon^2)
\end{align*}
Where the left term is reducible error and the right term is irreducible error. \\ \\ 
"Inference" - Sometimes we want to understand the association between $y$ and $X_1, ... ,X_p$ 
\begin{enumerate}
    \item Which predictors are associated with response?
    \item Is the association positive or negative?
    \item Is the relationship approximately linear?
\end{enumerate}

\subsection{Accuracy}
\underline{Training data} 
\begin{align*}
    (x_1, y_1),(x_2, y_2),...(x_n, y_n) \\ 
    \text{where } X_i = (X_{i1}, X_{i2}, ... , X_{ip})
\end{align*}
n = \# data points (e.g. $n=30$) \\
p = \# predictors (e.g. $p=3$) \\ \\
\noindent \underline{Mean Square Error (MSE)}
\begin{align*}
    \text{MSE} = \frac 1n \sum_{i=1}^{n} (y_i - \hat{f}(x_i))^2 \\
    \text{Defined on training data} \\ 
    \text{Test MSE} = \text{ Ave } (y_0 - \hat{f}(x_0))^2 \\ 
    \text{Defined on previously unseen data point } (x_0, y_0)
\end{align*}
\underline{Linear / affine model}
\begin{align*}
    f(x) = \beta_0 + \beta_1x_1 + ... + \beta_px_p
\end{align*}
"Fit/train the model" usually means run an optimization on your computer
that uses training data and outputs the parameters $\beta_0, \beta_1,...,\beta_p$ that make the 
MSE and hopefully the test MSE small. \\

\noindent \underline{Classification} \\ 
Data set $(x_1,y_1),(x_2,y_2,),..., (x_n,y_n)$ where $X_i=(x_{i1},...,x_{ip})$ and $y_i$ is qualitative. \\ \\ 
Error rate is the fraction of incorrect classifications. 
\begin{align*}
    \frac 1n \sum_{i=1}^{n} \mathbb{I} \{y_i \ne \hat{y}_i\}
\end{align*}
$\mathbb{I}\{ y_i \ne \hat{y}_i \} = \begin{cases}
    1, & y_i \ne \hat y_i \\
    0, & y_i = \hat y_i 
\end{cases}$ \\ \\
Test error rate 
\begin{align*}
    \text{Ave}(\mathbb{I} \{ y_0 \ne \hat y_0 \})
\end{align*}
Where $(x_0,y_0)$ is a previously unseen test data point. \\ \\
\underline{K-nearest neighbors (KNN)} \\ 
The KNN classifier is defined by letting $\eta_0$ = set of indices $i\in\{ 1,...,n \}$ for K nearest neighbors of $X_0$ among the input data. \\\\
$\hat y_0 = j$ where $j$ maximizes 
\begin{align*}
    \sum_{i\in \eta_0}^{} \mathbb{I} \{y_i=j\}
\end{align*}
which is the number of nearest neighbors (out of K) assigned label $j$.

\section{Linear Regression}
\subsection{Simple Linear Regression}
Underlying statistical model: $y=\beta_0 + \beta_1X + \epsilon$
where $y$ is the quantitative response and a single predictor $X$ which is why it is called simple. \\

\noindent Prediction model: $\hat y = \hat \beta_0 + \hat \beta_1X$ \\
$\hat \beta_0, \hat \beta_1$ are estimates based on training data, where hat means estimated quantity. \\

\noindent \underline{Estimating coefficients} \\
The residual sum of squares (RSS) 
\begin{align*}
    \text{RSS} &= \sum_{i=1}^{n}(y_i-\hat y_i)^2 \\
    &= \sum_{i=1}^{n}(y_i-\hat \beta_0 - \hat \beta_1x_i)^2 
\end{align*}
Choose $\hat \beta_0, \hat \beta_1$ to minimize RSS.
\begin{align*}
    \frac{\partial \text{RSS}}{\partial \hat\beta_0} &= \frac{\partial}{\partial \hat\beta_0}  \sum_{i=1}^{n}(y_i-\hat \beta_0 - \hat \beta_1X)^2  \\
    &= 2\sum_{i=1}^{n}(\hat \beta_0+\hat \beta_1X_i-Y_i) \\
    &\implies \text{RSS is } \begin{cases}
        \text{decreasing for really small } \hat \beta_0 \\
        \text{increasing for really big } \hat \beta_0 
    \end{cases} 
\end{align*}
RSS is optimal (in $\hat \beta_0$) when 
\begin{align*}
    0 &= \frac 1n \sum_{i=1}^{n} (\hat \beta_0 + \hat \beta_1 X_i - Y_i) \\
    &= \hat \beta_0 + \hat \beta_1 \Big(\frac 1n \sum_{i=1}^{n}X_i\Big) - \Big(\frac 1n \sum_{i=1}^{n}Y_i\Big) \\ 
    &= \hat \beta_0 +\hat \beta \bar x - \bar y \\
    \hat \beta_0 &=\bar y - \hat \beta_1 \bar x 
\end{align*}
Where $\begin{cases}
    \bar x = \frac 1n \sum_{i=1}^{n}X_i \text{ is the mean predictor} \\
    \bar y = \frac 1n \sum_{i=1}^{n}Y_i \text{ is the mean response}
\end{cases}$ \\  \\
\underline{Covariance}
\begin{align*}
    \text{Cov}  (X_i, Y_i)^n_{i=1} &= \frac 1n \sum_{i=1}^{n}X_iY_i - \bar X \bar Y \\
    &= \frac 1n \sum_{i=1}^{n} (X_i - \bar x) (Y_i - \bar y)
\end{align*}
\underline{Variance}
    \begin{align*}
        \text{Var}(X_i)^n_{i=1} &= \text{Cov} (X_i,X_i)_{i=1}^n \\
        &= \frac 1n \sum_{i=1}^{n}(X_i - \bar x)^2
    \end{align*}
From this we can get our optimal $\hat \beta_1$
\begin{align*}
    \hat \beta_1 &= \frac{\text{Cov}(X_i,Y_i)_{i=1}^n}{\text{Var}(X_i)_{i=1}^n} \\ 
    &= \frac{\frac 1n \sum_{i=1}^{n}(X_i-\bar x)(Y_i-\bar y)}{\frac 1n \sum_{i=1}^{n} (X_i-\bar x)^2}
\end{align*}

\subsection{Goodness of Fit \& Correlation}
Assume an underlying statistical model
\begin{align*}
    y = \beta_0 + \beta_1X +\epsilon
\end{align*}
We use training data $(X_i, Y_i)^n_{i=1}$ to generate \underline{prediction model} $\hat y = \hat \beta_0 + \hat \beta_1 X$

\noindent The formulas for $\hat \beta_0$ and $\hat \beta_1$ to minimize RSS:
\begin{align*}
    \text{RSS} = \sum_{i=1}^{n}(y_i-\hat y_i)^2 
    \begin{cases}
        \hat \beta_0 = \bar y - \hat \beta_1 \bar X \\ 
        \hat \beta_1 = \frac{\text{Cov}(X_i,Y_i)^n_{i=1}}{\text{Var}(X_i)^n_{i=1}}
    \end{cases}
\end{align*}
\underline{Goodness of fit}
\begin{align*}
    \text{RSS} &= \sum_{i=1}^{n}(y_i - \hat y_i)^2 \\
    &= \sum_{i=1}^{n}(y_i-\hat \beta_0 - \hat \beta_1x_i)^2 \\
    &= \sum_{i=1}^{n}\Big( (y_i-\bar y )-\hat \beta_1(x_i-\bar x)\Big)^2 \\
    &= \sum_{i=1}^{n}(y_i - \bar y)^2 - 2\hat \beta_1 \sum_{i=1}^{n}(x_i - \bar x)(y_i - \bar y) + \hat \beta_1^2 \sum_{i=1}^{n}(x_i - \bar x)^2 \\
    &= \sum_{i=1}^{n}(y_i - \bar y)^2 - \frac{\Big[ \sum_{i=1}^{n} (x_i - \bar x)(y_i - \bar y)\Big]^2}{\sum_{i=1}^{n} (x_i - \bar x)^2}
\end{align*}
RSS scales with the number of data points $n$ and also scales with $y^2$ units.
$\rightarrow$ We can fix the scaling and get out a goodness-of-fit score ($R^2$) between 0 and 1.

\begin{align*}
    R^2 = 1 - \frac{\text{RSS}}{\text{TSS}}
\end{align*}
\begin{align*}
    \text{RSS} = \sum_{i=1}^{n}(y_i - \hat y_i)^2 & & \text{TSS} = \sum_{i=1}^{n}(y_i - \bar y )^2
\end{align*}
\newpage
\noindent So for simple linear regression,
\begin{align*}
    R^2 &= 1 - \frac{\text{RSS}}{\text{TSS}} \\
    &= \frac{\text{TSS} - \text{RSS}}{\text{TSS}} \\ 
    &= \frac{\Big[ \sum_{i=1}^{n}(x_i - \bar x) (y_i - \bar y) \Big]^2}{\sum_{i=1}^{n}(x_i - \bar x)^2 \sum_{i=1}^{n}(y_i - \bar y)^2} \\
    R & = \frac{\text{Cov}((x_i),(y_i))}{\text{Var}(x_i)^{1/2}\text{Var}(y_i)^{1/2}}
\end{align*}
Long story short $R^2$ is the fraction of variance explained by the model. \\ \\
\boxed{ex} \quad $\hat{\text{sales}} = 7.0 + 0.048 \cdot\text{TV}$ \\
$R^2 = 0.61 = 61\%$ \qquad $r \approx 0.78$ pretty strong positive correlation

    \beta_1 \approx \hat \beta_1 \pm \text{SE}(\hat\beta_1)

\begin{center}
    $y_i = \beta_0 + \beta_1X_i + \epsilon_i$ \\
    $\text{where } \epsilon_i \text{ are independent }\eta(0,\sigma^2) \text{ random variables}$
\end{center}

$\begin{cases}

    \text{Var}(\hat \beta_1)^{1/2} \approx \hat{\text{Std. error}} (\hat \beta_1) \\ 
    \mathbb{E}(\hat \beta_0) = \beta_0 \\
    \mathbb{E}(\hat \beta_1) = \beta_1
\end{cases}$

\subsection{Multiple Linear Regression}
The underlying statistical model that we assume is true.
\begin{align*}
    y = \beta_0 + \beta_1x_1 + \beta_2x_2 + ... + \beta_px_p
\end{align*}
where $y$ is response, $\beta_0$ is the intercept, $\beta_1x_1,...,\beta_px_p$ are predictors, and $\epsilon$ is the mean zero noise. \\

\noindent \boxed{ex} \quad $\text{sales} = \beta_0 + \beta_1 \text{TV} + \beta_2 \text{Radio} + \beta_3 \text{Newspaper} + \epsilon$

\begin{align*}
    \hat y = \hat \beta_0 + \hat \beta_1 x_1 + ... + \hat \beta_p x_p
\end{align*}
We want to find the best $\hat \beta$ to minimize RSS, where $\hat \beta$ are estimated quantities.
\begin{align*}
    RSS &= \sum_{i=1}^{n}(y_i - \hat y_i)^2 \\ 
    &= \sum_{i=1}^{n} (y_i - \hat \beta_0 - \hat\beta_1x_{i1} - ... - \hat \beta x_{ip})^2 \\
    &= \Bigg|\Bigg|\begin{bmatrix}
        y_1 - \hat \beta_0 -\hat \beta_1 x_{i1} - ... - \hat \beta_p x_{ip}\\
        \vdots \\
        y_n - \hat \beta_0 - \hat \beta_1x_{n1} -... - \hat \beta_p x_{ip}
    \end{bmatrix} \Bigg|\Bigg| ^2 \\ 
    &= || Y - X\hat\beta ||^2
\end{align*}
where $y = \begin{bmatrix}
    y_1 \\ y_2 \\ \vdots \\ y_n
\end{bmatrix}$ \qquad 
$\hat \beta = \begin{bmatrix}
    \hat \beta_0 \\ \hat \beta_1 \\ \vdots \\ \hat \beta_p
\end{bmatrix}$ \qquad
$X = \begin{bmatrix}
1 & x_{11} & \dots & x_{1p} \\
1 & x_{21} & \dots & x_{1p} \\
& & \ddots \\
1 & x_{n1} & \dots & x_{np}
\end{bmatrix}$

\noindent The best $\hat \beta$ to minimize Euclidean norm$^2$ for $|| Y - X\hat\beta ||^2$ is $\hat \beta = X^+Y = (X^TX)^{-1}X^TY$

\begin{tabular}{c | c c c c}
    & Coefficient & Std. error & t-statistic & p-value\\
    Intercept & $\hat \beta_0$ 2.939 & 0.3119 & 9.42 & $<$0.0001\\ 
    TV & $\hat\beta_1$ 0.046 & 0.0014 & 32.81 & $<$0.0001\\ 
    Radio & $\hat\beta_2$ 0.189 & 0.0086 & 32.89 & $<$0.0001\\
    Newspaper & $\hat \beta_3$ -0.001 & 0.0059 & -0.18 & 0.8599
\end{tabular} \\ \\ \\ 
Note: the standard error is NOT $Var(\hat\beta_i)^2$ but rather $\hat{Var}(\hat\beta_i)^2$ \\

\noindent \underline{Hypothesis testing} \\

\noindent State a hypothesis, test it with data (testing the likelihood such data could possibly be generated if the hypothesis were true). \\

$H_0$ - the null hypothesis: $y = \beta_0 + \beta_1x_1 + \beta_2x_2 + \beta_3x_3 + \epsilon$ for some coefficients $B_0, \beta_2, \beta_3$ with $\beta_1 = 0$ 
with $\epsilon \sim N(0,\sigma^2)$ for some variance $\sigma^2 \ge 0$. Also, observations are independent.

$H_1$ - the alternative hypothesis: $y = \beta_0 + \beta_1x_1 + \beta_2x_2 + \beta_3x_3+\epsilon$ where everything above holds except $\beta_1 \ne 0$ \\ 

Assume $H_0$. Then \begin{align*}
    \frac{\hat \beta_1}{\hat{SE}(\hat \beta_1)} = t \sim T_{n-p-1}
\end{align*}

\section{Logistic Regression \& Generative Models}
\subsection{Logistic Regression}
We want to predict whether an event happens $(y=1)$ or not $(y=0)$, we 
really want to model the probability $p\in(0,1)$ of occurrence $\implies$
NOT LINEAR REGRESSION.\\
\underline{Before}
\begin{align*}
    y= \beta_0 + \beta_1X_ \epsilon
\end{align*}
\underline{Now}
\begin{align*}
    p &= \frac{\exp(\beta_0 + \beta_1X)}{1+\exp(\beta_0+\beta_1X)} \\
    1-p &= \frac{1}{1+\exp(\beta_0+\beta_1X)}\\
    \frac{p}{1-p} &= \exp(\beta_0+\beta_1X)
\end{align*} \\
Where the term $\frac{p}{1-p}$ is called "odds". \\
Note: $\log(\text{odds}) = \beta_0+\beta_1X$ \\

\noindent \boxed{ex} Defaulting on bank loans
\begin{align*}
    p &= \text{probability of defaulting} \\
    X &= \text{balance} \\ 
    y_i &= \begin{cases}
        0 & \text{default doesn't happen} \\
        1 & \text{default does happen}
    \end{cases} \\
    \log(\frac{p}{1-p}) &= \beta_0 + \beta_1 \cdot \text{X} 
\end{align*}

How do we use data/observations $(x_i,y_i)_{1\le i \le n}$ to bestimate $\beta_0, \beta_1$.

$\begin{cases}
    \text{Linear regression} \implies \text{minimize RSS}\\
    \text{Logistic regression} \implies \text{use max likelihood, do it on a computer} \\ 
    \text{Generative models} \implies \text{simple formulas for } \hat \Pi_k, \hat \mu_k, \hat \Sigma_k
\end{cases}$ \\

\noindent We choose estimates $\hat \beta_0, \hat \beta_1$ to maximize likelihood of observations $(x_i,y_i)_{1\le i \le n}$
if (a) the observations were independent and (b) they were generated exactly from the statistical model.

\begin{align*}
    \mathbb{P}\left((x_i,y_i)_{1\le i \le n}\right) &= \prod_{i=1}^{n} \mathbb{P}(x_i,y_i) \text{ where }y_i \in \{0,1\} \\
    &= \prod_{i:y_i=1} \frac{\exp(\beta_0+\beta_1X_i)}{1+\exp(\beta_0+\beta_1X_i)}\prod_{i:y_i=0}\frac{1}{1+\exp{\beta_0+\beta_1X_i}}
\end{align*}
Know that $\hat\beta_0, \hat\beta_1$ are chosen to maximize this probability/likelihood above. \\

\noindent \boxed{ex} $P_{\text{defaulting}}$ \\ \\
\begin{tabular}{l | c c c c}
              &  Coefficient & Std. error & z-statistic & p-value \\
              \hline
    Intercept & -10.6513 & 0.3612 & -29.5 & $<$0.0001 \\
    Balance   & .0055 & 0.0002 & 24.9 & $<$0.0001
\end{tabular} \\ \\ 

We want $\hat \beta_0 + \hat \beta_1X = 0$
\begin{align*}
    -10.6513 + 0.0055X &= 0 \\
    X &= \frac{10.6513}{0.0055} \\ 
    &= 1936.6
\end{align*}
Assume \begin{enumerate}[a.]
    \item The model is true 
    \item Observations $(x_i,y_i)_{1\le i\le n}$ are independent draws from the model
    \item Model is true with $\beta_0 = 0$ (null hypothesis)
\end{enumerate}


\subsection{Generative Models for Classification}
\begin{align*}
    &\text{Predictors} \\
    &X \text{ or } X_1,X_2,...,X_n \\
    &\text{Response} \\ 
    &y \in \{0,1\} \text{ or } y \in \{1,2,...,k\} 
\end{align*}

\noindent \underline{Review} \\
Simple logistic regression
\begin{align*}
    P = \text{prob}\{y=1\} = \frac{\exp(\beta_0+\beta_1X)}{1+\exp(\beta_0+\beta_1X)}
\end{align*}
Note: This doesn't make any assumptions about how predictors are distributed. \\ 

\noindent For each $k = 1,2,...,K$: \\
Our statistical model ('mixture of Gaussians') says $y=k$ with probability $\pi_k$ $(\sum_{k=1}^{K}\pi_k = 1)$

\noindent Given $y=k$, $X \sim N(\mu_k, \Sigma_k)$ where $\Sigma$ is the covariance matrix symbol. Here is the general vector-valued format: 
\begin{align*}
    X = (X_1,...,X-p) & & \Sigma_k = \begin{bmatrix}
    & &\\
    & &
    \end{bmatrix} \text{pxp matrix}
\end{align*}
Where $\mu_k$ has $Kp$ parameters and $\Sigma_k$ has $K\frac{p(p+1)}{2}$ parameters \\
\noindent When $p=1$, we can write 
\begin{align*}
    X\sim N (\mu_k, \sigma^2_k)
\end{align*}
In approaches like Linear Discriminant Analysis (LDA), Quadratic Discriminant
Analysis (QDA), and Naive Bayes (NB) use simple estimators for $\Pi_k, \mu_k, \Sigma_k$
that we can calculate fast. \\

\noindent \boxed{ex} Linear discriminant analysis (LDA) make assumptions of common covariance structure
$\Sigma = \Sigma_1 = \Sigma_2 = ... = \Sigma_k$ which reduces a lot of complexity.

\noindent Given observations $(X_i, y_i)_{1\le i \le n}$, we can estimate for $k = 1,...,K$
\begin{align*}
    \hat \Sigma &= \frac{n_k}{n}, \text{ where} \\
    n &= \# \{i:y_i=k\}\\ 
    &= \text{number of observations in class } k \\
    \hat \mu_k &= \frac{1}{n_k}\sum_{i:y_i=k}X_i
\end{align*}
Last estimate common covariance by 
\begin{align*}
    \hat \Sigma = \frac{1}{n-k} \sum_{k=1}^{K} \sum_{i:y_i=k} (X_i-\mu_k)(X_i-\mu_k)^T \\ 
    \mathbb{E} \hat \Sigma = \Sigma \text{ if the model is true}
\end{align*}

\noindent \boxed{ex} Quadratic discriminant analysis is pretty much the same except DIFFERENT covariances $\Sigma_1,...,\Sigma_k$
for each class. \\
$\implies \hat \Pi_k = \frac{n_k}{n}$, $\hat \mu_k = \frac{1}{n_k}\sum_{i:y_i=k} X_i$ \\
$\implies \hat \Sigma_k = \frac{1}{n_k-1} \sum_{i:y_i=k}(X_i-\mu_k)(X_i-\mu_k)^T$ \\
$\implies K\frac{p(p+1)}{2}$ covariance free parameters.

\noindent \boxed{ex} Naive Bayes is pretty much the same except DIFFERENT covariances $\Sigma_1, ..., \Sigma_k$
that are also diagonal matrices.\\
$\implies Kp$ covariance free parameters with the same estimators as QDA.
\begin{align*}
    \text{diag}\left(\sum_{k}^{n}\right) = \frac{1}{n_k-1} \sum_{i:y_i=k} (X_i-\mu_k)\odot(X_i-\mu_k)
\end{align*} \\

\noindent Suppose you have the statistical model with probability $\Pi_k$, we set $y=k$ and draw
$X\sim N(\mu_k,\Sigma_k)$. Suppose we have estimated the parameters well - see a new 
predictor vector $X = (X_1,...,X_p)$ - How do you determine class probabilities $P\{y= k |X\}$ and make predictions?\\

\noindent Prediction part is easy - just predict based on the highest probability class:
$\hat y = k$ for $k= \text{argmax}_{1\le j\le k}\  P\{y=j |X\}$ \\ 

\noindent So what is $P\{y=k |X\}$? Bayes' rule tells us 
\begin{align*}
    P\{y=k|X\} &= \frac{P\{x|y=k\}\ P\{y=k\}}{\sum_{j=1}^{k}P\{X|y=j\}P\{y=j\}} \\
    &= P(X) \\ 
    &= \frac{f_k(X)\Pi_k}{\sum_{j=1}^{K}f_j(X)\Pi_j} \\
    \text{ where Gaussian density } \Pi_j(X) &= \frac{1}{(2\pi)^{p/2}(\det\Sigma)^{1/2}}\exp(\frac{-1}{2}(X-\mu_j)\Sigma_j(X-\mu_j))
\end{align*}
